# 0037. 解数独

**题目描述**

编写一个程序，通过填充空格来解决数独问题。数独的解法需遵循如下规则：数字 1-9 在每一行只能出现一次；数字 1-9 在每一列只能出现一次；数字 1-9 在每一个以粗实线分隔的 3x3 宫内只能出现一次。数独部分空格内已填入了数字，空白格用 `'.'` 表示，需要填入合适的数字使整个数独有唯一解。

```
示例：
输入：一个 9x9 的二维字符数组 board，部分单元格是数字，其余是 '.'
输出：直接在 board 上完成填数（原地修改），使其成为一个解好的数独
```

## 思路

回溯：找到空格，尝试 1～9，同行、同列、同九宫格无冲突才填入并递归。失败则清空；某空格全部尝试失败立即返回，棋盘无空格即成功。

搜索时间上界 O(9ᵐ)，栈空间 O(m)，m 为空格数；固定 9×9 棋盘下空间 O(1)。

```python
# 依赖说明：range：Python 内置函数：生成按步长遍历的整数序列。

class Solution:
    # 方法：solveSudoku；按题目要求处理输入并返回结果，核心算法见方法体。
    def solveSudoku(self, board: list[list[str]]) -> None:
        # 方法：is_valid；按题目要求处理输入并返回结果，核心算法见方法体。
        def is_valid(row: int, col: int, val: str) -> bool:
            for i in range(9):
                if board[row][i] == val:        # 同行冲突
                    return False
                if board[i][col] == val:        # 同列冲突
                    return False
            box_row, box_col = (row // 3) * 3, (col // 3) * 3   # 所在宫格起始坐标
            for r in range(box_row, box_row + 3):
                for c in range(box_col, box_col + 3):
                    if board[r][c] == val:        # 同宫格冲突
                        return False
            return True

        # 方法：backtrack；按题目要求处理输入并返回结果，核心算法见方法体。
        def backtrack() -> bool:
            for row in range(9):
                for col in range(9):
                    if board[row][col] != '.':      # 已有数字，跳过
                        continue
                    for val in '123456789':            # 尝试填入 1-9
                        if not is_valid(row, col, val):
                            continue
                        board[row][col] = val    # 【选择】
                        if backtrack():             # 【递归】
                            return True                   # 已找到完整解，一路向上返回 True
                        board[row][col] = '.'      # 【撤销选择】
                    return False    # 1-9 都不行，当前分支无解
            return True    # 没有空格了，棋盘已填满

        backtrack()
```

// 方法：solveSudoku；按题目要求处理输入并返回结果，核心算法见方法体。
func solveSudoku(board [][]byte) {
    var isValid func(row, col int, val byte) bool
    isValid = func(row, col int, val byte) bool {
        for i := 0; i < 9; i++ {
            if board[row][i] == val {     // 同行冲突
                return false
            }
            if board[i][col] == val {     // 同列冲突
                return false
            }
        }
        boxRow, boxCol := (row/3)*3, (col/3)*3   // 所在宫格起始坐标
        for r := boxRow; r < boxRow+3; r++ {
            for c := boxCol; c < boxCol+3; c++ {
                if board[r][c] == val {     // 同宫格冲突
                    return false
                }
            }
        }
        return true
    }

    var backtrack func() bool
    backtrack = func() bool {
        for row := 0; row < 9; row++ {
            for col := 0; col < 9; col++ {
                if board[row][col] != '.' {    // 已有数字，跳过
                    continue
                }
                for val := byte('1'); val <= '9'; val++ {   // 尝试填入 1-9
                    if !isValid(row, col, val) {
                        continue
                    }
                    board[row][col] = val    // 【选择】
                    if backtrack() {           // 【递归】
                        return true                 // 已找到完整解，一路向上返回 true
                    }
                    board[row][col] = '.'      // 【撤销选择】
                }
                return false   // 1-9 都不行，当前分支无解
            }
        }
        return true   // 没有空格了，棋盘已填满
    }
    backtrack()
}

```cpp
// 依赖说明：vector：C++ STL 动态数组。

class Solution {
private:
    // 方法：isValid；按题目要求处理输入并返回结果，核心算法见方法体。
    bool isValid(vector<vector<char>>& board, int row, int col, char val) {
        for (int i = 0; i < 9; i++) {
            if (board[row][i] == val) return false;       // 同行冲突
            if (board[i][col] == val) return false;       // 同列冲突
        }
        int boxRow = (row / 3) * 3, boxCol = (col / 3) * 3;  // 所在 3x3 宫格起始坐标
        for (int r = boxRow; r < boxRow + 3; r++) {
            for (int c = boxCol; c < boxCol + 3; c++) {
                if (board[r][c] == val) return false;      // 同宫格冲突
            }
        }
        return true;
    }

    // 方法：backtrack；按题目要求处理输入并返回结果，核心算法见方法体。
    bool backtrack(vector<vector<char>>& board) {
        for (int row = 0; row < 9; row++) {
            for (int col = 0; col < 9; col++) {
                if (board[row][col] != '.') continue;   // 已有数字，跳过，处理下一个格子
                for (char val = '1'; val <= '9'; val++) {  // 横向遍历：尝试填入 1-9
                    if (!isValid(board, row, col, val)) continue;  // 冲突则跳过
                    board[row][col] = val;              // 【选择】填入数字
                    if (backtrack(board)) {             // 【递归】处理后续格子
                        return true;                        // 已找到完整解，一路向上返回 true，不再撤销
                    }
                    board[row][col] = '.';                // 【撤销选择】此数字走不通，恢复空格
                }
                return false;   // 1-9 都试过仍不行，说明当前分支无解，返回 false 让上一层换数字
            }
        }
        return true;   // 遍历完所有格子都没有再遇到空格，说明整个棋盘已经填满，成功
    }

public:
    // 方法：solveSudoku；按题目要求处理输入并返回结果，核心算法见方法体。
    void solveSudoku(vector<vector<char>>& board) {
        backtrack(board);   // 原地修改 board，题目不要求返回值
    }
};```

## 总结

- 数独是二维回溯（需要同时定位行 `row` 和列 `col` 两个维度），跟 N 皇后的一维行递归不一样，本质是因为每行可能有多个空格需要处理，不存在"每行只决策一次"的天然结构。
- 判断合法性需要同时检查行、列、3x3 宫格三个约束，宫格起始坐标用 `(row/3)*3` 和 `(col/3)*3` 计算，这是数独问题特有的第三重约束。
- 终止条件的设计是这题的核心难点：`backtrack` 函数必须返回布尔值表示"是否已经找到完整解"，找到解后要让 `true` 一路沿递归调用链向上传递直到最外层，中途不能撤销已经填好的数字，不然会破坏已经找到的正确解。
- 跟 0332 重新安排行程很相似，这题也是"只需要找到一个解就终止整个搜索"的回溯场景，这类题目的通用处理方式是让回溯函数带返回值，通过返回值的传递控制是否继续尝试其他分支。

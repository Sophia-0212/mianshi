# 0017. 电话号码的字母组合

**题目描述**

给定一个仅包含数字 `2-9` 的字符串，返回所有它能表示的字母组合，按九宫格键盘的映射关系（`2:abc`, `3:def`, `4:ghi`, `5:jkl`, `6:mno`, `7:pqrs`, `8:tuv`, `9:wxyz`）。答案可以按任意顺序返回。

```
示例：
输入：digits = "23"
输出：["ad","ae","af","bd","be","bf","cd","ce","cf"]

输入：digits = ""
输出：[]
```

## 思路

这题也是回溯法，但和 0077 题"树的宽度固定为 n"不一样，这里**树的宽度是随层数变化的**，这个区别得先区分清楚。

**递归树的结构**：树的深度是 `digits` 的长度（每一位数字对应一层），树的第 `i` 层的宽度是 `digits[i]` 这个数字在九宫格上对应的字母个数（比如 `7` 对应 `pqrs` 共 4 个字母，宽度就是 4；而 `2` 对应 `abc` 只有 3 个字母，宽度就是 3）。也就是说，这道题不再是"固定候选集合 1~n"了，而是"每一层的候选集合都不一样"，候选集合是由当前处理到的数字决定的。

**横向遍历与纵向递归**：

- **横向遍历（for 循环）**：在处理第 `index` 位数字时，遍历这个数字对应的字母表（比如 `digits[index]='2'` 就遍历 `"abc"` 这 3 个字母）。
- **纵向递归（backtrack 调用）**：选定一个字母后，递归处理 `digits` 的下一位（`index+1`），继续在下一位对应的字母表中做选择。

这题不需要 `startIndex` 参数，为什么呢？跟组合问题不一样，这里不需要避免重复选择同一个元素，因为每一位数字对应的字母表是独立的，不存在"同一个字母被选两次"的顾虑。参数用的是"当前处理到 digits 的第几位"，我们记作 `index`，它起到和 `startIndex` 类似的"往下走一层"的作用，但语义不太一样：这里的 `index` 是必选的推进，而不是"避免重复"的下界控制。

**边界情况**：`digits` 为空字符串的时候，直接返回空列表就行（树都不存在，不需要做任何递归）。

**复杂度分析**：设 `digits` 长度为 `m`，每一位平均对应 `t` 个字母（大部分数字对应 3 个字母，7 和 9 对应 4 个），时间复杂度是 O(t^m × m)（每个叶子节点需要 O(m) 时间拼接、拷贝字符串），空间复杂度是 O(m)（递归深度）。

## 代码

### C++

```cpp
class Solution {
private:
    // 数字到字母的映射表，下标 0-1 不使用（题目保证不出现 0、1）
    vector<string> letterMap = {
        "", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"
    };
    vector<string> result;
    string path;

    void backtrack(const string& digits, int index) {
        if (index == digits.size()) {          // 终止条件：每一位数字都已处理完
            result.push_back(path);            // 收集当前拼出的字母组合
            return;
        }
        int digit = digits[index] - '0';        // 取出当前处理的数字
        const string& letters = letterMap[digit]; // 该数字对应的字母表（本层候选集合）
        for (char c : letters) {                 // 横向遍历：本层的每个候选字母
            path.push_back(c);                   // 【选择】把字母加入路径
            backtrack(digits, index + 1);        // 【递归】处理下一位数字
            path.pop_back();                      // 【撤销选择】回退，尝试下一个字母
        }
    }

public:
    vector<string> letterCombinations(string digits) {
        if (digits.empty()) return result;   // 边界：空字符串直接返回空结果
        backtrack(digits, 0);
        return result;
    }
};
```

### Java

```java
class Solution {
    private String[] letterMap = {
        "", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"
    };
    private List<String> result = new ArrayList<>();
    private StringBuilder path = new StringBuilder();

    public List<String> letterCombinations(String digits) {
        if (digits == null || digits.length() == 0) return result;  // 边界：空字符串
        backtrack(digits, 0);
        return result;
    }

    private void backtrack(String digits, int index) {
        if (index == digits.length()) {          // 终止条件：每一位都已处理完
            result.add(path.toString());          // 收集当前拼出的字母组合
            return;
        }
        int digit = digits.charAt(index) - '0';   // 当前数字
        String letters = letterMap[digit];         // 本层候选字母表
        for (char c : letters.toCharArray()) {     // 横向遍历
            path.append(c);                         // 【选择】
            backtrack(digits, index + 1);           // 【递归】
            path.deleteCharAt(path.length() - 1);   // 【撤销选择】
        }
    }
}
```

### Python

标准库无直接对应函数可以一行解出九宫格映射结果（`itertools.product` 可以在拿到每位对应的字母表后做笛卡尔积，勉强算"库解"，但本质仍需手动构建字母表，这里只做简要提及）：

```python
from itertools import product

class Solution:
    def letterCombinations(self, digits: str) -> list[str]:
        if not digits:
            return []
        letter_map = {'2':'abc','3':'def','4':'ghi','5':'jkl','6':'mno','7':'pqrs','8':'tuv','9':'wxyz'}
        pools = [letter_map[d] for d in digits]      # 每一位对应的候选字母表
        return [''.join(p) for p in product(*pools)]  # 笛卡尔积，仅作对比，非回溯写法
```

手写回溯实现（还原递归树的逐层选择过程）：

```python
class Solution:
    def letterCombinations(self, digits: str) -> list[str]:
        if not digits:
            return []  # 边界：空字符串直接返回空结果

        letter_map = {
            '2': 'abc', '3': 'def', '4': 'ghi', '5': 'jkl',
            '6': 'mno', '7': 'pqrs', '8': 'tuv', '9': 'wxyz'
        }
        result = []
        path = []

        def backtrack(index: int):
            if index == len(digits):            # 终止条件：每一位数字都已处理完
                result.append(''.join(path))     # 收集拼出的字母组合
                return
            letters = letter_map[digits[index]]  # 当前数字对应的候选字母表
            for c in letters:                    # 横向遍历本层候选字母
                path.append(c)        # 【选择】
                backtrack(index + 1)  # 【递归】处理下一位数字
                path.pop()             # 【撤销选择】

        backtrack(0)
        return result
```

### Go

```go
func letterCombinations(digits string) []string {
    if len(digits) == 0 {
        return []string{}  // 边界：空字符串直接返回空结果
    }
    letterMap := map[byte]string{
        '2': "abc", '3': "def", '4': "ghi", '5': "jkl",
        '6': "mno", '7': "pqrs", '8': "tuv", '9': "wxyz",
    }
    var result []string
    path := make([]byte, 0, len(digits))

    var backtrack func(index int)
    backtrack = func(index int) {
        if index == len(digits) {              // 终止条件：每一位都已处理完
            result = append(result, string(path))
            return
        }
        letters := letterMap[digits[index]]     // 当前数字对应的候选字母表
        for i := 0; i < len(letters); i++ {     // 横向遍历本层候选字母
            path = append(path, letters[i])     // 【选择】
            backtrack(index + 1)                 // 【递归】
            path = path[:len(path)-1]            // 【撤销选择】
        }
    }
    backtrack(0)
    return result
}
```

## 总结

- 这题的关键区别在于"树的宽度不固定"：每一层的候选集合是由当前数字对应的字母表决定的，而不是像组合问题一样有一个固定的候选区间。
- 不需要 `startIndex` 去重，为什么呢？因为每一位处理的是不同位置的数字，候选字母表天然就不重叠，参数 `index` 只起到"往下推进一层"的作用。
- 递归深度等于 `digits` 的长度，每层的分支数等于该位数字对应的字母个数，这种"多叉树"结构在字符串拼接类回溯题目里很常见。
- 空字符串是一个明确的边界条件，必须在进入递归前单独处理，不然会返回一个包含空字符串的错误结果。

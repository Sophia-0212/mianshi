# 0332. 重新安排行程

**题目描述**

给定一个机票的字符串二维数组 `tickets`，其中 `tickets[i] = [fromi, toi]` 表示一张机票的起飞机场和降落机场。请你重新规划路线，使得从 `JFK` 出发能够刚好用掉所有机票。如果存在多种有效的行程，请返回按字典序排列最小的行程。假设所有机票至少存在一种合理的行程，且所有机票必须都用一次且只能用一次。

```
示例：
输入: tickets = [["MUC","LHR"],["JFK","MUC"],["LHR","SFO"]]
输出: ["JFK","MUC","LHR","SFO"]

输入: tickets = [["JFK","SFO"],["JFK","ATL"],["SFO","ATL"],["ATL","JFK"],["ATL","SFO"]]
输出: ["JFK","ATL","JFK","SFO","ATL","SFO"]
```

## 思路

- 排序＋回溯：从 `JFK` 出发，按字典序尝试未使用的机票；使用后标记，失败则撤销。路径含 `E+1` 个机场时立即返回，首个完整解即字典序最小。当前扫描邻接表的保守时间上界 O(E·E!)，空间 O(E)。
- 欧拉路径：按字典序消耗出边，走到无出边时记录机场，最后逆序（Hierholzer）。排序后时间 O(E log E)，空间 O(E)。

E 为机票数，重复机票须分别计数。

from collections import defaultdict

# 依赖说明：Python 标准库 collections：deque 是双端队列，Counter 是计数器，defaultdict 是带默认值的字典。；len：Python 内置函数：返回容器中元素的数量。；enumerate：Python 内置函数：遍历序列时同时得到下标和值。

class Solution:
    # 方法：findItinerary；按题目要求处理输入并返回结果，核心算法见方法体。
    def findItinerary(self, tickets: list[list[str]]) -> list[str]:
        targets = defaultdict(list)   # 机场 -> [目的地, ...]（未排序前先收集）
        for src, dst in tickets:
            targets[src].append(dst)
        for airport in targets:
            targets[airport].sort()    # 贪心排序：目的地按字典序排列，保证优先尝试小的

        used = {airport: [False] * len(dsts) for airport, dsts in targets.items()}
        path = ["JFK"]
        ticket_count = len(tickets)

        # 方法：backtrack；按题目要求处理输入并返回结果，核心算法见方法体。
        def backtrack(cur: str) -> bool:
            if len(path) == ticket_count + 1:   # 终止条件：用完所有机票
                return True                          # 找到答案，立即返回 True
            if cur not in targets:
                return False                          # 无出边，此分支失败
            for i, dst in enumerate(targets[cur]):    # 遍历已排序的目的地（贪心）
                if used[cur][i]:                        # 该机票已使用，跳过
                    continue
                used[cur][i] = True         # 【选择】标记已使用
                path.append(dst)             # 【选择】加入路径
                if backtrack(dst):            # 【递归】
                    return True                   # 已找到答案，直接向上传递 True，不再回溯
                path.pop()                     # 【撤销选择】
                used[cur][i] = False            # 【撤销选择】
            return False

        backtrack("JFK")
        return path```

```go
func findItinerary(tickets [][]string) []string {
    targets := make(map[string][]string)   // 机场 -> 目的地列表
    for _, t := range tickets {
        targets[t[0]] = append(targets[t[0]], t[1])
    }
    for airport := range targets {
        sort.Strings(targets[airport])      // 贪心排序：目的地按字典序排列
    }
    used := make(map[string][]bool)
    for airport, dsts := range targets {
        used[airport] = make([]bool, len(dsts))
    }
    path := []string{"JFK"}
    ticketCount := len(tickets)

    var backtrack func(cur string) bool
    backtrack = func(cur string) bool {
        if len(path) == ticketCount+1 {        // 终止条件：用完所有机票
            return true                             // 找到答案，立即返回 true
        }
        dsts, ok := targets[cur]
        if !ok {
            return false                             // 无出边，此分支失败
        }
        for i, dst := range dsts {               // 遍历已排序的目的地（贪心）
            if used[cur][i] {                        // 该机票已使用，跳过
                continue
            }
            used[cur][i] = true              // 【选择】
            path = append(path, dst)          // 【选择】
            if backtrack(dst) {                // 【递归】
                return true                          // 已找到答案，直接向上传递 true
            }
            path = path[:len(path)-1]           // 【撤销选择】
            used[cur][i] = false                 // 【撤销选择】
        }
        return false
    }
    backtrack("JFK")
    return path
}
```

```cpp
class Solution {
private:
    // targets[机场] = 该机场可飞往的目的地列表（已按字典序排序），
    // pair 的第二个元素表示这张机票是否已被使用
    unordered_map<string, vector<pair<string, bool>>> targets;
    vector<string> path;
    int ticketCount;

    // 方法：backtrack；按题目要求处理输入并返回结果，核心算法见方法体。
    bool backtrack(const string& cur) {
        if (path.size() == ticketCount + 1) {   // 终止条件：用完所有机票，路径长度=机票数+1
            return true;                            // 找到答案，直接返回 true，不再继续搜索
        }
        if (targets.find(cur) == targets.end()) {
            return false;                            // 当前机场没有可飞往的目的地，此分支失败
        }
        for (auto& target : targets[cur]) {          // 遍历已排序的目的地列表（贪心：优先字典序小）
            if (target.second) continue;               // 这张机票已经用过，跳过
            target.second = true;               // 【选择】标记这张机票已使用
            path.push_back(target.first);        // 【选择】加入路径
            if (backtrack(target.first)) {       // 【递归】
                return true;                          // 子递归已找到答案，直接向上传递 true，不再回溯
            }
            path.pop_back();                       // 【撤销选择】此分支失败，回退路径
            target.second = false;                  // 【撤销选择】恢复机票未使用状态
        }
        return false;                              // 所有目的地都尝试过仍未找到答案
    }

public:
    // 方法：findItinerary；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<string> findItinerary(vector<vector<string>>& tickets) {
        ticketCount = tickets.size();
        for (auto& t : tickets) {
            targets[t[0]].push_back({t[1], false});   // 建图：记录每张机票，初始未使用
        }
        for (auto& [airport, list] : targets) {
            sort(list.begin(), list.end());             // 贪心排序：目的地按字典序排列
        }
        path.push_back("JFK");                         // 起点固定为 JFK
        backtrack("JFK");
        return path;
    }
};
```

## 总结

- 这题是欧拉路径问题的回溯解法：把机票看作图的边，用回溯搜索一条能用完所有边的路径，而不是套用经典的 Hierholzer 欧拉路径算法，这样代码更符合回溯思维习惯。
- 贪心排序（每个机场的目的地列表预先按字典序排序）是保证结果字典序最小的关键，配合"找到答案就立即终止"的策略，第一条走到底的完整路径就是最优解，不需要枚举比较所有合法路径。
- 终止条件要通过返回布尔值的方式一路向上传递：一旦某层递归返回 `true`，所有调用它的上层函数都要立即返回 `true`，不能继续走 for 循环的其他分支，这是区别于普通回溯题"找到一个答案后仍需撤销继续搜索"的特殊之处。
- 标记"已使用"的对象是每一条具体的边（机票），而不是机场节点本身，为什么呢？因为同一个机场可以在行程中被经过多次，只有对应的某张机票用过一次后不能再用。

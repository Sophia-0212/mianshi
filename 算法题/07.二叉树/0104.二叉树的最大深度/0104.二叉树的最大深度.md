# 0104. 二叉树的最大深度

**题目描述**

给定一个二叉树的根节点 `root`，返回其最大深度。二叉树的最大深度是指从根节点到最远叶子节点的最长路径上的节点数。

```
示例：
输入: root = [3,9,20,null,null,15,7]
      3
     / \
    9  20
      /  \
     15   7
输出: 3
```

## 思路

这道题其实有两种很自然的思路，对应两种不同的遍历方式。

**第一种思路是后序遍历，也就是自底向上的方式。** 你想啊，一棵树的最大深度，等于 max(左子树最大深度, 右子树最大深度) + 1，这个公式天生就是递归的。为什么呢？因为要知道当前节点的深度，你得先知道它左右子树各自的深度是多少，所以必须先把左右子树递归处理完（这正好就是后序遍历"左右根"的顺序），然后再回到当前节点，拿子树算出来的结果去算当前节点为根的深度。递归的终止条件很简单，节点为空的时候，深度就是 0。

这种写法本质上是在算"高度"：以某个节点为根的子树，它的高度就等于这个节点到最远叶子节点的路径长度。那对根节点来说，高度和深度其实是同一个东西，所以这道题只要求出"根节点的高度"，答案就出来了。

**第二种思路呢，是用层序遍历，也就是 BFS，直接去数层数。** 用层序遍历那个模板（记录每一层的队列长度），每处理完一层，就把层数计数器加 1，遍历完了之后，这个计数器的值就是最大深度。这种写法其实更直观，你想啊，"深度"这个概念本身不就是"总共有多少层"嘛，用 BFS 一层一层去数，是最自然的对应关系。

这两种思路复杂度都是 O(n)，因为每个节点都只访问一次；空间复杂度的话，递归法是 O(h)，也就是调用栈的开销，层序遍历法则是 O(n)，最坏情况队列里要存下一整层的节点。

那这两种思路其实也对应了后面会讲到的"自顶向下"和"自底向上"这两种递归模式的选择。这道题用自底向上（也就是后序遍历，先把子树的结果算出来再往上合并）会更简洁一些，因为你不需要在递归过程中额外带一个深度参数从上往下传。

## 代码

二叉树节点定义：

```cpp
struct TreeNode {
    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode(int x) : val(x), left(NULL), right(NULL) {}
};
```

```python
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right
```

### C++

```cpp
class Solution {
public:
    // 思路一：后序遍历递归，自底向上求高度
    int maxDepth(TreeNode* root) {
        if (root == nullptr) return 0;             // 终止条件：空节点深度为0
        int leftDepth = maxDepth(root->left);      // 先求左子树深度
        int rightDepth = maxDepth(root->right);    // 再求右子树深度
        return max(leftDepth, rightDepth) + 1;     // 当前节点深度 = 较大子树深度 + 1（加上自己这一层）
    }

    // 思路二：层序遍历统计层数
    int maxDepthBFS(TreeNode* root) {
        if (root == nullptr) return 0;
        queue<TreeNode*> que;
        que.push(root);
        int depth = 0;
        while (!que.empty()) {
            int size = que.size();       // 记录当前层节点数
            for (int i = 0; i < size; i++) {
                TreeNode* node = que.front();
                que.pop();
                if (node->left) que.push(node->left);
                if (node->right) que.push(node->right);
            }
            depth++;                     // 处理完一层，深度加一
        }
        return depth;
    }
};
```

### Java

```java
class Solution {
    // 思路一：后序遍历递归
    public int maxDepth(TreeNode root) {
        if (root == null) return 0;               // 终止条件
        int leftDepth = maxDepth(root.left);       // 左子树深度
        int rightDepth = maxDepth(root.right);     // 右子树深度
        return Math.max(leftDepth, rightDepth) + 1; // 取较大值加上当前层
    }

    // 思路二：层序遍历统计层数
    public int maxDepthBFS(TreeNode root) {
        if (root == null) return 0;
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        int depth = 0;
        while (!queue.isEmpty()) {
            int size = queue.size();      // 当前层节点数
            for (int i = 0; i < size; i++) {
                TreeNode node = queue.poll();
                if (node.left != null) queue.offer(node.left);
                if (node.right != null) queue.offer(node.right);
            }
            depth++;                      // 处理完一层，深度加一
        }
        return depth;
    }
}
```

### Python

```python
class Solution:
    def maxDepth(self, root):
        # 思路一：后序遍历递归
        if root is None:
            return 0                          # 终止条件
        left_depth = self.maxDepth(root.left)   # 左子树深度
        right_depth = self.maxDepth(root.right) # 右子树深度
        return max(left_depth, right_depth) + 1  # 取较大值加上当前层

    def maxDepthBFS(self, root):
        # 思路二：层序遍历统计层数
        from collections import deque
        if root is None:
            return 0
        queue = deque([root])
        depth = 0
        while queue:
            size = len(queue)             # 当前层节点数
            for _ in range(size):
                node = queue.popleft()
                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)
            depth += 1                     # 处理完一层，深度加一
        return depth
```

### Go

```go
// 思路一：后序遍历递归
func maxDepth(root *TreeNode) int {
    if root == nil {
        return 0 // 终止条件
    }
    leftDepth := maxDepth(root.Left)   // 左子树深度
    rightDepth := maxDepth(root.Right) // 右子树深度
    if leftDepth > rightDepth {
        return leftDepth + 1 // 取较大值加上当前层
    }
    return rightDepth + 1
}

// 思路二：层序遍历统计层数
func maxDepthBFS(root *TreeNode) int {
    if root == nil {
        return 0
    }
    queue := []*TreeNode{root}
    depth := 0
    for len(queue) > 0 {
        size := len(queue) // 当前层节点数
        for i := 0; i < size; i++ {
            node := queue[0]
            queue = queue[1:]
            if node.Left != nil {
                queue = append(queue, node.Left)
            }
            if node.Right != nil {
                queue = append(queue, node.Right)
            }
        }
        depth++ // 处理完一层，深度加一
    }
    return depth
}
```

## 总结

- 求最大深度，说白了本质就是求"根节点的高度"，用后序遍历（自底向上）来做是天然合适的：先算出左右子树的高度，再用 max+1 合并成当前节点的高度。
- 用层序遍历数层数的思路更直观地对应了"深度等于层数"这个定义，拿 BFS 模板一层一层数就行。
- 两种解法复杂度都是 O(n)，实际面试中递归写法代码量更小，一般是更常用的首选。
- 这道题其实是"自底向上"递归模式的一个典型代表：不需要额外带参数从根往下传信息，而是让子树先算好结果，回溯的时候在父节点这里合并起来。

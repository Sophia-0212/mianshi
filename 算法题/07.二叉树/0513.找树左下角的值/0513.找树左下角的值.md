# 0513. 找树左下角的值

**题目描述**

给定一个二叉树的根节点 `root`，找出该二叉树的**最底层最左边**节点的值。

```
示例：
输入: root = [2,1,3]
输出: 1

输入: root = [1,2,3,4,null,5,6,null,null,7]
       1
      / \
     2   3
    /   / \
   4   5   6
  /
 7
输出: 7
```

## 思路

这道题要找的是"最底层"而且是"最左边"的节点，你看这天然就对应两种思路：一种是按层去处理，也就是 BFS，另一种是按深度去记录，也就是 DFS。

思路一是层序遍历，也就是 BFS，直接取每层的第一个节点。用标准的层序遍历模板，每处理一层的时候，只需要记录这一层第一个被处理到的节点的值就行了。为什么呢？因为层序遍历本身就保证了同一层里的节点是按从左到右的顺序被访问的，所以每一层"第一个访问到的节点"自然就是这一层最左边的那个。那遍历完所有层之后，最后一次记录下来的值，就是最底层最左边节点的值——这是因为后面的层会不断覆盖前面层记录的结果，最终留下来的一定是最后一层，也就是最底层的记录。

思路二是递归，也就是 DFS，记录最大深度，遇到更深层级的时候就更新答案。这里要维护两个全局状态：一个是当前记录的最大深度 `maxDepth`，初始值设成一个不可能达到的数，比如 -1；另一个是对应的结果值 `result`。递归遍历二叉树的时候用前序遍历，也就是先访问当前节点，再递归子树，每次进入一个新节点的时候把它的深度也传进去。这里的判断条件是：如果当前节点的深度大于目前记录的 `maxDepth`，那就说明这是目前遍历到的最深层级里第一次被访问到的节点——为什么呢？因为在前序遍历里，同一层最左边的节点一定比同一层其他节点先被访问到。这时候就更新 `maxDepth` 为当前深度，同时把 `result` 更新成当前节点的值。那后面同一层再访问到的节点，它们的深度不会超过 `maxDepth`，所以不会覆盖已经记录的最左值，只有真正进入更深一层的时候才会去更新结果。

这里有个地方必须注意：递归的时候一定要先递归左子树，再递归右子树，也就是保持前序遍历的左右顺序，这样才能保证同一深度里最左边的节点最先被访问到，"深度增加了才更新结果"这个判断逻辑才能成立。如果不小心写反了，先递归右子树再递归左子树，那逻辑就反过来了，找到的会变成"最右边"的节点，这个坑要注意。

两种思路的时间复杂度都是 O(n)，每个节点访问一次；空间复杂度这块，BFS 是 O(n)，因为队列最坏情况下要存一整层的节点，DFS 是 O(h)，也就是递归调用栈的开销。

## 代码

二叉树节点定义：

```cpp
struct TreeNode {
    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode(int x) : val(x), left(NULL), right(NULL) {}
};
```

```python
    def __init__(self, val=0, left=None, right=None):
```

### C++

```cpp
class Solution {
public:
    // 思路一：层序遍历，取每层第一个节点，最终保留最后一层的记录
    int findBottomLeftValueBFS(TreeNode* root) {
        queue<TreeNode*> que;
        que.push(root);
        int result = root->val;
        while (!que.empty()) {
            int size = que.size();
            for (int i = 0; i < size; i++) {
                TreeNode* node = que.front();
                que.pop();
                if (i == 0) result = node->val;   // 记录本层第一个访问到的节点（最左）
                if (node->left) que.push(node->left);
                if (node->right) que.push(node->right);
            }
        }
        return result;  // 遍历结束，result 保留的是最后一层（最底层）的最左值
    }

    // 思路二：DFS，记录最大深度，深度增加时更新结果
    int maxDepth = -1;
    int result = 0;
    void traverse(TreeNode* node, int depth) {
        if (node == nullptr) return;
        if (node->left == nullptr && node->right == nullptr) {
            // 到达叶子节点，检查是否比当前记录的最大深度更深
            if (depth > maxDepth) {
                maxDepth = depth;      // 更新最大深度
                result = node->val;    // 更新最左值（因为前序遍历先左后右，第一次到达更深层就是最左）
            }
        }
        traverse(node->left, depth + 1);  // 先递归左子树（保证左边先被访问到）
        traverse(node->right, depth + 1); // 再递归右子树
    }
    int findBottomLeftValue(TreeNode* root) {
        traverse(root, 0);
        return result;
    }
};
```

### Python

```python
class Solution:
    def findBottomLeftValueBFS(self, root):
        # 思路一：层序遍历
        from collections import deque
        queue = deque([root])
        result = root.val
        while queue:
            size = len(queue)
            for i in range(size):
                node = queue.popleft()
                if i == 0:
                    result = node.val         # 本层第一个节点即最左
                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)
        return result

    def findBottomLeftValue(self, root):
        # 思路二：DFS
        self.max_depth = -1
        self.result = 0

        def traverse(node, depth):
            if node is None:
                return
            if node.left is None and node.right is None:
                if depth > self.max_depth:
                    self.max_depth = depth   # 更新最大深度
                    self.result = node.val   # 更新最左值
            traverse(node.left, depth + 1)   # 先左
            traverse(node.right, depth + 1)  # 后右

        traverse(root, 0)
        return self.result
```

```go
func findBottomLeftValueBFS(root *TreeNode) int {
    queue := []*TreeNode{root}
    result := root.Val
    for len(queue) > 0 {
        size := len(queue)
        for i := 0; i < size; i++ {
            node := queue[0]
            queue = queue[1:]

                queue = append(queue, node.Left)

            if node.Right != nil {

        }
    }
    return result
}
```

## 总结

- 层序遍历这个解法是最直观的：每层只记录第一个被访问到的节点值，遍历结束之后留下来的就是最后一层，也就是最底层的记录，天然就满足了"最底层最左"这个要求。
- DFS 解法需要维护"当前最大深度"和"对应结果值"这两个状态，而且必须保证先递归左子树再递归右子树，这样同一深度里最左边的节点才会先被访问到，"深度增加了才更新结果"这个判断逻辑才能立得住。
- 两种解法的复杂度都是 O(n)，选哪种主要看个人偏好：BFS 写法更符合"层"的那种直觉，DFS 写法代码量稍微少一点，但要注意递归顺序不能写错。
- 这道题其实又体现了一个套路：跟"层"相关的问题优先考虑用 BFS，如果需要在遍历过程中携带额外状态（比如这里的深度），用 DFS 也能等价地解决。

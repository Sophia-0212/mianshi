# 1382. 将二叉搜索树变平衡

**题目描述**

给定一棵二叉搜索树（BST），请将其转换为一棵**高度平衡**的二叉搜索树，返回任意一个符合要求的新二叉搜索树的根节点即可（答案不唯一）。

```
示例：
输入: root = [1,null,2,null,3,null,4,null,null]  （一条向右倾斜的链）
输出: [2,1,3,null,null,null,4]  （或任意其他高度平衡的合法 BST）
解释: 原树退化成链表形态，高度为 4；转换后的树高度平衡，高度为 2 或 3
```

要求：转换后的树必须仍然是合法的 BST（中序遍历结果不变），且任意节点的左右子树高度差不超过 1。

## 思路

中序遍历得到有序数组，再递归取中点为根，左右半区间构造子树，得到平衡 BST。

时间 O(n)，空间 O(n)。

```python
class TreeNode:
    # 方法：__init__；按题目要求处理输入并返回结果，核心算法见方法体。
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    # 方法：balanceBST；按题目要求处理输入并返回结果，核心算法见方法体。
    def balanceBST(self, root: TreeNode) -> TreeNode:
        sorted_vals = []
        self._inorder(root, sorted_vals)  # 第一步：中序遍历得到有序数组
        return self._build_balanced(sorted_vals, 0, len(sorted_vals) - 1)  # 第二步：重建平衡 BST

    # 方法：_inorder；按题目要求处理输入并返回结果，核心算法见方法体。
    def _inorder(self, node: TreeNode, vals: list) -> None:
        if node is None:
            return
        self._inorder(node.left, vals)   # 先访问左子树
        vals.append(node.val)            # 再记录当前节点值（保证升序）
        self._inorder(node.right, vals)  # 最后访问右子树

    # 方法：_build_balanced；按题目要求处理输入并返回结果，核心算法见方法体。
    def _build_balanced(self, vals: list, left: int, right: int) -> TreeNode:
        if left > right:
            return None  # 区间为空
        mid = left + (right - left) // 2  # 取中点作为根
        node = TreeNode(vals[mid])
        node.left = self._build_balanced(vals, left, mid - 1)    # 左半区间构建左子树
        node.right = self._build_balanced(vals, mid + 1, right)  # 右半区间构建右子树
        return node
```

```go
    var sortedVals []int
    inorder(root, &sortedVals)                       // 第一步：中序遍历得到有序数组
    return buildBalanced(sortedVals, 0, len(sortedVals)-1) // 第二步：重建平衡 BST
}

// 方法：inorder；按题目要求处理输入并返回结果，核心算法见方法体。
func inorder(node *TreeNode, vals *[]int) {
    if node == nil {
        return
    }
    inorder(node.Left, vals)          // 先访问左子树
    *vals = append(*vals, node.Val)   // 再记录当前节点值
    inorder(node.Right, vals)         // 最后访问右子树
}

// 方法：buildBalanced；按题目要求处理输入并返回结果，核心算法见方法体。
func buildBalanced(vals []int, left, right int) *TreeNode {
    if left > right {
        return nil // 区间为空
    }
    mid := left + (right-left)/2 // 取中点作为根
    node := &TreeNode{Val: vals[mid]}
    node.Left = buildBalanced(vals, left, mid-1)   // 左半区间构建左子树
    node.Right = buildBalanced(vals, mid+1, right) // 右半区间构建右子树
    return node
}
```

    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode() : val(0), left(nullptr), right(nullptr) {}
    TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
    TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
};

class Solution {
public:
    TreeNode* balanceBST(TreeNode* root) {
        vector<int> sortedVals;
        inorder(root, sortedVals);           // 第一步：中序遍历得到有序数组
        return buildBalanced(sortedVals, 0, (int)sortedVals.size() - 1);  // 第二步：重建平衡 BST
    }

private:
    // 方法：inorder；按题目要求处理输入并返回结果，核心算法见方法体。
    void inorder(TreeNode* node, vector<int>& vals) {
        if (node == nullptr) return;
        inorder(node->left, vals);   // 先访问左子树
        vals.push_back(node->val);   // 再记录当前节点值（保证升序）
        inorder(node->right, vals);  // 最后访问右子树
    }

    TreeNode* buildBalanced(vector<int>& vals, int left, int right) {
        if (left > right) return nullptr;  // 区间为空，返回空节点

        TreeNode* node = new TreeNode(vals[mid]);
        node->left = buildBalanced(vals, left, mid - 1);   // 左半区间构建左子树
        node->right = buildBalanced(vals, mid + 1, right); // 右半区间构建右子树

```cpp
    }

// 方法：inorder；按题目要求处理输入并返回结果，核心算法见方法体。
    void inorder(TreeNode* node, vector<int>& vals) {
        if (node == nullptr) return;
        inorder(node->left, vals);   // 先访问左子树
        vals.push_back(node->val);   // 再记录当前节点值（保证升序）
        inorder(node->right, vals);  // 最后访问右子树
    }

TreeNode* buildBalanced(vector<int>& vals, int left, int right) {
        if (left > right) return nullptr;  // 区间为空，返回空节点
        int mid = left + (right - left) / 2;  // 取中点作为根，防止溢出
        TreeNode* node = new TreeNode(vals[mid]);
        node->left = buildBalanced(vals, left, mid - 1);   // 左半区间构建左子树
        node->right = buildBalanced(vals, mid + 1, right); // 右半区间构建右子树
        return node;
    }

}

int main() {
    // 用例 1：输入 ({'tree': [1, None, 2, None, 3, None, 4]},)；期望 [1, 2, 3, 4]。
    {
        auto* arg0 = buildTreeInput({1, nullopt, 2, nullopt, 3, nullopt, 4});
        auto result = interview::balanceBST(arg0);
        checkedHeight(result);
        cout << "用例 1: 期望=[1, 2, 3, 4] 实际=";
        show(inorderValues(result));
        cout << "\n";
    }
    // 用例 2：输入 ({'tree': [2, 1, 3]},)；期望 [1, 2, 3]。
    {
        auto* arg0 = buildTreeInput({2, 1, 3});
        auto result = interview::balanceBST(arg0);
        checkedHeight(result);
        cout << "用例 2: 期望=[1, 2, 3] 实际=";
        show(inorderValues(result));
        cout << "\n";
    }
    // 用例 3：输入 ({'tree': [1]},)；期望 [1]。
    {
        auto* arg0 = buildTreeInput({1});
        auto result = interview::balanceBST(arg0);
        checkedHeight(result);
        cout << "用例 3: 期望=[1] 实际=";
```

## 总结

- 核心技巧是"BST → 中序有序数组 → 平衡 BST"这个两段式转化，把一个新问题拆解成两个已知的经典子问题（中序遍历、108 题有序数组转平衡 BST）的组合。
- 这体现了一个通用规律：**任何 BST 都可以先转换成中序有序数组，再用同一套构建方法重建**，不管原树形态多么退化（比如完全变成一条链）都一样。
- 容易出错的地方在于取中点的下标计算要用 `left + (right - left) / 2` 来防止溢出，还有递归终止条件 `left > right` 不能写错（要是写成 `left >= right` 就会漏掉单节点区间）。
- 这题答案不唯一，只要中序遍历结果跟原树一致（也就是 BST 性质不变）而且高度平衡就行，不用纠结于非得构造出某个"标准"形态的树。

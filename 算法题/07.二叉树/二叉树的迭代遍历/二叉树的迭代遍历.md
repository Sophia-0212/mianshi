# 二叉树的迭代遍历

**内容概述**

你想啊，递归遍历靠的是系统调用栈自动帮你维护"待处理节点"的顺序，那迭代遍历呢，说白了就得自己用一个显式的栈来模拟这个过程。三种遍历里，前序最容易写，中序需要一个额外的指针配合，后序则可以用一个小技巧转化成前序的变体，挺巧妙的。

节点定义（后续代码统一使用）：

## 思路

- 前序：弹栈即访问，先压右孩子，再压左孩子。
- 中序：沿左链入栈；弹栈访问后转向右孩子。
- 后序：按“根→右→左”收集，再反转为“左→右→根”。

时间 O(n)，辅助栈空间 O(h)，结果空间 O(n)，h 为树高。

class Solution:
    # 方法：preorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    def preorderTraversal(self, root):
        # 前序遍历迭代法：根 -> 左 -> 右
        result = []
        if root is None:
            return result
        stack = [root]           # 初始把根节点压栈
        while stack:
            node = stack.pop()
            result.append(node.val)   # 弹出即访问
            if node.right:
                stack.append(node.right)  # 先压右孩子
            if node.left:
                stack.append(node.left)   # 再压左孩子，保证左先弹出
        return result

    # 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    def inorderTraversal(self, root):
        # 中序遍历迭代法：左 -> 根 -> 右
        result = []
        stack = []
        cur = root                # cur 跟踪当前探索节点
        while cur is not None or stack:
            if cur is not None:   # 一直向左压栈，不访问
                stack.append(cur)
                cur = cur.left
            else:                  # 左边走到头，弹出并访问
                cur = stack.pop()
                result.append(cur.val)
                cur = cur.right   # 转向右子树
        return result

    # 方法：postorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    def postorderTraversal(self, root):
        # 后序遍历迭代法：左 -> 右 -> 根
        result = []
        if root is None:
            return result
        stack = [root]
        while stack:
            node = stack.pop()
            result.append(node.val)
            if node.left:
                stack.append(node.left)   # 先压左孩子
            if node.right:
                stack.append(node.right)  # 再压右孩子，保证右先弹出
        return result[::-1]       # 反转"根右左"得到"左右根"
```

// 前序遍历迭代法：根 -> 左 -> 右
// 方法：preorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
func preorderTraversal(root *TreeNode) []int {
    result := []int{}
    if root == nil {
        return result
    }
    stack := []*TreeNode{root} // 初始把根节点压栈
    for len(stack) > 0 {
        node := stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append(result, node.Val) // 弹出即访问
        if node.Right != nil {
            stack = append(stack, node.Right) // 先压右孩子
        }
        if node.Left != nil {
            stack = append(stack, node.Left) // 再压左孩子，保证左先弹出
        }
    }
    return result
}

// 中序遍历迭代法：左 -> 根 -> 右
// 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
func inorderTraversal(root *TreeNode) []int {
    result := []int{}
    var stack []*TreeNode
    cur := root // cur 跟踪当前探索节点
    for cur != nil || len(stack) > 0 {
        if cur != nil { // 一直向左压栈，不访问
            stack = append(stack, cur)
            cur = cur.Left
        } else { // 左边走到头，弹出并访问
            cur = stack[len(stack)-1]
            stack = stack[:len(stack)-1]
            result = append(result, cur.Val)
            cur = cur.Right // 转向右子树
        }
    }
    return result
}

// 后序遍历迭代法：左 -> 右 -> 根
// 方法：postorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
func postorderTraversal(root *TreeNode) []int {
    result := []int{}
    if root == nil {
        return result
    }
    stack := []*TreeNode{root}
    for len(stack) > 0 {
        node := stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append(result, node.Val)
        if node.Left != nil {
            stack = append(stack, node.Left) // 先压左孩子
        }
        if node.Right != nil {
            stack = append(stack, node.Right) // 再压右孩子，保证右先弹出
        }
    }
    // 反转"根右左"得到"左右根"
    for i, j := 0, len(result)-1; i < j; i, j = i+1, j-1 {
        result[i], result[j] = result[j], result[i]
    }
    return result
}

```cpp
// 依赖说明：vector：C++ STL 动态数组。；stack：C++ STL 栈容器。

class Solution {
public:
    // 前序遍历迭代法：根 -> 左 -> 右
    // 方法：preorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> result;
        if (root == nullptr) return result;
        stack<TreeNode*> st;
        st.push(root);                    // 初始把根节点压栈
        while (!st.empty()) {
            TreeNode* node = st.top();
            st.pop();
            result.push_back(node->val);  // 弹出即访问，符合"先访问根"的要求
            if (node->right) st.push(node->right); // 先压右孩子
            if (node->left) st.push(node->left);   // 再压左孩子，保证左孩子先被弹出
        }
        return result;
    }

    // 中序遍历迭代法：左 -> 根 -> 右
    // 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        TreeNode* cur = root;             // cur 指针跟踪当前探索的节点
        while (cur != nullptr || !st.empty()) {
            if (cur != nullptr) {         // 只要还能往左走，就一直压栈不访问
                st.push(cur);
                cur = cur->left;
            } else {                       // 左边走到头，弹出栈顶并访问
                cur = st.top();
                st.pop();
                result.push_back(cur->val); // 此时左子树已完全处理，访问当前节点
                cur = cur->right;          // 转向右子树，重复上述过程
            }
        }
        return result;
    }

    // 后序遍历迭代法：左 -> 右 -> 根（用"根右左"逆序实现）
    // 方法：postorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> postorderTraversal(TreeNode* root) {
        vector<int> result;
        if (root == nullptr) return result;
        stack<TreeNode*> st;
        st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            st.pop();
            result.push_back(node->val);
            if (node->left) st.push(node->left);   // 先压左孩子
            if (node->right) st.push(node->right); // 再压右孩子，保证右孩子先被弹出（得到"根右左"）
        }
        reverse(result.begin(), result.end());     // 反转"根右左"得到"左右根"
        return result;
    }
};```

    if (values.empty() || !values[0]) return nullptr;
    auto* root = new TreeNode(*values[0]);
    queue<TreeNode*> q; q.push(root);
    size_t i = 1;
    while (!q.empty() && i < values.size()) {
        auto* node = q.front(); q.pop();
        if (i < values.size() && values[i]) { node->left = new TreeNode(*values[i]); q.push(node->left); }
        ++i;
        if (i < values.size() && values[i]) { node->right = new TreeNode(*values[i]); q.push(node->right); }
        ++i;
    }
    return root;
}
vector<optional<int>> treeValues(TreeNode* root) {
    vector<optional<int>> result;
    queue<TreeNode*> q; q.push(root);
    while (!q.empty()) {
        auto* node = q.front(); q.pop();
        if (node) { result.push_back(node->val); q.push(node->left); q.push(node->right); }
        else result.push_back(nullopt);
    }
    while (!result.empty() && !result.back()) result.pop_back();
    return result;
}
TreeNode* findNode(TreeNode* root, int val) {
    if (!root || root->val == val) return root;
    auto* left = findNode(root->left, val);
    return left ? left : findNode(root->right, val);
}
vector<int> inorderValues(TreeNode* root) {
    if (!root) return {};
    auto a = inorderValues(root->left), b = inorderValues(root->right);
    a.push_back(root->val); a.insert(a.end(), b.begin(), b.end()); return a;
}
int checkedHeight(TreeNode* root) {
    if (!root) return 0;
    int l = checkedHeight(root->left), r = checkedHeight(root->right);
    if (abs(l - r) > 1) throw runtime_error("tree is not balanced");
    return 1 + max(l, r);
}
vector<vector<int>> nextLevels(TreeNode* root) {
    vector<vector<int>> result;
    for (; root; root = root->left) {
        vector<int> row;
        for (auto* n = root; n; n = n->next) row.push_back(n->val);
        result.push_back(row);
    }
    return result;
}
void show(TreeNode* root) { show(treeValues(root)); }

// 普通函数与辅助函数声明。
vector<int> preorderTraversal(TreeNode* root);
vector<int> inorderTraversal(TreeNode* root);
vector<int> postorderTraversal(TreeNode* root);

// 前序遍历迭代法：根 -> 左 -> 右
    // 方法：preorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> result;
        if (root == nullptr) return result;
        stack<TreeNode*> st;
        st.push(root);                    // 初始把根节点压栈
        while (!st.empty()) {
            TreeNode* node = st.top();
            st.pop();
            result.push_back(node->val);  // 弹出即访问，符合"先访问根"的要求
            if (node->right) st.push(node->right); // 先压右孩子
            if (node->left) st.push(node->left);   // 再压左孩子，保证左孩子先被弹出
        }
        return result;
    }

// 中序遍历迭代法：左 -> 根 -> 右
    // 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        TreeNode* cur = root;             // cur 指针跟踪当前探索的节点
        while (cur != nullptr || !st.empty()) {
            if (cur != nullptr) {         // 只要还能往左走，就一直压栈不访问
                st.push(cur);
                cur = cur->left;
            } else {                       // 左边走到头，弹出栈顶并访问
                cur = st.top();
                st.pop();
                result.push_back(cur->val); // 此时左子树已完全处理，访问当前节点
                cur = cur->right;          // 转向右子树，重复上述过程
            }
        }
        return result;
    }

// 后序遍历迭代法：左 -> 右 -> 根（用"根右左"逆序实现）
    // 方法：postorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> postorderTraversal(TreeNode* root) {
        vector<int> result;
        if (root == nullptr) return result;
        stack<TreeNode*> st;
        st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            st.pop();
            result.push_back(node->val);
            if (node->left) st.push(node->left);   // 先压左孩子
            if (node->right) st.push(node->right); // 再压右孩子，保证右孩子先被弹出（得到"根右左"）
        }
        reverse(result.begin(), result.end());     // 反转"根右左"得到"左右根"
        return result;
    }

}

    {
        auto* arg0 = buildTreeInput({1, nullopt, 2, 3});
        auto result = interview::inorderTraversal(arg0);
        cout << "用例 1: 期望=[1, 3, 2] 实际=";

    // 用例 3：输入 ({'tree': []},)；期望 []。

## 总结

- 前序迭代最直观：栈的"后进先出"特性正好对应"先访问根再访问左"，压栈时要先压右孩子再压左孩子，这样才能保证左孩子先出栈。
- 中序迭代需要额外的 `cur` 指针配合栈来用：不断向左压栈直到走到头，再弹出访问，然后转向右子树，说白了本质就是在模拟"沿着左链一直往下，再折返"的过程。
- 后序迭代最省心的写法就是复用前序迭代的代码，把入栈顺序改为先左后右（这样能得到"根右左"），最后整体反转结果就得到"左右根"了，不需要单独设计新逻辑，挺讨巧的。
- 三种迭代写法的时间空间复杂度和递归版本是一致的，区别只是显式管理栈还是依赖系统调用栈，这也是为什么迭代法能避免递归深度过大导致栈溢出这个问题的原因。

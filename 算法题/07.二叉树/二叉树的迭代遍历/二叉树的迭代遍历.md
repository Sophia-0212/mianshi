# 二叉树的迭代遍历

**内容概述**

你想啊，递归遍历靠的是系统调用栈自动帮你维护"待处理节点"的顺序，那迭代遍历呢，说白了就得自己用一个显式的栈来模拟这个过程。三种遍历里，前序最容易写，中序需要一个额外的指针配合，后序则可以用一个小技巧转化成前序的变体，挺巧妙的。

节点定义（后续代码统一使用）：

```cpp
struct TreeNode {
    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode(int x) : val(x), left(NULL), right(NULL) {}
};
```

```python
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right
```

## 思路

**前序遍历（根左右）迭代法**：前序遍历是三种里最直观的，为什么呢？因为访问顺序和栈的"先进后出"特性可以直接对应上。用一个栈，先把根节点压进去。每次循环弹出栈顶节点并访问它，然后依次把它的右孩子、左孩子压入栈——这里要注意顺序：得先压右孩子再压左孩子，这样左孩子才会在下一次循环先被弹出，保证了"左子树先于右子树被访问"这个顺序。

**中序遍历（左根右）迭代法**：中序遍历不能直接套用前序的写法，为什么呢？因为中序要求先把左子树完全访问完，再访问根节点，最后才访问右子树——根节点的访问时机被"卡"在中间，不能像前序那样一弹出栈顶就直接访问。这里的关键是引入一个额外的指针 `cur` 来跟踪当前正在探索的节点：不断把 `cur` 及其左孩子依次压栈直到最左端，这一步只压栈不访问，等左边走到头（`cur` 为空）的时候，就弹出栈顶节点并访问它，然后把 `cur` 指向这个节点的右孩子，重复上述过程。这样就保证了每个节点都是在其左子树完全处理完之后才被访问的。

**后序遍历（左右根）迭代法**：后序遍历说实话最不直观，但有个经典技巧可以用：后序遍历的顺序是"左右根"，那如果把前序遍历"根左右"的压栈顺序改一下——压栈时先压左孩子再压右孩子，让右孩子先被弹出——就能得到"根右左"的访问顺序，这正好是"左右根"的**逆序**。所以你只需要用前序遍历的代码，把左右孩子入栈的顺序互换（先左后右），得到"根右左"的遍历结果，最后把整个结果数组反转一下，就得到了正确的"左右根"后序遍历结果。说白了这个技巧的本质就是"后序遍历 = 根右左遍历的逆序"，这样就避免了后序遍历本身需要额外状态去记录"是否已经访问过右子树"的复杂逻辑。

三种迭代写法的空间复杂度都是 O(h)（h 为树高，最坏 O(n)），时间复杂度都是 O(n)，跟递归版本是一致的，区别只是把递归调用栈换成了显式维护的栈而已。

## 代码

### C++

```cpp
class Solution {
public:
    // 前序遍历迭代法：根 -> 左 -> 右
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> result;
        if (root == nullptr) return result;
        stack<TreeNode*> st;
        st.push(root);                    // 初始把根节点压栈
        while (!st.empty()) {
            TreeNode* node = st.top();
            st.pop();
            result.push_back(node->val);  // 弹出即访问，符合"先访问根"的要求
            if (node->right) st.push(node->right); // 先压右孩子
            if (node->left) st.push(node->left);   // 再压左孩子，保证左孩子先被弹出
        }
        return result;
    }

    // 中序遍历迭代法：左 -> 根 -> 右
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        TreeNode* cur = root;             // cur 指针跟踪当前探索的节点
        while (cur != nullptr || !st.empty()) {
            if (cur != nullptr) {         // 只要还能往左走，就一直压栈不访问
                st.push(cur);
                cur = cur->left;
            } else {                       // 左边走到头，弹出栈顶并访问
                cur = st.top();
                st.pop();
                result.push_back(cur->val); // 此时左子树已完全处理，访问当前节点
                cur = cur->right;          // 转向右子树，重复上述过程
            }
        }
        return result;
    }

    // 后序遍历迭代法：左 -> 右 -> 根（用"根右左"逆序实现）
    vector<int> postorderTraversal(TreeNode* root) {
        vector<int> result;
        if (root == nullptr) return result;
        stack<TreeNode*> st;
        st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            st.pop();
            result.push_back(node->val);
            if (node->left) st.push(node->left);   // 先压左孩子
            if (node->right) st.push(node->right); // 再压右孩子，保证右孩子先被弹出（得到"根右左"）
        }
        reverse(result.begin(), result.end());     // 反转"根右左"得到"左右根"
        return result;
    }
};
```

```java
class Solution {
    // 前序遍历迭代法：根 -> 左 -> 右
    public List<Integer> preorderTraversal(TreeNode root) {
        List<Integer> result = new ArrayList<>();
        if (root == null) return result;
        Deque<TreeNode> stack = new ArrayDeque<>();
        stack.push(root);                     // 初始把根节点压栈
        while (!stack.isEmpty()) {
            TreeNode node = stack.pop();
            result.add(node.val);             // 弹出即访问
            if (node.right != null) stack.push(node.right); // 先压右孩子
            if (node.left != null) stack.push(node.left);   // 再压左孩子，保证左先弹出
        }
        return result;
    }

    // 中序遍历迭代法：左 -> 根 -> 右
    public List<Integer> inorderTraversal(TreeNode root) {
        List<Integer> result = new ArrayList<>();
        Deque<TreeNode> stack = new ArrayDeque<>();
        TreeNode cur = root;                  // cur 跟踪当前探索节点
        while (cur != null || !stack.isEmpty()) {
            if (cur != null) {                // 一直向左压栈，不访问
                stack.push(cur);
                cur = cur.left;
            } else {                           // 左边走到头，弹出并访问
                cur = stack.pop();
                result.add(cur.val);
                cur = cur.right;              // 转向右子树
            }
        }
        return result;
    }

    // 后序遍历迭代法：左 -> 右 -> 根
    public List<Integer> postorderTraversal(TreeNode root) {
        List<Integer> result = new ArrayList<>();
        if (root == null) return result;
        Deque<TreeNode> stack = new ArrayDeque<>();
        stack.push(root);
        while (!stack.isEmpty()) {
            TreeNode node = stack.pop();
            result.add(node.val);
            if (node.left != null) stack.push(node.left);   // 先压左孩子
            if (node.right != null) stack.push(node.right); // 再压右孩子，保证右先弹出
        }
        Collections.reverse(result);          // 反转"根右左"得到"左右根"
        return result;
    }
}
```

### Python

```python
class Solution:
    def preorderTraversal(self, root):
        # 前序遍历迭代法：根 -> 左 -> 右
        result = []
        if root is None:
            return result
        stack = [root]           # 初始把根节点压栈
        while stack:
            node = stack.pop()
            result.append(node.val)   # 弹出即访问
            if node.right:
                stack.append(node.right)  # 先压右孩子
            if node.left:
                stack.append(node.left)   # 再压左孩子，保证左先弹出
        return result

    def inorderTraversal(self, root):
        # 中序遍历迭代法：左 -> 根 -> 右
        result = []
        stack = []
        cur = root                # cur 跟踪当前探索节点
        while cur is not None or stack:
            if cur is not None:   # 一直向左压栈，不访问
                stack.append(cur)
                cur = cur.left
            else:                  # 左边走到头，弹出并访问
                cur = stack.pop()
                result.append(cur.val)
                cur = cur.right   # 转向右子树
        return result

    def postorderTraversal(self, root):
        # 后序遍历迭代法：左 -> 右 -> 根
        result = []
        if root is None:
            return result
        stack = [root]
        while stack:
            node = stack.pop()
            result.append(node.val)
            if node.left:
                stack.append(node.left)   # 先压左孩子
            if node.right:
                stack.append(node.right)  # 再压右孩子，保证右先弹出
        return result[::-1]       # 反转"根右左"得到"左右根"
```

### Go

```go
// 前序遍历迭代法：根 -> 左 -> 右
func preorderTraversal(root *TreeNode) []int {
    result := []int{}
    if root == nil {
        return result
    }
    stack := []*TreeNode{root} // 初始把根节点压栈
    for len(stack) > 0 {
        node := stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append(result, node.Val) // 弹出即访问
        if node.Right != nil {
            stack = append(stack, node.Right) // 先压右孩子
        }
        if node.Left != nil {
            stack = append(stack, node.Left) // 再压左孩子，保证左先弹出
        }
    }
    return result
}

// 中序遍历迭代法：左 -> 根 -> 右
func inorderTraversal(root *TreeNode) []int {
    result := []int{}
    var stack []*TreeNode
    cur := root // cur 跟踪当前探索节点
    for cur != nil || len(stack) > 0 {
        if cur != nil { // 一直向左压栈，不访问
            stack = append(stack, cur)
            cur = cur.Left
        } else { // 左边走到头，弹出并访问
            cur = stack[len(stack)-1]
            stack = stack[:len(stack)-1]
            result = append(result, cur.Val)
            cur = cur.Right // 转向右子树
        }
    }
    return result
}

// 后序遍历迭代法：左 -> 右 -> 根
func postorderTraversal(root *TreeNode) []int {
    result := []int{}
    if root == nil {
        return result
    }
    stack := []*TreeNode{root}
    for len(stack) > 0 {
        node := stack[len(stack)-1]
        stack = stack[:len(stack)-1]
        result = append(result, node.Val)
        if node.Left != nil {
            stack = append(stack, node.Left) // 先压左孩子
        }
        if node.Right != nil {
            stack = append(stack, node.Right) // 再压右孩子，保证右先弹出
        }
    }
    // 反转"根右左"得到"左右根"
    for i, j := 0, len(result)-1; i < j; i, j = i+1, j-1 {
        result[i], result[j] = result[j], result[i]
    }
    return result
}
```

## 总结

- 前序迭代最直观：栈的"后进先出"特性正好对应"先访问根再访问左"，压栈时要先压右孩子再压左孩子，这样才能保证左孩子先出栈。
- 中序迭代需要额外的 `cur` 指针配合栈来用：不断向左压栈直到走到头，再弹出访问，然后转向右子树，说白了本质就是在模拟"沿着左链一直往下，再折返"的过程。
- 后序迭代最省心的写法就是复用前序迭代的代码，把入栈顺序改为先左后右（这样能得到"根右左"），最后整体反转结果就得到"左右根"了，不需要单独设计新逻辑，挺讨巧的。
- 三种迭代写法的时间空间复杂度和递归版本是一致的，区别只是显式管理栈还是依赖系统调用栈，这也是为什么迭代法能避免递归深度过大导致栈溢出这个问题的原因。

# 二叉树的统一迭代法

**内容概述**

上一篇讲的迭代遍历里，前序和后序的写法很接近，但中序遍历需要额外引入一个 `cur` 指针，逻辑上和另外两种不统一，代码没法直接复用。这篇讲一种"标记法"，用同一套代码模板就能写出前中后序三种迭代遍历，只需要调整节点重新入栈的相对顺序就行。

节点定义（后续代码统一使用）：

## 思路

栈＋空标记区分“展开节点”和“访问节点”。未标记时按遍历顺序的逆序压入孩子及当前节点，并在当前节点后压入空标记；遇到标记，弹出下一个节点并访问。

时间 O(n)，辅助空间 O(h)，结果 O(n)，h 为树高。

class Solution:
    # 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    def inorderTraversal(self, root):
        # 中序遍历统一迭代法：左 -> 根 -> 右
        result = []
        stack = []
        if root:
            stack.append(root)
        while stack:
            node = stack[-1]
            if node is not None:
                stack.pop()
                if node.right:
                    stack.append(node.right)  # 右孩子先压
                stack.append(node)            # 根节点重新压回
                stack.append(None)            # 紧跟标记
                if node.left:
                    stack.append(node.left)   # 左孩子最后压
            else:
                stack.pop()                   # 弹出标记
                result.append(stack.pop().val)  # 弹出并访问真正的节点
        return result

    # 方法：preorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    def preorderTraversal(self, root):
        # 前序遍历统一迭代法：根 -> 左 -> 右
        result = []
        stack = []
        if root:
            stack.append(root)
        while stack:
            node = stack[-1]
            if node is not None:
                stack.pop()
                if node.right:
                    stack.append(node.right)
                if node.left:
                    stack.append(node.left)
                stack.append(node)
                stack.append(None)
            else:
                stack.pop()
                result.append(stack.pop().val)
        return result

    # 方法：postorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    def postorderTraversal(self, root):
        # 后序遍历统一迭代法：左 -> 右 -> 根
        result = []
        stack = []
        if root:
            stack.append(root)
        while stack:
            node = stack[-1]
            if node is not None:
                stack.pop()
                stack.append(node)
                stack.append(None)
                if node.right:
                    stack.append(node.right)
                if node.left:
                    stack.append(node.left)
            else:
                stack.pop()
                result.append(stack.pop().val)
        return result

def build_tree(values):
    """按层序构造树，None 表示空节点。"""
    if not values or values[0] is None:
        return None
    root = TreeNode(values[0])
    queue = deque([root])
    i = 1
    while queue and i < len(values):
        parent = queue.popleft()
        for side in ('left', 'right'):
            if i < len(values) and values[i] is not None:
                child = TreeNode(values[i])
                setattr(parent, side, child)
                queue.append(child)
            i += 1
    return root

    # 层序输出并去掉末尾空节点，方便和题目示例比较。

        node = queue.popleft()
        result.append(node.val if node else None)
        if node:
            queue.extend([node.left, node.right])
    while result and result[-1] is None:
        result.pop()
    return result

// 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
func inorderTraversal(root *TreeNode) []int {
    result := []int{}
    var stack []*TreeNode
    if root != nil {
        stack = append(stack, root)
    }
    for len(stack) > 0 {
        node := stack[len(stack)-1]
        if node != nil {
            stack = stack[:len(stack)-1]
            if node.Right != nil {
                stack = append(stack, node.Right) // 右先压
            }

            stack = stack[:len(stack)-1]

### LeetCode 写法与原有示例

```cpp
struct TreeNode {
    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode(int x) : val(x), left(NULL), right(NULL) {}
};
```

```cpp
// 依赖说明：vector：C++ STL 动态数组。；stack：C++ STL 栈容器。

class Solution {
public:
    // 中序遍历统一迭代法：左 -> 根 -> 右
    // 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != nullptr) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != nullptr) {
                st.pop();
                if (node->right) st.push(node->right); // 右孩子先压，最后出栈
                st.push(node);                          // 根节点重新压回
                st.push(nullptr);                       // 紧跟一个标记，表示下次弹出即访问
                if (node->left) st.push(node->left);    // 左孩子最后压，最先出栈
            } else {
                st.pop();                    // 弹出标记
                result.push_back(st.top()->val); // 标记后面就是真正要访问的节点
                st.pop();
            }
        }
        return result;
    }

    // 前序遍历统一迭代法：根 -> 左 -> 右
    // 方法：preorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != nullptr) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != nullptr) {
                st.pop();
                if (node->right) st.push(node->right); // 右孩子先压
                if (node->left) st.push(node->left);   // 左孩子次之
                st.push(node);                          // 根节点最后压，最先出栈访问
                st.push(nullptr);                       // 紧跟标记
            } else {
                st.pop();
                result.push_back(st.top()->val);
                st.pop();
            }
        }
        return result;
    }

    // 后序遍历统一迭代法：左 -> 右 -> 根
    // 方法：postorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> postorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != nullptr) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != nullptr) {
                st.pop();
                st.push(node);                          // 根节点先压（最后访问）
                st.push(nullptr);                       // 紧跟标记
                if (node->right) st.push(node->right);  // 右孩子次之
                if (node->left) st.push(node->left);    // 左孩子最后压，最先出栈
            } else {
                st.pop();
                result.push_back(st.top()->val);
                st.pop();
            }
        }
        return result;
    }
};```

struct TreeNode {
    int val;
    TreeNode *left, *right, *next;
    TreeNode(int value = 0, TreeNode* l = nullptr, TreeNode* r = nullptr)
        : val(value), left(l), right(r), next(nullptr) {}
};
using Node = TreeNode;

// 层序建树；nullopt 表示空节点。
TreeNode* buildTreeInput(const vector<optional<int>>& values) {
    if (values.empty() || !values[0]) return nullptr;
    auto* root = new TreeNode(*values[0]);
    queue<TreeNode*> q; q.push(root);
    size_t i = 1;
    while (!q.empty() && i < values.size()) {
        auto* node = q.front(); q.pop();
        if (i < values.size() && values[i]) { node->left = new TreeNode(*values[i]); q.push(node->left); }
        ++i;
        if (i < values.size() && values[i]) { node->right = new TreeNode(*values[i]); q.push(node->right); }
        ++i;
    }
    return root;
}
vector<optional<int>> treeValues(TreeNode* root) {
    vector<optional<int>> result;
    queue<TreeNode*> q; q.push(root);
    while (!q.empty()) {
        auto* node = q.front(); q.pop();
        if (node) { result.push_back(node->val); q.push(node->left); q.push(node->right); }
        else result.push_back(nullopt);
    }
    while (!result.empty() && !result.back()) result.pop_back();
    return result;
}
TreeNode* findNode(TreeNode* root, int val) {
    if (!root || root->val == val) return root;
    auto* left = findNode(root->left, val);
    return left ? left : findNode(root->right, val);
}
vector<int> inorderValues(TreeNode* root) {
    if (!root) return {};
    auto a = inorderValues(root->left), b = inorderValues(root->right);
    a.push_back(root->val); a.insert(a.end(), b.begin(), b.end()); return a;
}
int checkedHeight(TreeNode* root) {
    if (!root) return 0;
    int l = checkedHeight(root->left), r = checkedHeight(root->right);
    if (abs(l - r) > 1) throw runtime_error("tree is not balanced");
    return 1 + max(l, r);
}
vector<vector<int>> nextLevels(TreeNode* root) {
    vector<vector<int>> result;
    for (; root; root = root->left) {
        vector<int> row;
        for (auto* n = root; n; n = n->next) row.push_back(n->val);
        result.push_back(row);
    }
    return result;
}

// 中序遍历统一迭代法：左 -> 根 -> 右
    // 方法：inorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != nullptr) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != nullptr) {
                st.pop();
                if (node->right) st.push(node->right); // 右孩子先压，最后出栈
                st.push(node);                          // 根节点重新压回
                st.push(nullptr);                       // 紧跟一个标记，表示下次弹出即访问
                if (node->left) st.push(node->left);    // 左孩子最后压，最先出栈
            } else {
                st.pop();                    // 弹出标记
                result.push_back(st.top()->val); // 标记后面就是真正要访问的节点
                st.pop();
            }
        }
        return result;
    }

// 前序遍历统一迭代法：根 -> 左 -> 右
    // 方法：preorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != nullptr) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != nullptr) {
                st.pop();
                if (node->right) st.push(node->right); // 右孩子先压
                if (node->left) st.push(node->left);   // 左孩子次之
                st.push(node);                          // 根节点最后压，最先出栈访问
                st.push(nullptr);                       // 紧跟标记
            } else {
                st.pop();
                result.push_back(st.top()->val);
                st.pop();
            }
        }
        return result;
    }

// 后序遍历统一迭代法：左 -> 右 -> 根
    // 方法：postorderTraversal；按题目要求处理输入并返回结果，核心算法见方法体。
    vector<int> postorderTraversal(TreeNode* root) {
        vector<int> result;
        stack<TreeNode*> st;
        if (root != nullptr) st.push(root);
        while (!st.empty()) {
            TreeNode* node = st.top();
            if (node != nullptr) {
                st.pop();
                st.push(node);                          // 根节点先压（最后访问）
                st.push(nullptr);                       // 紧跟标记
                if (node->right) st.push(node->right);  // 右孩子次之
                if (node->left) st.push(node->left);    // 左孩子最后压，最先出栈
            } else {
                st.pop();
                result.push_back(st.top()->val);
                st.pop();
            }
        }
        return result;
    }

## 总结

- 统一迭代法的核心是给节点加一个"访问标记"：第一次遇到只入栈不访问，标记后紧跟着重新压回，第二次遇到（也就是弹出标记）的时候才真正去访问。
- 三种遍历顺序只需要调整"根、左、右"三者压栈的相对顺序，代码框架完全一致，不用再像中序那样单独引入额外指针，也不用像后序那样依赖"反转结果"的技巧。
- 压栈顺序跟出栈顺序是相反的：想要出栈顺序是"左、根、右"，压栈顺序就得反过来写成"右、根+标记、左"，这个地方要转个弯想一下。
- 这种写法虽然更统一，但引入了 `nullptr`/`None` 当标记，代码可读性上不如前一篇那种"专用写法"直观，实际面试中两种写法都可以选，理解思想比死记代码更重要。

# 0827. 最大人工岛

**题目描述**

给定一个 `n x n` 的二维网格 `grid`，每个格子的值是 `0`（水域）或 `1`（陆地）。规则规定：**最多可以选择一个海洋格子（值为 0），把它变成陆地（值为 1）**，求执行这个操作之后，能得到的最大岛屿面积（岛屿由上下左右相邻的陆地组成一片连通区域）。

需要注意两种边界情况：如果网格中已经没有任何海洋格子（全是陆地），说明无法再填海，直接返回整个网格的陆地总数；如果填了某个海洋格子后面积反而不比原来任何一个岛屿更大，也要正确地返回填海后能得到的最大值（而不是漏判）。

```
示例：
输入：
grid = [
  [1,0],
  [0,1]
]
输出：3
解释：网格里有两个独立的岛屿，面积都是 1。把 (0,1) 或 (1,0) 这个海洋格子填成陆地后，
可以把两个岛屿连起来，得到一个面积为 3 的岛屿（1 + 1 + 新填的这 1 格）。
```

## 思路

先用 DFS/BFS 给每个岛屿编号并记录面积；再枚举水格，将四邻不同岛屿的面积加一，取最大值。邻岛编号须去重，避免重复累计。

全陆地返回 n²，全水返回 1。时间、空间 O(n²)。

    # 方法：largestIsland；按题目要求处理输入并返回结果，核心算法见方法体。
    def largestIsland(self, grid: list[list[int]]) -> int:
        n = len(grid)
        area_of_island = {}  # 岛屿编号 -> 面积
        island_id = 2        # 编号从 2 开始，避免和原始的 0/1 混淆

                return 0  # 越界

            area += dfs(row - 1, col, island_id)  # 上
            area += dfs(row + 1, col, island_id)  # 下
            area += dfs(row, col - 1, island_id)  # 左
            area += dfs(row, col + 1, island_id)  # 右
            return area

        # 第一阶段：给每个岛屿编号并记录面积
        for i in range(n):
            for j in range(n):
                if grid[i][j] == 1:
                    area = dfs(i, j, island_id)
                    area_of_island[island_id] = area
                    island_id += 1

        max_area = max(area_of_island.values(), default=0)  # 先考虑不填海的情况

        dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]

        # 第二阶段：枚举每个海洋格子，计算填海后的面积
        for i in range(n):
            for j in range(n):
                if grid[i][j] != 0:
                    continue  # 只处理海洋格子
                neighbor_ids = set()  # 收集四周不同的岛屿编号，去重
                for dr, dc in dirs:
                    ni, nj = i + dr, j + dc
                    if ni < 0 or ni >= n or nj < 0 or nj >= n:
                        continue  # 越界跳过
                    if grid[ni][nj] == 0:
                        continue  # 仍是水域，跳过
                    neighbor_ids.add(grid[ni][nj])  # 记录这个方向的岛屿编号
                total = 1  # 填海本身贡献 1 格
                for island_id_ in neighbor_ids:
                    total += area_of_island[island_id_]  # 累加每个不同岛屿的面积
                max_area = max(max_area, total)

        return max_area

// 方法：largestIsland；按题目要求处理输入并返回结果，核心算法见方法体。
func largestIsland(grid [][]int) int {
    n := len(grid)
    areaOfIsland := make(map[int]int) // 岛屿编号 -> 面积
    islandId := 2                     // 编号从 2 开始，避免和原始的 0/1 混淆

    // 给 (row, col) 所在岛屿标记编号，同时统计面积
    var dfs func(row, col, id int) int
    dfs = func(row, col, id int) int {
        if row < 0 || row >= n || col < 0 || col >= n {
            return 0 // 越界
        }
        if grid[row][col] != 1 {
            return 0 // 不是待编号的陆地
        }
        grid[row][col] = id // 标记为当前岛屿编号
        area := 1
        area += dfs(row-1, col, id) // 上
        area += dfs(row+1, col, id) // 下
        area += dfs(row, col-1, id) // 左
        area += dfs(row, col+1, id) // 右
        return area
    }

    // 第一阶段：给每个岛屿编号并记录面积
    for i := 0; i < n; i++ {
        for j := 0; j < n; j++ {
            if grid[i][j] == 1 {
                area := dfs(i, j, islandId)
                areaOfIsland[islandId] = area
                islandId++
            }
        }
    }

    maxArea := 0
    for _, area := range areaOfIsland {
        if area > maxArea {
            maxArea = area // 先考虑不填海的情况
        }
    }

    dirs := [4][2]int{{-1, 0}, {1, 0}, {0, -1}, {0, 1}}

    // 第二阶段：枚举每个海洋格子，计算填海后的面积
    for i := 0; i < n; i++ {
        for j := 0; j < n; j++ {
            if grid[i][j] != 0 {
                continue // 只处理海洋格子
            }
            neighborIds := make(map[int]bool) // 收集四周不同的岛屿编号，去重
            for _, d := range dirs {
                ni, nj := i+d[0], j+d[1]
                if ni < 0 || ni >= n || nj < 0 || nj >= n {
                    continue // 越界跳过
                }
                if grid[ni][nj] == 0 {
                    continue // 仍是水域，跳过
                }
                neighborIds[grid[ni][nj]] = true // 记录这个方向的岛屿编号
            }
            total := 1 // 填海本身贡献 1 格
            for id := range neighborIds {
                total += areaOfIsland[id] // 累加每个不同岛屿的面积
            }
            if total > maxArea {
                maxArea = total
            }
        }
    }

    return maxArea
}

// 依赖说明：vector：C++ STL 动态数组。；unordered_map：C++ STL 哈希映射。；unordered_set：C++ STL 哈希集合。

class Solution {
public:
    // 方法：largestIsland；按题目要求处理输入并返回结果，核心算法见方法体。
    int largestIsland(vector<vector<int>>& grid) {
        int n = grid.size();
        unordered_map<int, int> areaOfIsland; // 岛屿编号 -> 面积
        int islandId = 2; // 编号从 2 开始，避免和原始的 0（水）、1（未编号陆地）混淆

        // 给 (row, col) 所在岛屿标记编号 id，同时统计面积
        function<int(int, int, int)> dfs = [&](int row, int col, int id) -> int {
            if (row < 0 || row >= n || col < 0 || col >= n) return 0; // 越界
            if (grid[row][col] != 1) return 0; // 不是待编号的陆地（水域或已编号的其他岛屿）
            grid[row][col] = id; // 标记为当前岛屿编号
            int area = 1; // 当前格子贡献 1 的面积
            area += dfs(row - 1, col, id); // 上
            area += dfs(row + 1, col, id); // 下
            area += dfs(row, col - 1, id); // 左
            area += dfs(row, col + 1, id); // 右
            return area;
        };

        // 第一阶段：给每个岛屿编号并记录面积
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1) {
                    int area = dfs(i, j, islandId);
                    areaOfIsland[islandId] = area;
                    islandId++;
                }
            }
        }

        int maxArea = 0;
        for (auto& [id, area] : areaOfIsland) {
            maxArea = max(maxArea, area); // 先考虑不填海的情况（网格里可能没有海洋格子）
        }

        int dirs[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

        // 第二阶段：枚举每个海洋格子，计算填海后的面积
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] != 0) continue; // 只处理海洋格子
                unordered_set<int> neighborIds; // 收集四周不同的岛屿编号，去重
                for (auto& d : dirs) {
                    int ni = i + d[0], nj = j + d[1];
                    if (ni < 0 || ni >= n || nj < 0 || nj >= n) continue; // 越界跳过
                    if (grid[ni][nj] == 0) continue; // 仍是水域，跳过
                    neighborIds.insert(grid[ni][nj]); // 记录这个方向的岛屿编号
                }
                int total = 1; // 填海本身贡献 1 格
                for (int id : neighborIds) {
                    total += areaOfIsland[id]; // 累加每个不同岛屿的面积（集合已去重，不会重复累加）
                }
                maxArea = max(maxArea, total);
            }
        }

        return maxArea;
    }
};```

    int largestIsland(vector<vector<int>>& grid) {
        int n = grid.size();
        unordered_map<int, int> areaOfIsland; // 岛屿编号 -> 面积
        int islandId = 2; // 编号从 2 开始，避免和原始的 0（水）、1（未编号陆地）混淆

        // 给 (row, col) 所在岛屿标记编号 id，同时统计面积
        function<int(int, int, int)> dfs = [&](int row, int col, int id) -> int {
            if (row < 0 || row >= n || col < 0 || col >= n) return 0; // 越界
            if (grid[row][col] != 1) return 0; // 不是待编号的陆地（水域或已编号的其他岛屿）
            grid[row][col] = id; // 标记为当前岛屿编号
            int area = 1; // 当前格子贡献 1 的面积
            area += dfs(row - 1, col, id); // 上
            area += dfs(row + 1, col, id); // 下
            area += dfs(row, col - 1, id); // 左
            area += dfs(row, col + 1, id); // 右
            return area;
        };

        // 第一阶段：给每个岛屿编号并记录面积
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1) {
                    int area = dfs(i, j, islandId);
                    areaOfIsland[islandId] = area;
                    islandId++;
                }
            }
        }

        int maxArea = 0;
        for (auto& [id, area] : areaOfIsland) {
            maxArea = max(maxArea, area); // 先考虑不填海的情况（网格里可能没有海洋格子）
        }

        int dirs[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

        // 第二阶段：枚举每个海洋格子，计算填海后的面积
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] != 0) continue; // 只处理海洋格子
                unordered_set<int> neighborIds; // 收集四周不同的岛屿编号，去重
                for (auto& d : dirs) {
                    int ni = i + d[0], nj = j + d[1];
                    if (ni < 0 || ni >= n || nj < 0 || nj >= n) continue; // 越界跳过
                    if (grid[ni][nj] == 0) continue; // 仍是水域，跳过
                    neighborIds.insert(grid[ni][nj]); // 记录这个方向的岛屿编号
                }
                int total = 1; // 填海本身贡献 1 格
                for (int id : neighborIds) {
                    total += areaOfIsland[id]; // 累加每个不同岛屿的面积（集合已去重，不会重复累加）
                }
                maxArea = max(maxArea, total);
            }
        }

        return maxArea;
    }

}

int main() {
    // 用例 1：输入 ([[1, 0], [0, 1]],)；期望 3。
    {
        vector<vector<int>> arg0 = {{1, 0}, {0, 1}};
        auto result = interview::largestIsland(arg0);
        cout << "用例 1: 期望=3 实际=";
        show(result);
        cout << "\n";
    }
    // 用例 2：输入 ([[1, 1], [1, 1]],)；期望 4。
    {
        vector<vector<int>> arg0 = {{1, 1}, {1, 1}};
        auto result = interview::largestIsland(arg0);
        cout << "用例 2: 期望=4 实际=";
        show(result);
        cout << "\n";
    }
    // 用例 3：输入 ([[0, 0], [0, 0]],)；期望 1。
    {
        vector<vector<int>> arg0 = {{0, 0}, {0, 0}};
        auto result = interview::largestIsland(arg0);
        cout << "用例 3: 期望=1 实际=";
        show(result);
        cout << "\n";
    }
    return 0;
}

## 总结

- 这道题是"岛屿最大面积"的进阶版，多了一步"填一格海洋，最多能连通多少个已有岛屿"，如果每次真的修改网格再重新统计，会导致复杂度退化到 O(n⁴)，必须用"预处理编号 + 枚举"的两阶段思路把复杂度控制在 O(n²)。
- 给岛屿分配唯一编号（而不是简单的布尔标记）是这道题的关键：只有编号能让"枚举海洋格子时判断四周是否属于同一岛屿"这件事变得可行，否则就没法区分不同方向的陆地是不是同一片，容易重复累加面积。
- 用集合对相邻岛屿编号去重，是避免"同一岛屿从多个方向被重复累加面积"这一常见错误的核心手段，体现了图论题里"先去重再累加"的通用处理套路。
- 必须单独处理"网格中没有海洋格子"的边界情况：这种情况下枚举阶段不会产生任何候选值，如果不特殊处理会漏判，正确做法是让 `maxArea` 在枚举前先取一次"所有已有岛屿面积的最大值"（这时候如果网格全是陆地，恰好等于 `n*n`，逻辑自然就覆盖了这个边界）。

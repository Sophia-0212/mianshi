# 0417. 太平洋大西洋水流问题

**题目描述**

给定一个 `m x n` 的二维矩阵 `heights`，`heights[r][c]` 表示坐标 `(r, c)` 处的地面高度。规定水流规则：**水只能从高处流向相邻的、高度小于或等于当前高度的地方**（可以流向更低处，也可以流向相同高度，但绝不能流向更高处）。

网格的左边界和上边界被认为与"太平洋"相邻，右边界和下边界被认为与"大西洋"相邻。请找出所有满足下面条件的坐标：从这个坐标出发的水，既能流到太平洋，也能流到大西洋。

```
示例：
输入：
heights = [
  [1,2,2,3,5],
  [3,2,3,4,4],
  [2,4,5,3,1],
  [6,7,1,4,5],
  [5,1,1,2,4]
]
输出：[[0,4],[1,3],[1,4],[2,2],[3,0],[3,1],[4,0]]
```

以坐标 `(0,4)` 为例：你看啊，它在最上面一行，天然就跟太平洋相邻了，同时它高度是 5，往下能流到高度更低（或者相等）的格子，最终到达右边界，从而流到大西洋，所以它是同时满足两个条件的。

## 思路

- 反向多源 DFS/BFS：分别从太平洋的上／左边界、大西洋的下／右边界出发，只向高度不低于当前格的邻居扩展。两次标记的交集即答案。时间、空间 O(mn)。
- 正向逐格搜索：每格单独判断能否流向两海。最坏时间 O(m²n²)，空间 O(mn)。

### LeetCode 写法与原有示例

这种图论题其实没有什么标准库函数能一行解决的，所以下面就直接给手写的 DFS 实现：

```python
# 依赖说明：len：Python 内置函数：返回容器中元素的数量。；range：Python 内置函数：生成按步长遍历的整数序列。

class Solution:
    # 方法：pacificAtlantic；按题目要求处理输入并返回结果，核心算法见方法体。
    def pacificAtlantic(self, heights: list[list[int]]) -> list[list[int]]:
        if not heights or not heights[0]:
            return []
        m, n = len(heights), len(heights[0])
        can_pacific = [[False] * n for _ in range(m)]   # 能流到太平洋的标记
        can_atlantic = [[False] * n for _ in range(m)]   # 能流到大西洋的标记
        dirs = [(-1, 0), (1, 0), (0, -1), (0, 1)]  # 上下左右

        # 方法：dfs；按题目要求处理输入并返回结果，核心算法见方法体。
        def dfs(row: int, col: int, visited: list[list[bool]]) -> None:
            visited[row][col] = True  # 标记当前格子已确定能到达对应大洋
            for dr, dc in dirs:
                nr, nc = row + dr, col + dc
                if nr < 0 or nr >= m or nc < 0 or nc >= n:
                    continue  # 越界跳过
                if visited[nr][nc]:
                    continue  # 已标记过，跳过
                if heights[nr][nc] < heights[row][col]:
                    continue  # 反向条件：邻居高度必须 >= 当前格子
                dfs(nr, nc, visited)

        # 从太平洋边界（最上行、最左列）出发做多源 DFS
        for i in range(m):
            dfs(i, 0, can_pacific)
        for j in range(n):
            dfs(0, j, can_pacific)

        # 从大西洋边界（最下行、最右列）出发做多源 DFS
        for i in range(m):
            dfs(i, n - 1, can_atlantic)
        for j in range(n):
            dfs(m - 1, j, can_atlantic)

        # 取两个标记数组的交集，即为答案
        result = []
        for i in range(m):
            for j in range(n):
                if can_pacific[i][j] and can_atlantic[i][j]:
                    result.append([i, j])
        return result
```

```go
func pacificAtlantic(heights [][]int) [][]int {
    m, n := len(heights), len(heights[0])
    canPacific := make([][]bool, m)  // 能流到太平洋的标记
    canAtlantic := make([][]bool, m) // 能流到大西洋的标记
    for i := range canPacific {
        canPacific[i] = make([]bool, n)
        canAtlantic[i] = make([]bool, n)
    }
    dirs := [4][2]int{{-1, 0}, {1, 0}, {0, -1}, {0, 1}} // 上下左右

    // 反向 DFS：只走向高度 >= 当前格子的相邻格子
    var dfs func(row, col int, visited [][]bool)
    dfs = func(row, col int, visited [][]bool) {
        visited[row][col] = true // 标记当前格子已确定能到达对应大洋
        for _, d := range dirs {
            nr, nc := row+d[0], col+d[1]
            if nr < 0 || nr >= m || nc < 0 || nc >= n {
                continue // 越界跳过
            }
            if visited[nr][nc] {
                continue // 已标记过，跳过
            }
            if heights[nr][nc] < heights[row][col] {
                continue // 反向条件：邻居高度必须 >= 当前格子
            }
            dfs(nr, nc, visited)
        }
    }

    // 从太平洋边界（最上行、最左列）出发做多源 DFS
    for i := 0; i < m; i++ {
        dfs(i, 0, canPacific)
    }
    for j := 0; j < n; j++ {
        dfs(0, j, canPacific)
    }

    // 从大西洋边界（最下行、最右列）出发做多源 DFS
    for i := 0; i < m; i++ {
        dfs(i, n-1, canAtlantic)
    }
    for j := 0; j < n; j++ {
        dfs(m-1, j, canAtlantic)
    }

    // 取两个标记数组的交集，即为答案
    result := [][]int{}
    for i := 0; i < m; i++ {
        for j := 0; j < n; j++ {
            if canPacific[i][j] && canAtlantic[i][j] {
                result = append(result, []int{i, j})
            }
        }
    }
    return result
}
```

### LeetCode 写法与原有示例

// 依赖说明：vector：C++ STL 动态数组。

class Solution {
public:
    vector<vector<int>> pacificAtlantic(vector<vector<int>>& heights) {
        int m = heights.size(), n = heights[0].size();
        vector<vector<bool>> canPacific(m, vector<bool>(n, false)); // 能流到太平洋的标记
        vector<vector<bool>> canAtlantic(m, vector<bool>(n, false)); // 能流到大西洋的标记

        // 方向数组：上下左右
        int dirs[4][2] = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

        // 反向 DFS：只走向高度 >= 当前格子的相邻格子（因为要模拟"谁能流到我这里"）
        function<void(int, int, vector<vector<bool>>&)> dfs = [&](int row, int col, vector<vector<bool>>& visited) {
            visited[row][col] = true; // 标记当前格子已确定能到达对应大洋
            for (auto& d : dirs) {
                int nr = row + d[0], nc = col + d[1];
                if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue; // 越界跳过
                if (visited[nr][nc]) continue; // 已标记过，跳过，避免重复访问
                if (heights[nr][nc] < heights[row][col]) continue; // 反向条件：邻居高度必须 >= 当前格子
                dfs(nr, nc, visited);
            }
        };

        // 从太平洋边界（最上行、最左列）出发做多源 DFS
        for (int i = 0; i < m; i++) dfs(i, 0, canPacific);
        for (int j = 0; j < n; j++) dfs(0, j, canPacific);

        // 从大西洋边界（最下行、最右列）出发做多源 DFS
        for (int i = 0; i < m; i++) dfs(i, n - 1, canAtlantic);
        for (int j = 0; j < n; j++) dfs(m - 1, j, canAtlantic);

        // 取两个标记数组的交集，即为答案
        vector<vector<int>> result;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (canPacific[i][j] && canAtlantic[i][j]) {
                    result.push_back({i, j});
                }
            }
        }
        return result;
    }
};```

```cpp
    }
    return 0;
}
```

## 总结

- 这道题最大的坑，就是很容易想到"正向枚举每个格子，分别判断能不能到达边界"，这个思路虽然直观，但复杂度是 O(m²n²)，网格稍微大一点就会超时。解题的关键其实是想到反过来，从边界出发做多源搜索。
- 反向搜索说白了就是图论里"反图"的思想：原图的边方向是"从高处指向不高于自己的相邻格子"，那要算"谁能到达边界集合"，就等价于在反图上算"边界集合能到达谁"，反图的边方向正好是反过来的，体现在代码里，就是把移动条件的大小关系反过来了——原本是"走向小于等于自己"，反向就变成了"走向大于等于自己"。
- 多源 DFS/BFS 效率上的优势，其实来自于"标记去重"这个机制：不管起点有多少个，只要维护一个全局的标记数组，每个格子最多也就被访问一次。所以从 O(m+n) 个边界起点出发的搜索，总代价还是 O(mn)，而不是"起点数乘以单次搜索代价"这么算。
- 最终答案就是两次独立标记结果的交集，这其实提示了一种挺通用的解题模式：当问题要求"同时满足两个独立条件"的时候，你可以把两个条件分别求出各自的可行解集合，最后再取交集，不用非得在一次搜索里同时判断两个条件。

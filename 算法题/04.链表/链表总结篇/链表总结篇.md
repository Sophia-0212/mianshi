# 链表总结篇

链表模式复习与普通函数实践。

## 题意与思路

虚拟头节点统一边界；快慢指针找中点或判环；固定间距找倒数节点；相交比较节点身份而非值；反转前保存后继。下面以迭代反转复习指针更新顺序：保存 next → 当前节点指向 prev → 推进两指针。

迭代反转时间 O(n)，辅助空间 O(1)；测试构造/输出另用 O(n)。

## 输入约定与测试

输入无环链表；空链表、单节点直接成立。递归反转需 O(n) 栈；不用于带环链表。

- 正常：`[[1, 2, 3]]` → `[3, 2, 1]`。
- 空链表：`[[]]` → `[]`。
- 单节点：`[[7]]` → `[7]`。
- 重复值：`[[1, 1]]` → `[1, 1]`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
class Node:

    def __init__(self, value=0, next=None):
        (self.value, self.next) = (value, next)

def build(values):
    dummy = tail = Node()
    for value in values:
        tail.next = Node(value)
        tail = tail.next
    return dummy.next

def to_values(head):
    result = []
    while head:
        result.append(head.value)
        head = head.next
    return result

def reverseList(head):
    prev = None
    while head:
        following = head.next
        head.next = prev
        prev = head
        head = following
    return prev

def solve(values):
    return to_values(reverseList(build(values)))
if __name__ == '__main__':
    actual = solve([1, 2, 3])
    expected = [3, 2, 1]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = []
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([7])
    expected = [7]
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, 1])
    expected = [1, 1]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

type Node struct {
	value int
	next  *Node
}

func build(values []int) *Node {
	dummy := &Node{}
	tail := dummy
	for _, v := range values {
		tail.next = &Node{value: v}
		tail = tail.next
	}
	return dummy.next
}
func toValues(head *Node) []int {
	result := []int{}
	for ; head != nil; head = head.next {
		result = append(result, head.value)
	}
	return result
}

func reverseList(head *Node) *Node {
	var prev *Node
	for head != nil {
		following := head.next
		head.next = prev
		prev = head
		head = following
	}
	return prev
}
func solve(values []int) []int { return toValues(reverseList(build(values))) }

func main() {
	{ // 正常
		actual := solve([]int{1, 2, 3})
		expected := []int{3, 2, 1}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空链表
		actual := solve([]int{})
		expected := []int{}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单节点
		actual := solve([]int{7})
		expected := []int{7}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复值
		actual := solve([]int{1, 1})
		expected := []int{1, 1}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
}
```

## C++

保存为 `main.cpp`，运行 `c++ -std=c++17 main.cpp -o main && ./main`。

```cpp
#include <cassert>
#include <iostream>
#include <memory>
#include <vector>

using namespace std;
struct Node {
    int value;
    Node* next=nullptr;
    explicit Node(int v=0):value(v) {}
};
// arena 只管理所有权，链表算法仍通过真实 next 指针操作。
Node* build(const vector<int>& values,vector<unique_ptr<Node>>& arena) {
    Node dummy;
    Node* tail=&dummy;
    for(int v:values) {
        arena.push_back(make_unique<Node>(v));
        tail->next=arena.back().get();
        tail=tail->next;
    }
    return dummy.next;
}
vector<int> toValues(Node* head) {
    vector<int> result;
    for(;head;head=head->next)result.push_back(head->value);
    return result;
}
Node* reverseList(Node* head) {
    Node* prev=nullptr;
    while(head) {
        auto* following=head->next;
        head->next=prev;
        prev=head;
        head=following;
    }
    return prev;
}
vector<int> solve(const vector<int>& values) {
    vector<unique_ptr<Node>> arena;
    return toValues(reverseList(build(values,arena)));
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常
        auto actual = solve(vector<int> {1, 2, 3});
        auto expected = vector<int> {3, 2, 1};
        cout << "用例 1：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空链表
        auto actual = solve(vector<int> {});
        auto expected = vector<int> {};
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 单节点
        auto actual = solve(vector<int> {7});
        auto expected = vector<int> {7};
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 重复值
        auto actual = solve(vector<int> {1, 1});
        auto expected = vector<int> {1, 1};
        cout << "用例 4：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    }
}
```

# 0234. 回文链表

**题目描述**

给定一个单链表的头节点 `head`，判断该链表是否为回文链表。回文的含义是：从前往后读和从后往前读，序列完全一致。

```
示例：
输入: head = [1,2,2,1]
输出: true

输入: head = [1,2]
输出: false
```

## 思路

- 快慢指针＋反转：找中点，反转后半段，与前半段逐项比较；比较后可再次反转恢复链表。时间 O(n)，空间 O(1)。
- 数组：保存节点值，双指针比较首尾。时间 O(n)，空间 O(n)。



## Python

### LeetCode 写法与原有示例

标准库思路说明：Python 没有直接针对链表回文判断的库函数，不过如果先把链表值收集成一个列表，就可以用切片反转 `lst == lst[::-1]` 这种简洁写法快速验证回文，原理是切片 `[::-1]` 会生成一个反向拷贝的新列表，再跟原列表逐元素比较就行。这个写法说白了本质上还是 O(n) 空间的数组法，适合追求代码简洁、但不追求空间最优的场景：

```python
class Solution:
    # 方法：isPalindrome；按题目要求处理输入并返回结果，核心算法见方法体。
    def isPalindrome(self, head: 'Optional[ListNode]') -> bool:
        vals = []
        node = head
        while node:               # 遍历链表，收集所有节点值
            vals.append(node.val)
            node = node.next
        return vals == vals[::-1]  # 与反转后的列表比较，等价于回文判断
```


手写实现（这是面试的考察点，O(1) 空间的快慢指针 + 反转 + 双指针比较）：

```python
class ListNode:
    # 方法：__init__；按题目要求处理输入并返回结果，核心算法见方法体。
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

class Solution:
    # 方法：isPalindrome；按题目要求处理输入并返回结果，核心算法见方法体。
    def isPalindrome(self, head: 'Optional[ListNode]') -> bool:
        if not head or not head.next:  # 空链表或单节点必为回文
            return True

        # 第一步：快慢指针找中点
        slow, fast = head, head
        while fast.next and fast.next.next:
            slow = slow.next        # 慢指针走一步
            fast = fast.next.next   # 快指针走两步

        # 第二步：反转后半部分链表
        second_head = self._reverse(slow.next)

        # 第三步：双指针比较前半部分与反转后的后半部分
        p1, p2 = head, second_head
        result = True
        while p2:                   # 以较短的后半部分为终止条件
            if p1.val != p2.val:
                result = False      # 发现不相等
                break
            p1 = p1.next
            p2 = p2.next

        slow.next = self._reverse(second_head)  # 还原链表
        return result

    # 方法：_reverse；按题目要求处理输入并返回结果，核心算法见方法体。
    def _reverse(self, head):
        prev = None
        while head:
            nxt = head.next   # 暂存下一节点
            head.next = prev  # 反转指向
            prev = head
            head = nxt
        return prev
```

```go
// 方法：isPalindrome；按题目要求处理输入并返回结果，核心算法见方法体。
func isPalindrome(head *ListNode) bool {
	if head == nil || head.Next == nil {
		return true // 空链表或单节点必为回文
	}

	// 第一步：快慢指针找中点
	slow, fast := head, head
	for fast.Next != nil && fast.Next.Next != nil {
		slow = slow.Next       // 慢指针走一步
		fast = fast.Next.Next  // 快指针走两步
	}

	// 第二步：反转后半部分链表
	secondHead := reverse(slow.Next)

	// 第三步：双指针比较前半部分与反转后的后半部分
	p1, p2 := head, secondHead
	result := true
	for p2 != nil { // 以较短的后半部分为终止条件
		if p1.Val != p2.Val {
			result = false // 发现不相等
			break
		}
		p1 = p1.Next
		p2 = p2.Next
	}

	slow.Next = reverse(secondHead) // 还原链表
	return result
}

// 方法：reverse；按题目要求处理输入并返回结果，核心算法见方法体。
func reverse(head *ListNode) *ListNode {
	var prev *ListNode
	for head != nil {
		next := head.Next // 暂存下一节点
		head.Next = prev  // 反转指向
		prev = head
		head = next
	}
	return prev
}
```

### LeetCode 写法与原有示例

```cpp
/**
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(nullptr) {}
 * };
 */
class Solution {
public:
    // 方法：isPalindrome；按题目要求处理输入并返回结果，核心算法见方法体。
    bool isPalindrome(ListNode* head) {
        if (head == nullptr || head->next == nullptr) return true; // 空链表或单节点必为回文

        // 第一步：快慢指针找中点
        ListNode *slow = head, *fast = head;
        while (fast->next != nullptr && fast->next->next != nullptr) {
            slow = slow->next;       // 慢指针走一步
            fast = fast->next->next; // 快指针走两步
        }
        // 此时 slow 停在前半部分的最后一个节点（奇数长度时是中间节点）

        // 第二步：反转后半部分链表
        ListNode* secondHead = reverse(slow->next);

        // 第三步：双指针比较前半部分与反转后的后半部分
        ListNode *p1 = head, *p2 = secondHead;
        bool result = true;
        while (p2 != nullptr) { // 以较短的后半部分为终止条件
            if (p1->val != p2->val) {
                result = false; // 发现不相等，标记为非回文
                break;
            }
            p1 = p1->next;
            p2 = p2->next;
        }

        slow->next = reverse(secondHead); // 还原链表，恢复原始结构
        return result;
    }

private:
    ListNode* reverse(ListNode* head) {
        ListNode* prev = nullptr;
        while (head != nullptr) {
            ListNode* next = head->next; // 暂存下一个节点
            head->next = prev;           // 当前节点指向前一个节点，完成反转
            prev = head;
            head = next;
        }
        return prev; // prev 是反转后的新头节点
    }
};
```

## 总结

- 判断回文的这两种思路，本质上是空间换时间的权衡：数组法用 O(n) 空间换来实现简单，快慢指针加反转的方法用 O(1) 空间，但需要更精细的指针操作。面试中优先展示后者更能体现链表功底。
- 找中点、反转链表、双指针比较，这三个操作各自都是独立的高频考点，这道题把它们组合在一起，是检验综合运用链表基本操作能力的典型题目。
- 反转后半部分链表会破坏原链表的结构，如果题目要求不能修改链表，就需要在比较完成之后再反转回去还原，这个细节很容易被忽略。
- 找中点时循环条件怎么写（`fast != nullptr && fast->next != nullptr` 还是 `fast->next != nullptr && fast->next->next != nullptr`）会影响奇偶长度下慢指针最终停留的位置，写代码之前得先想清楚要哪种停留位置，再决定用哪种循环条件。

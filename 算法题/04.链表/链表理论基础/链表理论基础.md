# 链表理论基础

链表模式复习与普通函数实践。

## 题意与思路

节点通过 next 指针连接，不要求连续内存。按下标访问 O(n)；已知前驱后的插入/删除 O(1)。虚拟头节点统一头部操作。下面用普通函数实现按位置插入与删除。

单次定位 O(n)，指针重连 O(1)；链表空间 O(n)。输入输出数组仅用于构造与验证。

## 输入约定与测试

输入无环链表值数组；操作 [0,index,value] 插入、[1,index] 删除。插入允许 index==长度；负下标、越界或未知操作拒绝。

- 正常插入删除：`[[1, 3], [[0, 1, 2], [1, 0]]]` → `[2, 3]`。
- 空链表头插：`[[], [[0, 0, 7]]]` → `[7]`。
- 删除唯一节点：`[[1], [[1, 0]]]` → `[]`。
- 空操作：`[[], []]` → `[]`。
- 越界删除：`[[1], [[1, 1]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
class Node:

    def __init__(self, value=0, next=None):
        (self.value, self.next) = (value, next)

def build(values):
    dummy = tail = Node()
    for value in values:
        tail.next = Node(value)
        tail = tail.next
    return dummy.next

def to_values(head):
    result = []
    while head:
        result.append(head.value)
        head = head.next
    return result

def insertAt(head, index, value):
    if index < 0:
        raise ValueError('负下标')
    dummy = Node(next=head)
    prev = dummy
    for _ in range(index):
        if prev.next is None:
            raise ValueError('插入越界')
        prev = prev.next
    prev.next = Node(value, prev.next)
    return dummy.next

def deleteAt(head, index):
    if index < 0:
        raise ValueError('负下标')
    dummy = Node(next=head)
    prev = dummy
    for _ in range(index):
        if prev.next is None:
            raise ValueError('删除越界')
        prev = prev.next
    if prev.next is None:
        raise ValueError('删除越界')
    prev.next = prev.next.next
    return dummy.next

def solve(values, operations):
    head = build(values)
    for op in operations:
        if not op or op[0] not in (0, 1) or len(op) != (3 if op[0] == 0 else 2):
            raise ValueError('非法操作')
        head = insertAt(head, op[1], op[2]) if op[0] == 0 else deleteAt(head, op[1])
    return to_values(head)
if __name__ == '__main__':
    actual = solve([1, 3], [[0, 1, 2], [1, 0]])
    expected = [2, 3]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], [[0, 0, 7]])
    expected = [7]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1], [[1, 0]])
    expected = []
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], [])
    expected = []
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([1], [[1, 1]])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

type Node struct {
	value int
	next  *Node
}

func build(values []int) *Node {
	dummy := &Node{}
	tail := dummy
	for _, v := range values {
		tail.next = &Node{value: v}
		tail = tail.next
	}
	return dummy.next
}
func toValues(head *Node) []int {
	result := []int{}
	for ; head != nil; head = head.next {
		result = append(result, head.value)
	}
	return result
}

func insertAt(head *Node, index, value int) *Node {
	if index < 0 {
		panic("负下标")
	}
	dummy := &Node{next: head}
	prev := dummy
	for i := 0; i < index; i++ {
		if prev.next == nil {
			panic("插入越界")
		}
		prev = prev.next
	}
	prev.next = &Node{value: value, next: prev.next}
	return dummy.next
}
func deleteAt(head *Node, index int) *Node {
	if index < 0 {
		panic("负下标")
	}
	dummy := &Node{next: head}
	prev := dummy
	for i := 0; i < index; i++ {
		if prev.next == nil {
			panic("删除越界")
		}
		prev = prev.next
	}
	if prev.next == nil {
		panic("删除越界")
	}
	prev.next = prev.next.next
	return dummy.next
}
func solve(values []int, operations [][]int) []int {
	head := build(values)
	for _, op := range operations {
		if len(op) == 0 || (op[0] != 0 && op[0] != 1) {
			panic("非法操作")
		}
		if (op[0] == 0 && len(op) != 3) || (op[0] == 1 && len(op) != 2) {
			panic("非法参数")
		}
		if op[0] == 0 {
			head = insertAt(head, op[1], op[2])
		} else {
			head = deleteAt(head, op[1])
		}
	}
	return toValues(head)
}

func main() {
	{ // 正常插入删除
		actual := solve([]int{1, 3}, [][]int{[]int{0, 1, 2}, []int{1, 0}})
		expected := []int{2, 3}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空链表头插
		actual := solve([]int{}, [][]int{[]int{0, 0, 7}})
		expected := []int{7}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 删除唯一节点
		actual := solve([]int{1}, [][]int{[]int{1, 0}})
		expected := []int{}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空操作
		actual := solve([]int{}, [][]int{})
		expected := []int{}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([]int{1}, [][]int{[]int{1, 1}})
	}()
}
```

## C++

保存为 `main.cpp`，运行 `c++ -std=c++17 main.cpp -o main && ./main`。

```cpp
#include <cassert>
#include <iostream>
#include <memory>
#include <stdexcept>
#include <vector>

using namespace std;
struct Node {
    int value;
    Node* next=nullptr;
    explicit Node(int v=0):value(v) {}
};
// arena 只管理所有权，链表算法仍通过真实 next 指针操作。
Node* build(const vector<int>& values,vector<unique_ptr<Node>>& arena) {
    Node dummy;
    Node* tail=&dummy;
    for(int v:values) {
        arena.push_back(make_unique<Node>(v));
        tail->next=arena.back().get();
        tail=tail->next;
    }
    return dummy.next;
}
vector<int> toValues(Node* head) {
    vector<int> result;
    for(;head;head=head->next)result.push_back(head->value);
    return result;
}
Node* insertAt(Node* head,int index,int value,vector<unique_ptr<Node>>& arena) {
    if(index<0)throw invalid_argument("negative index");
    Node dummy;
    dummy.next=head;
    Node* prev=&dummy;
    for(int i=0;i<index;++i) {
        if(!prev->next)throw invalid_argument("out of range");
        prev=prev->next;
    }
    arena.push_back(make_unique<Node>(value));
    auto* node=arena.back().get();
    node->next=prev->next;
    prev->next=node;
    return dummy.next;
}
Node* deleteAt(Node* head,int index) {
    if(index<0)throw invalid_argument("negative index");
    Node dummy;
    dummy.next=head;
    Node* prev=&dummy;
    for(int i=0;i<index;++i) {
        if(!prev->next)throw invalid_argument("out of range");
        prev=prev->next;
    }
    if(!prev->next)throw invalid_argument("out of range");
    prev->next=prev->next->next;
    return dummy.next;
}
vector<int> solve(const vector<int>& values,const vector<vector<int>>& operations) {
    vector<unique_ptr<Node>> arena;
    auto* head=build(values,arena);
    for(auto& op:operations) {
        if(op.empty()||(op[0]!=0&&op[0]!=1)||op.size()!=(op[0]==0?3U:2U))throw invalid_argument("bad operation");
        head=op[0]==0?insertAt(head,op[1],op[2],arena):deleteAt(head,op[1]);
    }
    return toValues(head);
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常插入删除
        auto actual = solve(vector<int> {1, 3}, vector<vector<int>> {
            vector<int> {0, 1, 2}, vector<int> {1, 0}
        });
        auto expected = vector<int> {2, 3};
        cout << "用例 1：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空链表头插
        auto actual = solve(vector<int> {}, vector<vector<int>> {
            vector<int> {0, 0, 7}
        });
        auto expected = vector<int> {7};
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 删除唯一节点
        auto actual = solve(vector<int> {1}, vector<vector<int>> {
            vector<int> {1, 0}
        });
        auto expected = vector<int> {};
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空操作
        auto actual = solve(vector<int> {}, vector<vector<int>> {});
        auto expected = vector<int> {};
        cout << "用例 4：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        bool rejected=false;
        try {
            solve(vector<int> {1}, vector<vector<int>> {
                vector<int> {1, 1}
            });
        } catch(const invalid_argument&) {
            rejected=true;
        }
        assert(rejected);
        cout << "用例 5：已拒绝非法输入\n";
    }
}
```

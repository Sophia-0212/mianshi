# 0460. LFU 缓存

[LeetCode 原题](https://leetcode.com/problems/lfu-cache/)

## 题意与思路

维护 key→值与频率、频率→LRU 链表、最小频率 minFreq。命中后从旧频次桶移到新桶；淘汰最小频次桶内最旧的键。新键频率为 1，更新已有键也加频率。

get/put 平均 O(1)，状态空间 O(capacity)。题目不要求计数衰减。

## 输入约定与测试

操作编码：`[0,key]` 为 get，`[1,key,value]` 为 put；返回所有 get 结果，未命中为 -1。容量 0 不存储，负容量或非法操作拒绝。

- 访问后淘汰：`[2, [[1, 1, 1], [1, 2, 2], [0, 1], [1, 3, 3], [0, 2], [0, 3]]]` → `[1, -1, 3]`。
- 零容量：`[0, [[1, 1, 1], [0, 1]]]` → `[-1]`。
- 空操作：`[1, []]` → `[]`。
- 更新同一键：`[1, [[1, 1, 1], [1, 1, 9], [0, 1]]]` → `[9]`。
- 负容量：`[-1, []]` → `拒绝非法输入`。
- 非法操作：`[1, [[0]]]` → `拒绝非法输入`。
- 同频按 LRU：`[2, [[1, 1, 10], [1, 2, 20], [0, 1], [0, 2], [1, 3, 30], [0, 1], [0, 2], [0, 3]]]` → `[10, 20, -1, 20, 30]`。
- 高频保留：`[2, [[1, 1, 10], [0, 1], [0, 1], [1, 2, 20], [1, 3, 30], [0, 1], [0, 2]]]` → `[10, 10, 10, -1]`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
from collections import OrderedDict

def solve(capacity, operations):
    if capacity < 0:
        raise ValueError('负容量')
    values = {}
    frequencies = {}
    buckets = {}
    minimum = 0
    answers = []

    def touch(key):
        nonlocal minimum
        f = frequencies[key]
        del buckets[f][key]
        if not buckets[f]:
            del buckets[f]
            if minimum == f:
                minimum = f + 1
        frequencies[key] = f + 1
        buckets.setdefault(f + 1, OrderedDict())[key] = None
    for op in operations:
        if not op or op[0] not in (0, 1) or len(op) != (2 if op[0] == 0 else 3):
            raise ValueError('非法操作')
        key = op[1]
        if op[0] == 0:
            if key not in values:
                answers.append(-1)
            else:
                touch(key)
                answers.append(values[key])
        elif capacity:
            if key in values:
                values[key] = op[2]
                touch(key)
            else:
                if len(values) == capacity:
                    (old, _) = buckets[minimum].popitem(last=False)
                    del values[old]
                    del frequencies[old]
                    if not buckets[minimum]:
                        del buckets[minimum]
                values[key] = op[2]
                frequencies[key] = 1
                minimum = 1
                buckets.setdefault(1, OrderedDict())[key] = None
    return answers
if __name__ == '__main__':
    actual = solve(2, [[1, 1, 1], [1, 2, 2], [0, 1], [1, 3, 3], [0, 2], [0, 3]])
    expected = [1, -1, 3]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(0, [[1, 1, 1], [0, 1]])
    expected = [-1]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(1, [])
    expected = []
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(1, [[1, 1, 1], [1, 1, 9], [0, 1]])
    expected = [9]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve(-1, [])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    try:
        solve(1, [[0]])
    except (ValueError, IndexError):
        print('用例 6：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    actual = solve(2, [[1, 1, 10], [1, 2, 20], [0, 1], [0, 2], [1, 3, 30], [0, 1], [0, 2], [0, 3]])
    expected = [10, 20, -1, 20, 30]
    print('用例 7：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(2, [[1, 1, 10], [0, 1], [0, 1], [1, 2, 20], [1, 3, 30], [0, 1], [0, 2]])
    expected = [10, 10, 10, -1]
    print('用例 8：期望', expected, '实际', actual)
    assert actual == expected
```

## Go

```go
package main

import (
	"container/list"
	"fmt"
	"reflect"
)

type Entry struct {
	value, freq int
	position    *list.Element
}

func solve(capacity int, operations [][]int) []int {
	if capacity < 0 {
		panic("负容量")
	}
	data := map[int]*Entry{}
	buckets := map[int]*list.List{}
	minimum := 0
	ans := []int{}
	bucket := func(f int) *list.List {
		if buckets[f] == nil {
			buckets[f] = list.New()
		}
		return buckets[f]
	}
	touch := func(key int) {
		e := data[key]
		f := e.freq
		buckets[f].Remove(e.position)
		if buckets[f].Len() == 0 {
			delete(buckets, f)
			if minimum == f {
				minimum = f + 1
			}
		}
		e.freq++
		e.position = bucket(e.freq).PushBack(key)
	}
	for _, op := range operations {
		if len(op) == 0 || (op[0] != 0 && op[0] != 1) {
			panic("非法操作")
		}
		if (op[0] == 0 && len(op) != 2) || (op[0] == 1 && len(op) != 3) {
			panic("非法参数")
		}
		key := op[1]
		e := data[key]
		if op[0] == 0 {
			if e == nil {
				ans = append(ans, -1)
			} else {
				touch(key)
				ans = append(ans, e.value)
			}
		} else if capacity > 0 {
			if e != nil {
				e.value = op[2]
				touch(key)
			} else {
				if len(data) == capacity {
					q := buckets[minimum]
					old := q.Front()
					delete(data, old.Value.(int))
					q.Remove(old)
					if q.Len() == 0 {
						delete(buckets, minimum)
					}
				}
				data[key] = &Entry{op[2], 1, bucket(1).PushBack(key)}
				minimum = 1
			}
		}
	}
	return ans
}

func main() {
	{ // 访问后淘汰
		actual := solve(2, [][]int{[]int{1, 1, 1}, []int{1, 2, 2}, []int{0, 1}, []int{1, 3, 3}, []int{0, 2}, []int{0, 3}})
		expected := []int{1, -1, 3}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 零容量
		actual := solve(0, [][]int{[]int{1, 1, 1}, []int{0, 1}})
		expected := []int{-1}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空操作
		actual := solve(1, [][]int{})
		expected := []int{}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 更新同一键
		actual := solve(1, [][]int{[]int{1, 1, 1}, []int{1, 1, 9}, []int{0, 1}})
		expected := []int{9}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve(-1, [][]int{})
	}()
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 6：已拒绝非法输入")
		}()
		solve(1, [][]int{[]int{0}})
	}()
	{ // 同频按 LRU
		actual := solve(2, [][]int{[]int{1, 1, 10}, []int{1, 2, 20}, []int{0, 1}, []int{0, 2}, []int{1, 3, 30}, []int{0, 1}, []int{0, 2}, []int{0, 3}})
		expected := []int{10, 20, -1, 20, 30}
		fmt.Printf("用例 7：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
```

```cpp
#include <algorithm>
#include <cassert>
#include <iostream>
#include <list>
#include <stdexcept>
#include <unordered_map>
#include <vector>
#include <iterator>

using namespace std;
struct Entry {
    int value,freq;
    list<int>::iterator position;
};
vector<int> solve(int capacity,const vector<vector<int>>& operations) {
    if(capacity<0)throw invalid_argument("negative capacity");
    unordered_map<int,Entry> data;
    unordered_map<int,list<int>> buckets;
    int minimum=0;
    vector<int> ans;
    auto touch=[&](int key) {
        auto& e=data.at(key);
        int f=e.freq;
        buckets.at(f).erase(e.position);
        if(buckets.at(f).empty()) {
            buckets.erase(f);
            if(minimum==f)minimum=f+1;
        } ++e.freq;
        auto& q=buckets[e.freq];
        q.push_back(key);
        e.position=prev(q.end());
    };
    for(auto& op:operations) {
        if(op.empty()||(op[0]!=0&&op[0]!=1)||op.size()!=(op[0]==0?2U:3U))throw invalid_argument("bad operation");
        int key=op[1];
        auto it=data.find(key);
        if(op[0]==0) {
            if(it==data.end())ans.push_back(-1);
            else {
                touch(key);
                ans.push_back(it->second.value);
            }
        } else if(capacity) {
            if(it!=data.end()) {
                it->second.value=op[2];
                touch(key);
            } else {
                if((int)data.size()==capacity) {
                    auto& q=buckets.at(minimum);
                    int old=q.front();
                    q.pop_front();
                    data.erase(old);
                    if(q.empty())buckets.erase(minimum);
                }
                auto& q=buckets[1];
                q.push_back(key);
                data.emplace(key,Entry {op[2],1,prev(q.end())});
                minimum=1;
            }
        }
    }
    return ans;
}
```

# 0210. 课程表 II

[LeetCode 原题](https://leetcode.com/problems/course-schedule-ii/)

## 题意与思路

与课程表相同，使用 Kahn 拓扑排序，并返回出队顺序。有环返回空数组；合法拓扑序可能不唯一。

时间 O(V+E)，空间 O(V+E)。

## 输入约定与测试

零课程返回空数组；非法课程编号或负课程数拒绝。示例采用固定遍历顺序。

- 正常：`[4, [[1, 0], [2, 0], [3, 1], [3, 2]]]` → `[0, 1, 2, 3]`。
- 环：`[2, [[1, 0], [0, 1]]]` → `[]`。
- 空课程：`[0, []]` → `[]`。
- 独立课程：`[3, []]` → `[0, 1, 2]`。
- 非法边：`[1, [[1, 0]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def course_order(n, edges):
    if n < 0:
        raise ValueError('课程数不能为负')
    graph = [[] for _ in range(n)]
    degree = [0] * n
    for edge in edges:
        if len(edge) != 2 or any((x < 0 or x >= n for x in edge)):
            raise ValueError('非法边')
        (a, b) = edge
        graph[b].append(a)
        degree[a] += 1
    queue = [i for (i, d) in enumerate(degree) if d == 0]
    head = 0
    while head < len(queue):
        u = queue[head]
        head += 1
        for v in graph[u]:
            degree[v] -= 1
            if degree[v] == 0:
                queue.append(v)
    return queue if len(queue) == n else []

def solve(n, edges):
    return course_order(n, edges)
if __name__ == '__main__':
    actual = solve(4, [[1, 0], [2, 0], [3, 1], [3, 2]])
    expected = [0, 1, 2, 3]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(2, [[1, 0], [0, 1]])
    expected = []
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(0, [])
    expected = []
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(3, [])
    expected = [0, 1, 2]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve(1, [[1, 0]])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

package main

import (
	"fmt"
	"reflect"
)

func courseOrder(n int, edges [][]int) []int {
	if n < 0 {
		panic("课程数不能为负")
	}
	graph := make([][]int, n)
	degree := make([]int, n)
	for _, e := range edges {
		if len(e) != 2 || e[0] < 0 || e[0] >= n || e[1] < 0 || e[1] >= n {
			panic("非法边")
		}
		a, b := e[0], e[1]
		graph[b] = append(graph[b], a)
		degree[a]++
	}
	q := []int{}
	for i, d := range degree {
		if d == 0 {
			q = append(q, i)
		}
	}
	for head := 0; head < len(q); head++ {
		for _, v := range graph[q[head]] {
			degree[v]--
			if degree[v] == 0 {
				q = append(q, v)
			}
		}
	}
	if len(q) != n {
		return []int{}
	}
	return q
}

func solve(n int, edges [][]int) []int { return courseOrder(n, edges) }

func main() {
	{ // 正常
		actual := solve(4, [][]int{[]int{1, 0}, []int{2, 0}, []int{3, 1}, []int{3, 2}})
		expected := []int{0, 1, 2, 3}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 环
		actual := solve(2, [][]int{[]int{1, 0}, []int{0, 1}})
		expected := []int{}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空课程
		actual := solve(0, [][]int{})
		expected := []int{}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 独立课程
		actual := solve(3, [][]int{})
		expected := []int{0, 1, 2}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve(1, [][]int{[]int{1, 0}})
	}()
}

#include <cassert>
#include <iostream>
#include <stdexcept>
#include <vector>

using namespace std;
vector<int> courseOrder(int n,const vector<vector<int>>& edges) {
    if(n<0)throw invalid_argument("negative count");
    vector<vector<int>> graph(n);
    vector<int> degree(n);
    for(auto& e:edges) {
        if(e.size()!=2||e[0]<0||e[0]>=n||e[1]<0||e[1]>=n)throw invalid_argument("bad edge");
        graph[e[1]].push_back(e[0]);
        ++degree[e[0]];
    }
    vector<int> q;
    for(int i=0;i<n;++i)if(!degree[i])q.push_back(i);
    for(size_t h=0;h<q.size();++h)for(int v:graph[q[h]])if(--degree[v]==0)q.push_back(v);
    if((int)q.size()!=n)return {};
    return q;
}
vector<int> solve(int n,const vector<vector<int>>& edges) {
    return courseOrder(n,edges);
}
template<class T> void show(const vector<T>& xs);

    }
    cout << "]";
}

        cout << " 实际 ";

        cout << "\n";

    } {
        // 环

            vector<int> {1, 0}, vector<int> {0, 1}
        });

        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空课程

        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 独立课程
        auto actual = solve(3, vector<vector<int>> {});

        cout << "用例 4：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);

            });

            rejected=true;
        }
        assert(rejected);
        cout << "用例 5：已拒绝非法输入\n";
    }
}

# 0622. 设计循环队列

[LeetCode 原题](https://leetcode.com/problems/design-circular-queue/)

## 题意与思路

固定数组 + head + size。入队位置 `(head+size)%capacity`；出队只移动 head。FIFO 出队与满容量自动淘汰不同：本题队列满时入队失败。

每次操作 O(1)，空间 O(capacity)。

## 输入约定与测试

操作：[0,x] 入队、[1] 出队、[2] Front、[3] Rear、[4] isEmpty、[5] isFull。结果 true/false 用 1/0 表示；空队头尾返回 -1。零容量扩展为既空又满；负容量或非法操作拒绝。

- 正常与环绕：`[3, [[0, 1], [0, 2], [0, 3], [0, 4], [3], [5], [1], [0, 4], [3]]]` → `[1, 1, 1, 0, 3, 1, 1, 1, 4]`。
- 空与单元素：`[1, [[2], [1], [0, 7], [3], [1], [4]]]` → `[-1, 0, 1, 7, 1, 1]`。
- 零容量：`[0, [[0, 1], [4], [5]]]` → `[0, 1, 1]`。
- 空操作：`[2, []]` → `[]`。
- 负容量：`[-1, []]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(capacity, operations):
    if capacity < 0:
        raise ValueError('负容量')
    data = [0] * capacity
    head = size = 0
    answers = []
    for op in operations:
        if not op or op[0] not in range(6) or len(op) != (2 if op[0] == 0 else 1):
            raise ValueError('非法操作')
        kind = op[0]
        if kind == 0:
            ok = size < capacity
            if ok:
                data[(head + size) % capacity] = op[1]
                size += 1
            answers.append(int(ok))
        elif kind == 1:
            ok = size > 0
            if ok:
                head = (head + 1) % capacity
                size -= 1
            answers.append(int(ok))
        elif kind == 2:
            answers.append(data[head] if size else -1)
        elif kind == 3:
            answers.append(data[(head + size - 1) % capacity] if size else -1)
        elif kind == 4:
            answers.append(int(size == 0))
        else:
            answers.append(int(size == capacity))
    return answers
if __name__ == '__main__':
    actual = solve(3, [[0, 1], [0, 2], [0, 3], [0, 4], [3], [5], [1], [0, 4], [3]])
    expected = [1, 1, 1, 0, 3, 1, 1, 1, 4]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(1, [[2], [1], [0, 7], [3], [1], [4]])
    expected = [-1, 0, 1, 7, 1, 1]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(0, [[0, 1], [4], [5]])
    expected = [0, 1, 1]
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(2, [])
    expected = []
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve(-1, [])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(capacity int, operations [][]int) []int {
	if capacity < 0 {
		panic("负容量")
	}
	data := make([]int, capacity)
	head, size := 0, 0
	ans := []int{}
	for _, op := range operations {
		if len(op) == 0 || op[0] < 0 || op[0] > 5 {
			panic("非法操作")
		}
		if (op[0] == 0 && len(op) != 2) || (op[0] != 0 && len(op) != 1) {
			panic("非法参数")
		}
		result := 0
		switch op[0] {
		case 0:
			if size < capacity {
				data[(head+size)%capacity] = op[1]
				size++
				result = 1
			}
		case 1:
			if size > 0 {
				head = (head + 1) % capacity
				size--
				result = 1
			}
		case 2:
			result = -1
			if size > 0 {
				result = data[head]
			}
		case 3:
			result = -1
			if size > 0 {
				result = data[(head+size-1)%capacity]
			}
		case 4:
			if size == 0 {
				result = 1
			}
		case 5:
			if size == capacity {
				result = 1
			}
		}
		ans = append(ans, result)
	}
	return ans
}

func main() {
	{ // 正常与环绕
		actual := solve(3, [][]int{[]int{0, 1}, []int{0, 2}, []int{0, 3}, []int{0, 4}, []int{3}, []int{5}, []int{1}, []int{0, 4}, []int{3}})
		expected := []int{1, 1, 1, 0, 3, 1, 1, 1, 4}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空与单元素
		actual := solve(1, [][]int{[]int{2}, []int{1}, []int{0, 7}, []int{3}, []int{1}, []int{4}})
		expected := []int{-1, 0, 1, 7, 1, 1}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 零容量
		actual := solve(0, [][]int{[]int{0, 1}, []int{4}, []int{5}})
		expected := []int{0, 1, 1}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空操作
		actual := solve(2, [][]int{})
		expected := []int{}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve(-1, [][]int{})
	}()
}
```

```cpp
#include <cassert>
#include <iostream>
#include <stdexcept>
#include <vector>

using namespace std;
vector<int> solve(int capacity,const vector<vector<int>>& operations) {
    if(capacity<0)throw invalid_argument("negative capacity");
    vector<int> data(capacity),ans;
    int head=0,size=0;
    for(auto& op:operations) {
        if(op.empty()||op[0]<0||op[0]>5||op.size()!=(op[0]==0?2U:1U))throw invalid_argument("bad operation");
        int result=0;
        switch(op[0]) {
            case 0:if(size<capacity) {
                data[(head+size)%capacity]=op[1];
                ++size;
                result=1;
            }
            break;
            case 1:if(size) {
                head=(head+1)%capacity;
                --size;
                result=1;
            }
            break;
            case 2:result=size?data[head]:-1;
            break;
            case 3:result=size?data[(head+size-1)%capacity]:-1;
            break;
            case 4:result=size==0;
            break;
            case 5:result=size==capacity;
            break;
        }
        ans.push_back(result);
    }
    return ans;
}
```

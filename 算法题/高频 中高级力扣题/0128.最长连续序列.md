# 0128. 最长连续序列

[LeetCode 原题](https://leetcode.com/problems/longest-consecutive-sequence/)

## 题意与思路

哈希集合去重；仅从没有前驱 x-1 的数字开始向右扩展，避免对每个数字重复扫描。

平均时间 O(n)，空间 O(n)。

## 输入约定与测试

输入整数数组；空数组返回 0。C++ 扩展为 64 位集合，避免 int 边界加减溢出。

- 正常：`[[100, 4, 200, 1, 3, 2]]` → `4`。
- 空数组：`[[]]` → `0`。
- 重复：`[[1, 2, 2, 3]]` → `3`。
- 负数：`[[-2, -1, 0]]` → `3`。
- 整数边界：`[[2147483647, -2147483648]]` → `1`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(nums):
    values = set(nums)
    best = 0
    for x in values:
        if x - 1 in values:
            continue
        end = x
        while end + 1 in values:
            end += 1
        best = max(best, end - x + 1)
    return best
if __name__ == '__main__':
    actual = solve([100, 4, 200, 1, 3, 2])
    expected = 4
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = 0
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, 2, 2, 3])
    expected = 3
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([-2, -1, 0])
    expected = 3
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([2147483647, -2147483648])
    expected = 1
    print('用例 5：期望', expected, '实际', actual)
    assert actual == expected
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(a []int) int {
	values := map[int64]bool{}
	for _, x := range a {
		values[int64(x)] = true
	}
	best := 0
	for x := range values {
		if values[x-1] {
			continue
		}
		end := x
		for values[end+1] {
			end++
		}
		if size := int(end - x + 1); size > best {
			best = size
		}
	}
	return best
}

func main() {
	{ // 正常
		actual := solve([]int{100, 4, 200, 1, 3, 2})
		expected := 4
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空数组
		actual := solve([]int{})
		expected := 0
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复
		actual := solve([]int{1, 2, 2, 3})
		expected := 3
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 负数
		actual := solve([]int{-2, -1, 0})
		expected := 3
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 整数边界
		actual := solve([]int{2147483647, -2147483648})

		fmt.Printf("用例 5：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
}
```

```cpp
#include <algorithm>
#include <cassert>
#include <iostream>
#include <unordered_set>
#include <vector>

using namespace std;
int solve(const vector<int>& nums) {
    unordered_set<long long> values(nums.begin(),nums.end());
    int best=0;
    for(long long x:values) {
        if(values.count(x-1)) continue;
        long long end=x;
        while(values.count(end+1)) ++end;
        best=max(best,(int)(end-x+1));
    }
    return best;
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}

    cout << "[";

        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
```

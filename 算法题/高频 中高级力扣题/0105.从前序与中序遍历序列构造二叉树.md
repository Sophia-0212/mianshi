# 0105. 从前序与中序遍历序列构造二叉树

[LeetCode 原题](https://leetcode.com/problems/construct-binary-tree-from-preorder-and-inorder-traversal/)

## 题意与思路

前序首元素为根；用中序下标将区间分成左右子树，再递归构造。哈希表避免每次扫描中序序列。入口返回新树的先序序列（含 # 空节点）便于验证树形。

时间 O(n)，空间 O(n)，包含映射和递归栈；退化树递归深度 O(n)，超长链状树应改迭代实现。

## 输入约定与测试

两序列等长、元素唯一且相容；不相容或重复值拒绝。空树序列化为 ["#"]。

- 正常：`[[3, 9, 20, 15, 7], [9, 3, 15, 20, 7]]` → `['3', '9', '#', '#', '20', '15', '#', '#', '7', '#', '#']`。
- 空树：`[[], []]` → `['#']`。
- 单节点：`[[1], [1]]` → `['1', '#', '#']`。
- 不相容：`[[1, 2], [2, 3]]` → `拒绝非法输入`。
- 重复值：`[[1, 1], [1, 1]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
class Node:

    def __init__(self, value):
        self.value = value
        self.left = self.right = None

def buildTree(preorder, inorder):
    if len(preorder) != len(inorder) or len(set(preorder)) != len(preorder) or set(preorder) != set(inorder):
        raise ValueError('遍历不相容')
    positions = {v: i for (i, v) in enumerate(inorder)}
    next_root = 0

    def build(left, right):
        nonlocal next_root
        if left >= right:
            return None
        if next_root >= len(preorder):
            raise ValueError('遍历不相容')
        value = preorder[next_root]
        next_root += 1
        mid = positions[value]
        if not left <= mid < right:
            raise ValueError('遍历不相容')
        node = Node(value)
        node.left = build(left, mid)
        node.right = build(mid + 1, right)
        return node
    return build(0, len(inorder))

def solve(preorder, inorder):
    root = buildTree(preorder, inorder)
    ans = []
    stack = [root]
    while stack:
        node = stack.pop()
        if node is None:
            ans.append('#')
        else:
            ans.append(str(node.value))
            stack.extend([node.right, node.left])
    return ans
if __name__ == '__main__':
    actual = solve([3, 9, 20, 15, 7], [9, 3, 15, 20, 7])
    expected = ['3', '9', '#', '#', '20', '15', '#', '#', '7', '#', '#']
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], [])
    expected = ['#']
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1], [1])
    expected = ['1', '#', '#']
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([1, 2], [2, 3])
    except (ValueError, IndexError):
        print('用例 4：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    try:
        solve([1, 1], [1, 1])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

type Node struct {
	value       int
	left, right *Node
}

func buildTree(pre, in []int) *Node {
	if len(pre) != len(in) {
		panic("长度不一致")
	}
	pos := map[int]int{}
	seen := map[int]bool{}
	for i, v := range in {
		if _, ok := pos[v]; ok {
			panic("重复值")
		}
		pos[v] = i
	}
	for _, v := range pre {
		_, ok := pos[v]
		if !ok || seen[v] {
			panic("遍历不相容")
		}
		seen[v] = true
	}
	index := 0
	var build func(int, int) *Node
	build = func(l, r int) *Node {
		if l >= r {
			return nil
		}
		if index >= len(pre) {
			panic("遍历不相容")
		}
		v := pre[index]
		index++
		m := pos[v]
		if m < l || m >= r {
			panic("遍历不相容")
		}
		node := &Node{value: v}
		node.left = build(l, m)
		node.right = build(m+1, r)
		return node
	}
	return build(0, len(in))
}
func solve(pre, in []int) []string {
	root := buildTree(pre, in)
	ans := []string{}
	stack := []*Node{root}
	for len(stack) > 0 {
		p := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		if p == nil {
			ans = append(ans, "#")
		} else {
			ans = append(ans, fmt.Sprint(p.value))
			stack = append(stack, p.right, p.left)
		}
	}
	return ans
}

func main() {
	{ // 正常
		actual := solve([]int{3, 9, 20, 15, 7}, []int{9, 3, 15, 20, 7})
		expected := []string{"3", "9", "#", "#", "20", "15", "#", "#", "7", "#", "#"}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空树
		actual := solve([]int{}, []int{})
		expected := []string{"#"}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单节点
		actual := solve([]int{1}, []int{1})
		expected := []string{"1", "#", "#"}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 4：已拒绝非法输入")
		}()
		solve([]int{1, 2}, []int{2, 3})
	}()
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([]int{1, 1}, []int{1, 1})
	}()
}
```

```cpp
#include <cassert>
#include <functional>
#include <iostream>
#include <memory>
#include <stdexcept>
#include <string>
#include <unordered_map>
#include <unordered_set>
#include <vector>

using namespace std;
struct Node {
    int value;
    unique_ptr<Node> left,right;
    explicit Node(int v):value(v) {}
};
unique_ptr<Node> buildTree(const vector<int>& pre,const vector<int>& in) {
    if(pre.size()!=in.size())throw invalid_argument("length mismatch");
    unordered_map<int,int> pos;
    unordered_set<int> seen;
    for(int i=0;i<(int)in.size();++i)if(!pos.emplace(in[i],i).second)throw invalid_argument("duplicate");
    for(int v:pre)if(!pos.count(v)||!seen.insert(v).second)throw invalid_argument("incompatible");
    int index=0;
    function<unique_ptr<Node>(int,int)> build=[&](int l,int r)->unique_ptr<Node> {
        if(l>=r)return nullptr;
        if(index>=(int)pre.size())throw invalid_argument("incompatible");
        int v=pre[index++],m=pos.at(v);
        if(m<l||m>=r)throw invalid_argument("incompatible");
        auto node=make_unique<Node>(v);
        node->left=build(l,m);
        node->right=build(m+1,r);
        return node;
    };
    return build(0,in.size());
}
vector<string> solve(const vector<int>& pre,const vector<int>& in) {
    auto root=buildTree(pre,in);
    vector<string> ans;
    vector<Node*> stack {root.get()};
    while(!stack.empty()) {
        auto* p=stack.back();
        stack.pop_back();
        if(!p)ans.push_back("#");
        else {
            ans.push_back(to_string(p->value));
            stack.push_back(p->right.get());
            stack.push_back(p->left.get());
        }
    }
    return ans;
}

}

    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}

        cout << " 实际 ";
        show(actual);
        cout << "\n";

    } {
        // 空树
        auto actual = solve(vector<int> {}, vector<int> {});

        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 单节点
        auto actual = solve(vector<int> {1}, vector<int> {1});

        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {

            rejected=true;
        }
        assert(rejected);

    }
}
```

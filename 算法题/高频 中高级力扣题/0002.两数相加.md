# 0002. 两数相加

[LeetCode 原题](https://leetcode.com/problems/add-two-numbers/)

## 题意与思路

两个非负整数按低位在前保存在链表中。逐位相加并携带进位；任一链表或进位未结束就继续。`addTwoNumbers` 操作节点，`solve` 负责数组与链表转换。

核心时间 O(max(m,n))，结果空间 O(max(m,n))。

## 输入约定与测试

节点值为 0～9；测试入口空数组代表空链表，两者均空返回空数组；非法数字拒绝。

- 正常：`[[2, 4, 3], [5, 6, 4]]` → `[7, 0, 8]`。
- 连续进位：`[[9, 9], [1]]` → `[0, 0, 1]`。
- 空链表：`[[], []]` → `[]`。
- 单侧为空：`[[], [0]]` → `[0]`。
- 非法数字：`[[12], [1]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
class Node:

    def __init__(self, value=0, next=None):
        (self.value, self.next) = (value, next)

def build(values):
    dummy = tail = Node()
    for value in values:
        tail.next = Node(value)
        tail = tail.next
    return dummy.next

def to_values(head):
    result = []
    while head:
        result.append(head.value)
        head = head.next
    return result

def addTwoNumbers(a, b):
    dummy = tail = Node()
    carry = 0
    while a or b or carry:
        total = carry + (a.value if a else 0) + (b.value if b else 0)
        tail.next = Node(total % 10)
        tail = tail.next
        carry = total // 10
        a = a.next if a else None
        b = b.next if b else None
    return dummy.next

def solve(a, b):
    if any((x < 0 or x > 9 for x in a + b)):
        raise ValueError('非法数字')
    return to_values(addTwoNumbers(build(a), build(b)))
if __name__ == '__main__':
    actual = solve([2, 4, 3], [5, 6, 4])
    expected = [7, 0, 8]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([9, 9], [1])
    expected = [0, 0, 1]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], [])
    expected = []
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], [0])
    expected = [0]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([12], [1])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

type Node struct {
	value int
	next  *Node
}

func build(values []int) *Node {
	dummy := &Node{}
	tail := dummy
	for _, v := range values {
		tail.next = &Node{value: v}
		tail = tail.next
	}
	return dummy.next
}
func toValues(head *Node) []int {
	result := []int{}
	for ; head != nil; head = head.next {
		result = append(result, head.value)
	}
	return result
}

func addTwoNumbers(a, b *Node) *Node {
	dummy := &Node{}
	tail := dummy
	carry := 0
	for a != nil || b != nil || carry != 0 {
		total := carry
		if a != nil {
			total += a.value
			a = a.next
		}
		if b != nil {
			total += b.value
			b = b.next
		}
		tail.next = &Node{value: total % 10}
		tail = tail.next
		carry = total / 10
	}
	return dummy.next
}
func solve(a, b []int) []int {
	for _, values := range [][]int{a, b} {
		for _, x := range values {
			if x < 0 || x > 9 {
				panic("非法数字")
			}
		}
	}
	return toValues(addTwoNumbers(build(a), build(b)))
}

func main() {
	{ // 正常
		actual := solve([]int{2, 4, 3}, []int{5, 6, 4})
		expected := []int{7, 0, 8}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 连续进位
		actual := solve([]int{9, 9}, []int{1})
		expected := []int{0, 0, 1}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空链表
		actual := solve([]int{}, []int{})
		expected := []int{}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单侧为空
		actual := solve([]int{}, []int{0})
		expected := []int{0}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([]int{12}, []int{1})
	}()
}
```

#include <cassert>
#include <iostream>
#include <memory>
#include <stdexcept>
#include <vector>

using namespace std;
struct Node {
    int value;
    Node* next=nullptr;
    explicit Node(int v=0):value(v) {}
};
// arena 只管理所有权，链表算法仍通过真实 next 指针操作。
Node* build(const vector<int>& values,vector<unique_ptr<Node>>& arena) {
    Node dummy;
    Node* tail=&dummy;
    for(int v:values) {
        arena.push_back(make_unique<Node>(v));
        tail->next=arena.back().get();
        tail=tail->next;
    }
    return dummy.next;
}
vector<int> toValues(Node* head) {
    vector<int> result;
    for(;head;head=head->next)result.push_back(head->value);
    return result;
}
Node* addTwoNumbers(Node* a,Node* b,vector<unique_ptr<Node>>& arena) {
    Node dummy;
    Node* tail=&dummy;
    int carry=0;
    while(a||b||carry) {
        int total=carry;
        if(a) {
            total+=a->value;
            a=a->next;
        }
        if(b) {
            total+=b->value;
            b=b->next;
        }
        arena.push_back(make_unique<Node>(total%10));
        tail->next=arena.back().get();
        tail=tail->next;
        carry=total/10;
    }
    return dummy.next;
}
vector<int> solve(const vector<int>& a,const vector<int>& b) {
    for(auto* values: {&a,&b})for(int x:*values)if(x<0||x>9)throw invalid_argument("bad digit");
    vector<unique_ptr<Node>> arena;
    auto* x=build(a,arena);
    auto* y=build(b,arena);
    return toValues(addTwoNumbers(x,y,arena));
}
template<class T> void show(const vector<T>& xs);

        cout << " 实际 ";

        cout << " 实际 ";

        cout << "\n";

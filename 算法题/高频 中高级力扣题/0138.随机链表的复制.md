# 0138. 随机链表的复制

[LeetCode 原题](https://leetcode.com/problems/copy-list-with-random-pointer/)

## 题意与思路

两遍哈希映射：第一遍为每个旧节点创建新节点；第二遍按旧→新映射连接 next 和 random。节点值相同也必须分别复制。

时间 O(n)，空间 O(n)。

## 输入约定与测试

测试输入为 [值,random下标]，-1 表示空；next 按数组顺序连接。空数组返回空；非法下标拒绝。断言检查新旧节点不共享引用。

- 正常：`[[[7, -1], [13, 0], [11, 4], [10, 2], [1, 0]]]` → `[[7, -1], [13, 0], [11, 4], [10, 2], [1, 0]]`。
- 空：`[[]]` → `[]`。
- 自引用：`[[[1, 0]]]` → `[[1, 0]]`。
- 相同值不同节点：`[[[1, 1], [1, 0]]]` → `[[1, 1], [1, 0]]`。
- 越界随机指针：`[[[1, 2]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
class Node:

    def __init__(self, value):
        self.value = value
        self.next = self.random = None

def copyRandomList(head):
    copies = {None: None}
    node = head
    while node:
        copies[node] = Node(node.value)
        node = node.next
    node = head
    while node:
        copies[node].next = copies[node.next]
        copies[node].random = copies[node.random]
        node = node.next
    return copies[head]

def solve(data):
    if any((len(p) != 2 or p[1] < -1 or p[1] >= len(data) for p in data)):
        raise ValueError('非法节点')
    nodes = [Node(p[0]) for p in data]
    for (i, (_, r)) in enumerate(data):
        nodes[i].next = nodes[i + 1] if i + 1 < len(nodes) else None
        nodes[i].random = nodes[r] if r >= 0 else None
    head = copyRandomList(nodes[0] if nodes else None)
    copied = []
    node = head
    while node:
        copied.append(node)
        node = node.next
    assert not set(nodes).intersection(copied)
    indices = {node: i for (i, node) in enumerate(copied)}
    return [[node.value, indices.get(node.random, -1)] for node in copied]
if __name__ == '__main__':
    actual = solve([[7, -1], [13, 0], [11, 4], [10, 2], [1, 0]])
    expected = [[7, -1], [13, 0], [11, 4], [10, 2], [1, 0]]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = []
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([[1, 0]])
    expected = [[1, 0]]
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([[1, 1], [1, 0]])
    expected = [[1, 1], [1, 0]]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([[1, 2]])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

type Node struct {
	value        int
	next, random *Node
}

func copyRandomList(head *Node) *Node {
	copies := map[*Node]*Node{nil: nil}
	for p := head; p != nil; p = p.next {
		copies[p] = &Node{value: p.value}
	}
	for p := head; p != nil; p = p.next {
		copies[p].next = copies[p.next]
		copies[p].random = copies[p.random]
	}
	return copies[head]
}
func solve(data [][]int) [][]int {
	nodes := make([]*Node, len(data))
	original := map[*Node]bool{}
	for i, p := range data {
		if len(p) != 2 || p[1] < -1 || p[1] >= len(data) {
			panic("非法节点")
		}
		nodes[i] = &Node{value: p[0]}
		original[nodes[i]] = true
	}
	for i, p := range data {
		if i+1 < len(nodes) {
			nodes[i].next = nodes[i+1]
		}
		if p[1] >= 0 {
			nodes[i].random = nodes[p[1]]
		}
	}
	var head *Node
	if len(nodes) > 0 {
		head = nodes[0]
	}
	copied := copyRandomList(head)
	indices := map[*Node]int{}
	for p := copied; p != nil; p = p.next {
		if original[p] {
			panic("未深拷贝")
		}
		indices[p] = len(indices)
	}
	ans := [][]int{}
	for p := copied; p != nil; p = p.next {
		r := -1
		if p.random != nil {
			r = indices[p.random]
		}
		ans = append(ans, []int{p.value, r})
	}
	return ans
}

func main() {
	{ // 正常
		actual := solve([][]int{[]int{7, -1}, []int{13, 0}, []int{11, 4}, []int{10, 2}, []int{1, 0}})
		expected := [][]int{[]int{7, -1}, []int{13, 0}, []int{11, 4}, []int{10, 2}, []int{1, 0}}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空
		actual := solve([][]int{})
		expected := [][]int{}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 自引用
		actual := solve([][]int{[]int{1, 0}})
		expected := [][]int{[]int{1, 0}}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 相同值不同节点
		actual := solve([][]int{[]int{1, 1}, []int{1, 0}})
		expected := [][]int{[]int{1, 1}, []int{1, 0}}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([][]int{[]int{1, 2}})
	}()
}
```

#include <cassert>
#include <iostream>
#include <memory>
#include <stdexcept>
#include <unordered_map>
#include <unordered_set>
#include <vector>

using namespace std;
struct Node {
    int value;
    Node* next=nullptr;
    Node* random=nullptr;
    explicit Node(int v):value(v) {}
};
Node* copyRandomList(Node* head,vector<unique_ptr<Node>>& arena) {
    unordered_map<Node*,Node*> copies {
        {nullptr,nullptr}
    };
    for(auto* p=head;p;p=p->next) {
        arena.push_back(make_unique<Node>(p->value));
        copies[p]=arena.back().get();
    }
    for(auto* p=head;p;p=p->next) {
        copies[p]->next=copies[p->next];
        copies[p]->random=copies[p->random];
    }
    return copies[head];
}
vector<vector<int>> solve(const vector<vector<int>>& data) {
    vector<unique_ptr<Node>> original,copied;
    vector<Node*> nodes;
    for(auto& p:data) {
        if(p.size()!=2||p[1]<-1||p[1]>=(int)data.size())throw invalid_argument("bad node");
        original.push_back(make_unique<Node>(p[0]));
        nodes.push_back(original.back().get());
    }
    for(size_t i=0;i<data.size();++i) {
        if(i+1<nodes.size())nodes[i]->next=nodes[i+1];
        if(data[i][1]>=0)nodes[i]->random=nodes[data[i][1]];
    }
    auto* head=copyRandomList(nodes.empty()?nullptr:nodes[0],copied);
    unordered_map<Node*,int> index;
    unordered_set<Node*> old(nodes.begin(),nodes.end());
    for(auto* p=head;p;p=p->next) {
        assert(!old.count(p));
        int i=index.size();
        index[p]=i;
    }
    vector<vector<int>> ans;
    for(auto* p=head;p;p=p->next)ans.push_back( {p->value,p->random?index.at(p->random):-1});
    return ans;
}

# 0148. 排序链表

[LeetCode 原题](https://leetcode.com/problems/sort-list/)

## 题意与思路

归并排序：快慢指针找中点并断链，分别排序两半，再合并有序链表。`sortList` 操作原节点，`solve` 仅为测试封装。

时间 O(n log n)，递归栈 O(log n)。进阶要求 O(1) 辅助空间时改用自底向上的迭代归并。

## 输入约定与测试

输入为无环链表；空链表返回空；支持重复值、负数。

- 正常：`[[4, 2, 1, 3]]` → `[1, 2, 3, 4]`。
- 空：`[[]]` → `[]`。
- 单节点：`[[1]]` → `[1]`。
- 重复与负数：`[[-1, 5, 3, 3, 0]]` → `[-1, 0, 3, 3, 5]`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
class Node:

    def __init__(self, value=0, next=None):
        (self.value, self.next) = (value, next)

def build(values):
    dummy = tail = Node()
    for value in values:
        tail.next = Node(value)
        tail = tail.next
    return dummy.next

def to_values(head):
    result = []
    while head:
        result.append(head.value)
        head = head.next
    return result

def sortList(head):
    if not head or not head.next:
        return head
    (slow, fast) = (head, head.next)
    while fast and fast.next:
        slow = slow.next
        fast = fast.next.next
    right = slow.next
    slow.next = None
    (a, b) = (sortList(head), sortList(right))
    dummy = tail = Node()
    while a and b:
        if a.value <= b.value:
            tail.next = a
            a = a.next
        else:
            tail.next = b
            b = b.next
        tail = tail.next
    tail.next = a or b
    return dummy.next

def solve(values):
    return to_values(sortList(build(values)))
if __name__ == '__main__':
    actual = solve([4, 2, 1, 3])
    expected = [1, 2, 3, 4]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = []
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1])
    expected = [1]
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([-1, 5, 3, 3, 0])
    expected = [-1, 0, 3, 3, 5]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
```

## Go

```go
package main

import (
	"fmt"
	"reflect"
)

type Node struct {
	value int
	next  *Node
}

func build(values []int) *Node {
	dummy := &Node{}
	tail := dummy
	for _, v := range values {
		tail.next = &Node{value: v}
		tail = tail.next
	}
	return dummy.next
}
func toValues(head *Node) []int {
	result := []int{}
	for ; head != nil; head = head.next {
		result = append(result, head.value)
	}
	return result
}

func sortList(head *Node) *Node {
	if head == nil || head.next == nil {
		return head
	}
	slow, fast := head, head.next
	for fast != nil && fast.next != nil {
		slow = slow.next
		fast = fast.next.next
	}
	right := slow.next
	slow.next = nil
	a, b := sortList(head), sortList(right)
	dummy := &Node{}
	tail := dummy
	for a != nil && b != nil {
		if a.value <= b.value {
			tail.next = a
			a = a.next
		} else {
			tail.next = b
			b = b.next
		}
		tail = tail.next
	}
	if a != nil {
		tail.next = a
	} else {
		tail.next = b
	}
	return dummy.next
}
func solve(values []int) []int { return toValues(sortList(build(values))) }

func main() {
	{ // 正常
		actual := solve([]int{4, 2, 1, 3})
		expected := []int{1, 2, 3, 4}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空
		actual := solve([]int{})
		expected := []int{}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单节点
		actual := solve([]int{1})
		expected := []int{1}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复与负数
		actual := solve([]int{-1, 5, 3, 3, 0})
		expected := []int{-1, 0, 3, 3, 5}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
}
```

```cpp
#include <cassert>
#include <iostream>
#include <memory>
#include <vector>

using namespace std;
struct Node {
    int value;
    Node* next=nullptr;
    explicit Node(int v=0):value(v) {}
};
// arena 只管理所有权，链表算法仍通过真实 next 指针操作。
Node* build(const vector<int>& values,vector<unique_ptr<Node>>& arena) {
    Node dummy;
    Node* tail=&dummy;
    for(int v:values) {
        arena.push_back(make_unique<Node>(v));
        tail->next=arena.back().get();
        tail=tail->next;
    }
    return dummy.next;
}
vector<int> toValues(Node* head) {
    vector<int> result;
    for(;head;head=head->next)result.push_back(head->value);
    return result;
}
Node* sortList(Node* head) {
    if(!head||!head->next)return head;
    auto* slow=head;
    auto* fast=head->next;
    while(fast&&fast->next) {
        slow=slow->next;
        fast=fast->next->next;
    }
    auto* right=slow->next;
    slow->next=nullptr;
    auto* a=sortList(head);
    auto* b=sortList(right);
    Node dummy;
    Node* tail=&dummy;
    while(a&&b) {
        if(a->value<=b->value) {
            tail->next=a;
            a=a->next;
        }else {
            tail->next=b;
            b=b->next;
        }
        tail=tail->next;
    }
    tail->next=a?a:b;
    return dummy.next;
}
vector<int> solve(const vector<int>& values) {
    vector<unique_ptr<Node>> arena;
    return toValues(sortList(build(values,arena)));
}
```

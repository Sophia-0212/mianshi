# 0049. 字母异位词分组

[LeetCode 原题](https://leetcode.com/problems/group-anagrams/)

## 题意与思路

将每个单词排序后的字符串作为哈希键；键相同的单词放入同一组。输出再排序，仅为方便三语言结果对比。

分组时间 O(n·k log k)，辅助空间 O(nk)；输出规范化另需排序。

## 输入约定与测试

单词仅含小写英文字母；空数组返回空分组，空字符串可成组。

- 正常：`[['eat', 'tea', 'tan', 'ate', 'nat', 'bat']]` → `[['ate', 'eat', 'tea'], ['bat'], ['nat', 'tan']]`。
- 空数组：`[[]]` → `[]`。
- 空字符串：`[['', '']]` → `[['', '']]`。
- 重复单词：`[['a', 'a']]` → `[['a', 'a']]`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(words):
    groups = {}
    for word in words:
        key = ''.join(sorted(word))
        groups.setdefault(key, []).append(word)
    return sorted((sorted(group) for group in groups.values()))
if __name__ == '__main__':
    actual = solve(['eat', 'tea', 'tan', 'ate', 'nat', 'bat'])
    expected = [['ate', 'eat', 'tea'], ['bat'], ['nat', 'tan']]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = []
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(['', ''])
    expected = [['', '']]
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(['a', 'a'])
    expected = [['a', 'a']]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
	"sort"
)

func solve(words []string) [][]string {
	groups := map[string][]string{}
	for _, word := range words {
		chars := []byte(word)
		sort.Slice(chars, func(i, j int) bool { return chars[i] < chars[j] })
		key := string(chars)
		groups[key] = append(groups[key], word)
	}
	ans := [][]string{}
	for _, group := range groups {
		sort.Strings(group)
		ans = append(ans, group)
	}
	sort.Slice(ans, func(i, j int) bool {
		a, b := ans[i], ans[j]
		for k := 0; k < len(a) && k < len(b); k++ {
			if a[k] != b[k] {
				return a[k] < b[k]
			}
		}
		return len(a) < len(b)
	})
	return ans
}

func main() {
	{ // 正常
		actual := solve([]string{"eat", "tea", "tan", "ate", "nat", "bat"})
		expected := [][]string{[]string{"ate", "eat", "tea"}, []string{"bat"}, []string{"nat", "tan"}}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空数组
		actual := solve([]string{})
		expected := [][]string{}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空字符串
		actual := solve([]string{"", ""})
		expected := [][]string{[]string{"", ""}}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复单词
		actual := solve([]string{"a", "a"})
		expected := [][]string{[]string{"a", "a"}}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
}
```

## C++

```cpp
#include <algorithm>
#include <cassert>
#include <iostream>
#include <string>
#include <unordered_map>
#include <vector>

using namespace std;
vector<vector<string>> solve(const vector<string>& words) {
    unordered_map<string,vector<string>> groups;
    for(auto word:words) {
        auto key=word;
        sort(key.begin(),key.end());
        groups[key].push_back(word);
    }
    vector<vector<string>> ans;
    for(auto& entry:groups) {
        sort(entry.second.begin(),entry.second.end());
        ans.push_back(entry.second);
    }
    sort(ans.begin(),ans.end());
    return ans;
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常
        auto actual = solve(vector<string> {"eat", "tea", "tan", "ate", "nat", "bat"});
        auto expected = vector<vector<string>> {
            vector<string> {"ate", "eat", "tea"}, vector<string> {"bat"}, vector<string> {"nat", "tan"}
        };
        cout << "用例 1：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空数组
        auto actual = solve(vector<string> {});
        auto expected = vector<vector<string>> {};
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空字符串
        auto actual = solve(vector<string> {"", ""});

            vector<string> {"", ""}
        };
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 重复单词
        auto actual = solve(vector<string> {"a", "a"});

            vector<string> {"a", "a"}
        };
        cout << "用例 4：期望 ";

        cout << " 实际 ";
        show(actual);

    }
}
```

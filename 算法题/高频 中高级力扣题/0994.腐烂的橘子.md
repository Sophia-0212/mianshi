# 0994. 腐烂的橘子

[LeetCode 原题](https://leetcode.com/problems/rotting-oranges/)

## 题意与思路

多源 BFS：所有腐烂橘子同时入队；每扩散一层就是一分钟。新鲜数降为零时结束，仍有无法到达的新鲜橘子则返回 -1。

时间 O(mn)，空间 O(mn)。

## 输入约定与测试

0 空格、1 新鲜、2 腐烂；空网格或无新鲜橘子返回 0；不规则网格、非法格值拒绝。复制网格避免修改调用者数据。

- 正常：`[[[2, 1, 1], [1, 1, 0], [0, 1, 1]]]` → `4`。
- 无法到达：`[[[2, 1, 1], [0, 1, 1], [1, 0, 1]]]` → `-1`。
- 空网格：`[[]]` → `0`。
- 无新鲜：`[[[0, 2]]]` → `0`。
- 无感染源：`[[[1]]]` → `-1`。
- 不规则网格：`[[[1], [1, 2]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(grid):
    if not grid:
        return 0
    width = len(grid[0])
    if any((len(row) != width or any((x not in (0, 1, 2) for x in row)) for row in grid)):
        raise ValueError('非法网格')
    g = [row[:] for row in grid]
    queue = []
    fresh = 0
    for (i, row) in enumerate(g):
        for (j, x) in enumerate(row):
            if x == 2:
                queue.append((i, j))
            if x == 1:
                fresh += 1
    head = minutes = 0
    while head < len(queue) and fresh:
        end = len(queue)
        while head < end:
            (i, j) = queue[head]
            head += 1
            for (x, y) in [(i - 1, j), (i + 1, j), (i, j - 1), (i, j + 1)]:
                if 0 <= x < len(g) and 0 <= y < width and (g[x][y] == 1):
                    g[x][y] = 2
                    fresh -= 1
                    queue.append((x, y))
        minutes += 1
    return minutes if fresh == 0 else -1
if __name__ == '__main__':

    actual = solve([[2, 1, 1], [0, 1, 1], [1, 0, 1]])
    expected = -1
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])

    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected

    expected = 0
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([[1]])

    print('用例 5：期望', expected, '实际', actual)

        solve([[1], [1, 2]])

    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(grid [][]int) int {
	if len(grid) == 0 {
		return 0
	}
	n, m := len(grid), len(grid[0])
	g := make([][]int, n)
	q := [][2]int{}
	fresh := 0
	for i, row := range grid {
		if len(row) != m {
			panic("不规则网格")
		}
		g[i] = append([]int{}, row...)
		for j, x := range row {
			if x < 0 || x > 2 {
				panic("非法格值")
			}
			if x == 1 {
				fresh++
			}
			if x == 2 {
				q = append(q, [2]int{i, j})
			}
		}
	}
	dirs := [][2]int{{-1, 0}, {1, 0}, {0, -1}, {0, 1}}
	head, minutes := 0, 0
	for head < len(q) && fresh > 0 {
		end := len(q)
		for head < end {
			p := q[head]
			head++
			for _, d := range dirs {
				x, y := p[0]+d[0], p[1]+d[1]
				if x >= 0 && x < n && y >= 0 && y < m && g[x][y] == 1 {
					g[x][y] = 2
					fresh--
					q = append(q, [2]int{x, y})
				}
			}
		}
		minutes++
	}
	if fresh > 0 {
		return -1
	}
	return minutes
}

func main() {
	{ // 正常
		actual := solve([][]int{[]int{2, 1, 1}, []int{1, 1, 0}, []int{0, 1, 1}})
		expected := 4
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 无法到达
		actual := solve([][]int{[]int{2, 1, 1}, []int{0, 1, 1}, []int{1, 0, 1}})
		expected := -1
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空网格
		actual := solve([][]int{})
		expected := 0
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 无新鲜
		actual := solve([][]int{[]int{0, 2}})
		expected := 0
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 无感染源
		actual := solve([][]int{[]int{1}})
		expected := -1
		fmt.Printf("用例 5：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 6：已拒绝非法输入")
		}()
		solve([][]int{[]int{1}, []int{1, 2}})
	}()
}
```

```cpp
#include <cassert>
#include <iostream>
#include <queue>
#include <stdexcept>
#include <utility>
#include <vector>

using namespace std;
int solve(vector<vector<int>> g) {
    if(g.empty())return 0;
    int n=g.size(),m=g[0].size(),fresh=0,minutes=0;
    queue<pair<int,int>> q;
    for(int i=0;i<n;++i) {
        if((int)g[i].size()!=m)throw invalid_argument("ragged grid");
        for(int j=0;j<m;++j) {
            int x=g[i][j];
            if(x<0||x>2)throw invalid_argument("bad cell");
            if(x==1)++fresh;
            if(x==2)q.push( {i,j});
        }
    }
    int dirs[5]= {-1,0,1,0,-1};
    while(!q.empty()&&fresh) {
        int count=q.size();
        while(count--) {
            auto [i,j]=q.front();
            q.pop();
            for(int d=0;d<4;++d) {
                int x=i+dirs[d],y=j+dirs[d+1];
                if(x>=0&&x<n&&y>=0&&y<m&&g[x][y]==1) {
                    g[x][y]=2;
                    --fresh;
                    q.push( {x,y});
                }
            }
        }++minutes;
    }
    return fresh?-1:minutes;
}
```

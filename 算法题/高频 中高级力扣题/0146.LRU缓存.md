# 0146. LRU 缓存

[LeetCode 原题](https://leetcode.com/problems/lru-cache/)

## 题意与思路

哈希表定位节点，双向链表维护访问顺序；头部最新、尾部最旧。get 命中或 put 更新都移动到头部；容量满时删除尾部。

get/put 平均 O(1)，状态空间 O(capacity)；批量测试输出另计。

## 输入约定与测试

操作编码：`[0,key]` 为 get，`[1,key,value]` 为 put；返回所有 get 结果，未命中为 -1。容量 0 不存储，负容量或非法操作拒绝。

- 访问后淘汰：`[2, [[1, 1, 1], [1, 2, 2], [0, 1], [1, 3, 3], [0, 2], [0, 3]]]` → `[1, -1, 3]`。
- 零容量：`[0, [[1, 1, 1], [0, 1]]]` → `[-1]`。
- 空操作：`[1, []]` → `[]`。
- 更新同一键：`[1, [[1, 1, 1], [1, 1, 9], [0, 1]]]` → `[9]`。
- 负容量：`[-1, []]` → `拒绝非法输入`。
- 非法操作：`[1, [[0]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
class Node:

    def __init__(self, key=0, value=0):
        (self.key, self.value) = (key, value)
        self.prev = self.next = None

def detach(node):
    node.prev.next = node.next
    node.next.prev = node.prev

def push_front(head, node):
    node.prev = head
    node.next = head.next
    head.next.prev = node
    head.next = node

def solve(capacity, operations):
    if capacity < 0:
        raise ValueError('负容量')
    data = {}
    (head, tail) = (Node(), Node())
    head.next = tail
    tail.prev = head
    answers = []
    for op in operations:
        if not op or op[0] not in (0, 1) or len(op) != (2 if op[0] == 0 else 3):
            raise ValueError('非法操作')
        key = op[1]
        node = data.get(key)
        if op[0] == 0:
            if node is None:
                answers.append(-1)
            else:
                detach(node)
                push_front(head, node)
                answers.append(node.value)
        elif capacity:
            if node is not None:
                node.value = op[2]
                detach(node)
                push_front(head, node)
            else:
                node = Node(key, op[2])
                data[key] = node
                push_front(head, node)
                if len(data) > capacity:
                    old = tail.prev
                    detach(old)
                    del data[old.key]
    return answers
if __name__ == '__main__':
    actual = solve(2, [[1, 1, 1], [1, 2, 2], [0, 1], [1, 3, 3], [0, 2], [0, 3]])
    expected = [1, -1, 3]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(0, [[1, 1, 1], [0, 1]])
    expected = [-1]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(1, [])
    expected = []
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(1, [[1, 1, 1], [1, 1, 9], [0, 1]])
    expected = [9]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve(-1, [])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    try:
        solve(1, [[0]])
    except (ValueError, IndexError):
        print('用例 6：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"container/list"
	"fmt"
	"reflect"
)

type Entry struct{ key, value int }

func solve(capacity int, operations [][]int) []int {
	if capacity < 0 {
		panic("负容量")
	}
	order := list.New()
	data := map[int]*list.Element{}
	ans := []int{}
	for _, op := range operations {
		if len(op) == 0 || (op[0] != 0 && op[0] != 1) {
			panic("非法操作")
		}
		if (op[0] == 0 && len(op) != 2) || (op[0] == 1 && len(op) != 3) {
			panic("非法参数")
		}
		key := op[1]
		node := data[key]
		if op[0] == 0 {
			if node == nil {
				ans = append(ans, -1)
			} else {
				order.MoveToFront(node)
				ans = append(ans, node.Value.(Entry).value)
			}
		} else if capacity > 0 {
			if node != nil {
				node.Value = Entry{key, op[2]}
				order.MoveToFront(node)
			} else {
				data[key] = order.PushFront(Entry{key, op[2]})
				if len(data) > capacity {
					old := order.Back()
					delete(data, old.Value.(Entry).key)
					order.Remove(old)
				}
			}
		}
	}
	return ans
}

func main() {
	{ // 访问后淘汰
		actual := solve(2, [][]int{[]int{1, 1, 1}, []int{1, 2, 2}, []int{0, 1}, []int{1, 3, 3}, []int{0, 2}, []int{0, 3}})
		expected := []int{1, -1, 3}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 零容量
		actual := solve(0, [][]int{[]int{1, 1, 1}, []int{0, 1}})
		expected := []int{-1}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空操作
		actual := solve(1, [][]int{})
		expected := []int{}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 更新同一键
		actual := solve(1, [][]int{[]int{1, 1, 1}, []int{1, 1, 9}, []int{0, 1}})
		expected := []int{9}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve(-1, [][]int{})
	}()
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 6：已拒绝非法输入")
		}()
		solve(1, [][]int{[]int{0}})
	}()
}
```

```cpp
#include <algorithm>
#include <cassert>
#include <iostream>
#include <list>
#include <stdexcept>
#include <unordered_map>
#include <utility>
#include <vector>

using namespace std;
vector<int> solve(int capacity,const vector<vector<int>>& operations) {
    if(capacity<0)throw invalid_argument("negative capacity");
    list<pair<int,int>> order;
    unordered_map<int,list<pair<int,int>>::iterator> data;
    vector<int> ans;
    for(auto& op:operations) {
        if(op.empty()||(op[0]!=0&&op[0]!=1)||op.size()!=(op[0]==0?2U:3U))throw invalid_argument("bad operation");
        int key=op[1];
        auto found=data.find(key);
        if(op[0]==0) {
            if(found==data.end())ans.push_back(-1);
            else {
                order.splice(order.begin(),order,found->second);
                ans.push_back(found->second->second);
            }
        }else if(capacity) {
            if(found!=data.end()) {
                found->second->second=op[2];
                order.splice(order.begin(),order,found->second);
            } else {
                order.push_front( {key,op[2]});
                data[key]=order.begin();
                if((int)data.size()>capacity) {
                    data.erase(order.back().first);
                    order.pop_back();
                }
            }
        }
    }
    return ans;
}
```

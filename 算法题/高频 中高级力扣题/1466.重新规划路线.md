# 1466. 重新规划路线

[LeetCode 原题](https://leetcode.com/problems/reorder-routes-to-make-all-paths-lead-to-the-city-zero/)

## 题意与思路

原图底层是树。对原边 a→b 建两条遍历边：(a,b,1)、(b,a,0)。从 0 出发，沿原方向向外走的边需要反转，累加代价。

时间 O(n)，空间 O(n)。

## 输入约定与测试

n 至少为 1，边数 n-1，编号合法且底层无向图连通；否则拒绝。

- 正常：`[6, [[0, 1], [1, 3], [2, 3], [4, 0], [4, 5]]]` → `3`。
- 无需反转：`[3, [[1, 0], [2, 0]]]` → `0`。
- 单城空边：`[1, []]` → `0`。
- 非法规模：`[0, []]` → `拒绝非法输入`。
- 非树：`[3, [[0, 1], [0, 1]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(n, edges):
    if n < 1 or len(edges) != n - 1:
        raise ValueError('输入必须为树')
    graph = [[] for _ in range(n)]
    for e in edges:
        if len(e) != 2 or any((x < 0 or x >= n for x in e)):
            raise ValueError('非法边')
        (a, b) = e
        graph[a].append((b, 1))
        graph[b].append((a, 0))
    seen = {0}
    stack = [0]
    answer = 0
    while stack:
        u = stack.pop()
        for (v, cost) in graph[u]:
            if v not in seen:
                seen.add(v)
                stack.append(v)
                answer += cost
    if len(seen) != n:
        raise ValueError('输入必须连通')
    return answer
if __name__ == '__main__':
    actual = solve(6, [[0, 1], [1, 3], [2, 3], [4, 0], [4, 5]])
    expected = 3
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(3, [[1, 0], [2, 0]])
    expected = 0
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(1, [])
    expected = 0
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve(0, [])
    except (ValueError, IndexError):
        print('用例 4：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    try:
        solve(3, [[0, 1], [0, 1]])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

package main

import (
	"fmt"
	"reflect"
)

func solve(n int, edges [][]int) int {
	if n < 1 || len(edges) != n-1 {
		panic("输入必须为树")
	}
	g := make([][][2]int, n)
	for _, e := range edges {
		if len(e) != 2 || e[0] < 0 || e[0] >= n || e[1] < 0 || e[1] >= n {
			panic("非法边")
		}
		a, b := e[0], e[1]
		g[a] = append(g[a], [2]int{b, 1})
		g[b] = append(g[b], [2]int{a, 0})
	}
	seen := make([]bool, n)
	seen[0] = true
	stack := []int{0}
	count, answer := 1, 0
	for len(stack) > 0 {
		u := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		for _, e := range g[u] {
			v := e[0]
			if !seen[v] {
				seen[v] = true
				count++
				answer += e[1]
				stack = append(stack, v)
			}
		}
	}
	if count != n {
		panic("输入必须连通")
	}
	return answer
}

func main() {
	{ // 正常
		actual := solve(6, [][]int{[]int{0, 1}, []int{1, 3}, []int{2, 3}, []int{4, 0}, []int{4, 5}})
		expected := 3
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 无需反转
		actual := solve(3, [][]int{[]int{1, 0}, []int{2, 0}})
		expected := 0
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单城空边
		actual := solve(1, [][]int{})
		expected := 0
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 4：已拒绝非法输入")
		}()
		solve(0, [][]int{})
	}()
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve(3, [][]int{[]int{0, 1}, []int{0, 1}})
	}()
}

#include <cassert>
#include <iostream>
#include <stdexcept>
#include <utility>
#include <vector>

using namespace std;
int solve(int n,const vector<vector<int>>& edges) {
    if(n<1||(int)edges.size()!=n-1)throw invalid_argument("not a tree");
    vector<vector<pair<int,int>>> g(n);
    for(auto& e:edges) {
        if(e.size()!=2||e[0]<0||e[0]>=n||e[1]<0||e[1]>=n)throw invalid_argument("bad edge");
        g[e[0]].push_back( {e[1],1});
        g[e[1]].push_back( {e[0],0});
    }
    vector<bool> seen(n);
    seen[0]=true;
    vector<int> stack {0};
    int count=1,answer=0;
    while(!stack.empty()) {
        int u=stack.back();
        stack.pop_back();
        for(auto [v,c]:g[u])if(!seen[v]) {
            seen[v]=true;
            ++count;
            answer+=c;
            stack.push_back(v);
        }
    }
    if(count!=n)throw invalid_argument("disconnected");
    return answer;
}

# 0003. 无重复字符的最长子串

[LeetCode 原题](https://leetcode.com/problems/longest-substring-without-repeating-characters/)

## 题意与思路

滑动窗口保存不重复区间。记录字符上次出现的位置；重复时左端跳到旧位置之后，但不能后退。

时间 O(n)，空间 O(字符集大小)。

## 输入约定与测试

按字符处理；空串返回 0。

- 正常：`['abcabcbb']` → `3`。
- 全部重复：`['bbbbb']` → `1`。
- 空串：`['']` → `0`。
- 左端不能后退：`['abba']` → `2`。
- Unicode 字符：`['你好吗你']` → `3`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(s):
    (last, left, best) = ({}, 0, 0)
    for (right, ch) in enumerate(s):
        left = max(left, last.get(ch, -1) + 1)
        last[ch] = right
        best = max(best, right - left + 1)
    return best
if __name__ == '__main__':
    actual = solve('abcabcbb')
    expected = 3
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('bbbbb')
    expected = 1
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('')
    expected = 0
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('abba')
    expected = 2
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('你好吗你')
    expected = 3
    print('用例 5：期望', expected, '实际', actual)

```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(s string) int {
	last := map[rune]int{}
	left, best := 0, 0
	for right, ch := range []rune(s) {
		if old, ok := last[ch]; ok && old >= left {
			left = old + 1
		}
		last[ch] = right
		if size := right - left + 1; size > best {
			best = size
		}
	}
	return best
}

func main() {
	{ // 正常
		actual := solve("abcabcbb")
		expected := 3
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 全部重复
		actual := solve("bbbbb")
		expected := 1
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空串
		actual := solve("")
		expected := 0
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)

		}
```

```cpp
#include <algorithm>
#include <cassert>
#include <iostream>
#include <string>
#include <unordered_map>
#include <vector>

using namespace std;
// 使用 UTF-8 码点的字节串作为键；输入要求是有效 UTF-8。
int solve(const string& s) {
    unordered_map<string,int> last;
    int left=0, best=0, right=0;
    for (size_t i=0; i<s.size(); ++right) {
        unsigned char c=s[i];
        size_t width=c<0x80?1:(c<0xE0?2:(c<0xF0?3:4));
        string ch=s.substr(i,width);
        i+=width;
        if(last.count(ch)) left=max(left,last[ch]+1);
        last[ch]=right;
        best=max(best,right-left+1);
    }
    return best;
}
```

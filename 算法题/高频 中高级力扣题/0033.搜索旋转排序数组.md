# 0033. 搜索旋转排序数组

[LeetCode 原题](https://leetcode.com/problems/search-in-rotated-sorted-array/)

## 题意与思路

每次二分至少有一半有序。先判断哪半有序，再判断目标是否落在该半区。

时间 O(log n)，空间 O(1)。

## 输入约定与测试

数组来自无重复的升序数组旋转；不额外扫描验证此前提。空数组或找不到返回 -1。

- 正常：`[[4, 5, 6, 7, 0, 1, 2], 0]` → `4`。
- 不存在：`[[4, 5, 6, 7, 0, 1, 2], 3]` → `-1`。
- 空数组：`[[], 1]` → `-1`。
- 单元素：`[[1], 1]` → `0`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(nums, target):
    (l, r) = (0, len(nums) - 1)
    while l <= r:
        m = (l + r) // 2
        if nums[m] == target:
            return m
        if nums[l] <= nums[m]:
            if nums[l] <= target < nums[m]:
                r = m - 1
            else:
                l = m + 1
        elif nums[m] < target <= nums[r]:
            l = m + 1
        else:
            r = m - 1
    return -1
if __name__ == '__main__':
    actual = solve([4, 5, 6, 7, 0, 1, 2], 0)
    expected = 4
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([4, 5, 6, 7, 0, 1, 2], 3)
    expected = -1
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], 1)
    expected = -1
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1], 1)
    expected = 0
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(a []int, target int) int {
	l, r := 0, len(a)-1
	for l <= r {
		m := l + (r-l)/2
		if a[m] == target {
			return m
		}
		if a[l] <= a[m] {
			if a[l] <= target && target < a[m] {
				r = m - 1
			} else {
				l = m + 1
			}
		} else {
			if a[m] < target && target <= a[r] {
				l = m + 1
			} else {
				r = m - 1
			}
		}
	}
	return -1
}

func main() {
	{ // 正常
		actual := solve([]int{4, 5, 6, 7, 0, 1, 2}, 0)
		expected := 4
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 不存在
		actual := solve([]int{4, 5, 6, 7, 0, 1, 2}, 3)
		expected := -1
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空数组
		actual := solve([]int{}, 1)
		expected := -1
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单元素
		actual := solve([]int{1}, 1)
		expected := 0
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
}
```

```cpp
        int m=l+(r-l)/2;
```

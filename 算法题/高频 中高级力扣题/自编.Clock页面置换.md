# Clock 页面置换（自编模拟）

**自编模拟题，无 LeetCode 题号。**

## 题意与思路

页面槽位排成环，命中时访问位置 1。缺页时指针扫描：遇到 1 清零并前进；遇到 0 或空槽就装入，置 1 后前进。本题返回总缺页数。

本实现用哈希表定位命中：平均查找 O(1)，单次置换最坏 O(capacity)；空间 O(capacity)。

## 输入约定与测试

输入为页面编号序列与容量。空序列缺页数为 0；容量 0 每次访问均缺页；负容量拒绝。

- 正常置换：`[[1, 2, 3, 1, 4, 1, 2], 3]` → `6`。
- 空序列：`[[], 3]` → `0`。
- 重复命中：`[[1, 1, 1], 1]` → `1`。
- 零容量：`[[1, 2], 0]` → `2`。
- 负容量：`[[1], -1]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(pages, capacity):
    if capacity < 0:
        raise ValueError('负容量')
    if capacity == 0:
        return len(pages)
    slots = [None] * capacity
    bits = [0] * capacity
    where = {}
    hand = faults = 0
    for page in pages:
        if page in where:
            bits[where[page]] = 1
            continue
        faults += 1
        while slots[hand] is not None and bits[hand]:
            bits[hand] = 0
            hand = (hand + 1) % capacity
        if slots[hand] is not None:
            del where[slots[hand]]
        slots[hand] = page
        bits[hand] = 1
        where[page] = hand
        hand = (hand + 1) % capacity
    return faults
if __name__ == '__main__':
    actual = solve([1, 2, 3, 1, 4, 1, 2], 3)
    expected = 6
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], 3)
    expected = 0
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, 1, 1], 1)
    expected = 1
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, 2], 0)
    expected = 2
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([1], -1)
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(pages []int, capacity int) int {
	if capacity < 0 {
		panic("负容量")
	}
	if capacity == 0 {
		return len(pages)
	}
	slots := make([]int, capacity)
	bits := make([]bool, capacity)
	used := make([]bool, capacity)
	where := map[int]int{}
	hand, faults := 0, 0
	for _, page := range pages {
		if index, ok := where[page]; ok {
			bits[index] = true
			continue
		}
		faults++
		for used[hand] && bits[hand] {
			bits[hand] = false
			hand = (hand + 1) % capacity
		}
		if used[hand] {
			delete(where, slots[hand])
		}
		slots[hand] = page
		used[hand] = true
		bits[hand] = true
		where[page] = hand
		hand = (hand + 1) % capacity
	}
	return faults
}

func main() {
	{ // 正常置换
		actual := solve([]int{1, 2, 3, 1, 4, 1, 2}, 3)
		expected := 6
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空序列
		actual := solve([]int{}, 3)
		expected := 0
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复命中
		actual := solve([]int{1, 1, 1}, 1)
		expected := 1
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 零容量
		actual := solve([]int{1, 2}, 0)
		expected := 2
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([]int{1}, -1)
	}()
}
```

## C++

保存为 `main.cpp`，运行 `c++ -std=c++17 main.cpp -o main && ./main`。

#include <algorithm>
#include <cassert>
#include <iostream>
#include <stdexcept>
#include <unordered_map>
#include <vector>

using namespace std;
int solve(const vector<int>& pages,int capacity) {
    if(capacity<0)throw invalid_argument("negative capacity");
    if(!capacity)return pages.size();
    vector<int> slots(capacity);
    vector<bool> used(capacity),bits(capacity);
    unordered_map<int,int> where;
    int hand=0,faults=0;
    for(int page:pages) {
        auto it=where.find(page);
        if(it!=where.end()) {
            bits[it->second]=true;
            continue;
        }++faults;
        while(used[hand]&&bits[hand]) {
            bits[hand]=false;
            hand=(hand+1)%capacity;
        }
        if(used[hand])where.erase(slots[hand]);
        slots[hand]=page;
        used[hand]=bits[hand]=true;
        where[page]=hand;
        hand=(hand+1)%capacity;
    }
    return faults;
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常置换
        auto actual = solve(vector<int> {1, 2, 3, 1, 4, 1, 2}, 3);
        auto expected = 6;
        cout << "用例 1：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空序列
        auto actual = solve(vector<int> {}, 3);
        auto expected = 0;
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 重复命中
        auto actual = solve(vector<int> {1, 1, 1}, 1);
        auto expected = 1;
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 零容量
        auto actual = solve(vector<int> {1, 2}, 0);

        cout << "用例 4：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {

        try {
            solve(vector<int> {1}, -1);
        } catch(const invalid_argument&) {
            rejected=true;
        }
        assert(rejected);
        cout << "用例 5：已拒绝非法输入\n";
    }
}

# 3508. 设计路由器（FIFO 淘汰）

[LeetCode 原题](https://leetcode.com/problems/implement-router/)

## 题意与思路

队列维护包的进入顺序；容量满时移除最早的包。哈希集合按 (source,destination,timestamp) 去重。每个目的地维护有序时间戳及已移除的前缀，用两次二分统计时间范围内的包数。

add/forward 平均 O(1)，getCount 为 O(log q)；存活包空间 O(capacity)，教学实现的时间戳历史空间 O(q)。历史前缀可定期压缩。

## 输入约定与测试

操作 [0,s,d,t] 添加、[1] 转发、[2,d,start,end] 统计；返回二维数组：布尔或计数装为单元素数组，空转发为 []。添加调用时间戳必须非递减；容量至少 1，编号/时间为正；非法参数拒绝。

- 正常、重复、满容量淘汰：`[3, [[0, 1, 4, 90], [0, 2, 5, 90], [0, 1, 4, 90], [0, 3, 5, 95], [0, 4, 5, 105], [1], [0, 5, 2, 110], [2, 5, 100, 110]]]` → `[[1], [1], [0], [1], [1], [2, 5, 90], [1], [1]]`。
- 容量一与空转发：`[1, [[1], [0, 1, 2, 1], [0, 2, 2, 2], [2, 2, 1, 2], [1], [1]]]` → `[[], [1], [1], [1], [2, 2, 2], []]`。
- 空操作：`[2, []]` → `[]`。
- 非法容量：`[0, []]` → `拒绝非法输入`。
- 时间逆序：`[2, [[0, 1, 2, 2], [0, 1, 2, 1]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
from collections import deque
from bisect import bisect_left, bisect_right

def solve(capacity, operations):
    if capacity < 1:
        raise ValueError('容量至少为 1')
    packets = deque()
    present = set()
    times = {}
    starts = {}
    last_time = 0
    answers = []

    def forward():
        if not packets:
            return []
        packet = packets.popleft()
        present.remove(packet)
        starts[packet[1]] += 1
        return list(packet)
    for op in operations:
        if not op or op[0] not in (0, 1, 2) or len(op) != {0: 4, 1: 1, 2: 4}[op[0]]:
            raise ValueError('非法操作')
        if op[0] == 0:
            (_, source, destination, t) = op
            if min(source, destination, t) < 1 or t < last_time:
                raise ValueError('非法添加参数')
            last_time = t
            packet = (source, destination, t)
            if packet in present:
                answers.append([0])
                continue
            if len(packets) == capacity:
                forward()
            packets.append(packet)
            present.add(packet)
            times.setdefault(destination, []).append(t)
            starts.setdefault(destination, 0)
            answers.append([1])
        elif op[0] == 1:
            answers.append(forward())
        else:
            (_, destination, left, right) = op
            if destination < 1 or left < 1 or left > right:
                raise ValueError('非法查询')
            a = times.get(destination, [])
            start = starts.get(destination, 0)
            answers.append([bisect_right(a, right, lo=start) - bisect_left(a, left, lo=start)])
    return answers
if __name__ == '__main__':
    actual = solve(3, [[0, 1, 4, 90], [0, 2, 5, 90], [0, 1, 4, 90], [0, 3, 5, 95], [0, 4, 5, 105], [1], [0, 5, 2, 110], [2, 5, 100, 110]])
    expected = [[1], [1], [0], [1], [1], [2, 5, 90], [1], [1]]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(1, [[1], [0, 1, 2, 1], [0, 2, 2, 2], [2, 2, 1, 2], [1], [1]])
    expected = [[], [1], [1], [1], [2, 2, 2], []]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(2, [])
    expected = []
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve(0, [])
    except (ValueError, IndexError):
        print('用例 4：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    try:
        solve(2, [[0, 1, 2, 2], [0, 1, 2, 1]])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

```go
package main

import (
	"container/list"
	"fmt"
	"reflect"
	"sort"
)

func solve(capacity int, operations [][]int) [][]int {
	if capacity < 1 {
		panic("容量至少为 1")
	}
	packets := list.New()
	present := map[[3]int]bool{}
	times := map[int][]int{}
	starts := map[int]int{}
	last := 0
	ans := [][]int{}
	forward := func() []int {
		if packets.Len() == 0 {
			return []int{}
		}
		e := packets.Front()
		p := e.Value.([3]int)
		packets.Remove(e)
		delete(present, p)
		starts[p[1]]++
		return []int{p[0], p[1], p[2]}
	}
	for _, op := range operations {
		if len(op) == 0 || op[0] < 0 || op[0] > 2 {
			panic("非法操作")
		}
		required := 4
		if op[0] == 1 {
			required = 1
		}
		if len(op) != required {
			panic("非法参数")
		}
		switch op[0] {
		case 0:
			s, d, t := op[1], op[2], op[3]
			if s < 1 || d < 1 || t < 1 || t < last {
				panic("非法添加参数")
			}
			last = t
			p := [3]int{s, d, t}
			if present[p] {
				ans = append(ans, []int{0})
				continue
			}
			if packets.Len() == capacity {
				forward()
			}
			packets.PushBack(p)
			present[p] = true
			times[d] = append(times[d], t)
			ans = append(ans, []int{1})
		case 1:
			ans = append(ans, forward())
		case 2:
			d, l, r := op[1], op[2], op[3]
			if d < 1 || l < 1 || l > r {
				panic("非法查询")
			}
			a := times[d]
			a = a[starts[d]:]
			count := sort.Search(len(a), func(i int) bool { return a[i] > r }) - sort.Search(len(a), func(i int) bool { return a[i] >= l })
			ans = append(ans, []int{count})
		}
	}
	return ans
}

func main() {
	{ // 正常、重复、满容量淘汰

		expected := [][]int{[]int{1}, []int{1}, []int{0}, []int{1}, []int{1}, []int{2, 5, 90}, []int{1}, []int{1}}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 容量一与空转发
		actual := solve(1, [][]int{[]int{1}, []int{0, 1, 2, 1}, []int{0, 2, 2, 2}, []int{2, 2, 1, 2}, []int{1}, []int{1}})
		expected := [][]int{[]int{}, []int{1}, []int{1}, []int{1}, []int{2, 2, 2}, []int{}}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空操作
		actual := solve(2, [][]int{})
		expected := [][]int{}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 4：已拒绝非法输入")
		}()
		solve(0, [][]int{})
	}()
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve(2, [][]int{[]int{0, 1, 2, 2}, []int{0, 1, 2, 1}})
	}()
}
```

```cpp
#include <algorithm>
#include <array>
#include <cassert>
#include <functional>
#include <iostream>
#include <queue>
#include <stdexcept>
#include <unordered_map>
#include <unordered_set>
#include <vector>

using namespace std;
struct PacketHash {
    size_t operator()(const array<int,3>& p)const {
        size_t h=0;
        for(int x:p)h^=hash<int> {}(x)+0x9e3779b9+(h<<6)+(h>>2);
        return h;
    }
};
vector<vector<int>> solve(int capacity,const vector<vector<int>>& operations) {
    if(capacity<1)throw invalid_argument("bad capacity");
    queue<array<int,3>> packets;
    unordered_set<array<int,3>,PacketHash> present;
    unordered_map<int,vector<int>> times;
    unordered_map<int,size_t> starts;
    int last=0;
    vector<vector<int>> ans;
    auto forward=[&]()->vector<int> {
        if(packets.empty())return {};
        auto p=packets.front();
        packets.pop();
        present.erase(p);
        ++starts[p[1]];
        return {p[0],p[1],p[2]};
    };
    for(auto& op:operations) {
        if(op.empty()||op[0]<0||op[0]>2||op.size()!=(op[0]==1?1U:4U))throw invalid_argument("bad operation");
        if(op[0]==0) {
            int s=op[1],d=op[2],t=op[3];
            if(s<1||d<1||t<1||t<last)throw invalid_argument("bad packet");
            last=t;
            array<int,3> p {s,d,t};
            if(present.count(p)) {
                ans.push_back( {0});
                continue;
            }
            if((int)packets.size()==capacity)forward();
            packets.push(p);
            present.insert(p);
            times[d].push_back(t);
            ans.push_back( {1});
        }else if(op[0]==1)ans.push_back(forward());
        else {
            int d=op[1],l=op[2],r=op[3];
            if(d<1||l<1||l>r)throw invalid_argument("bad query");
            auto& a=times[d];
            auto begin=a.begin()+starts[d];
            ans.push_back( {(int)(upper_bound(begin,a.end(),r)-lower_bound(begin,a.end(),l))});
        }
    }
    return ans;
}
```

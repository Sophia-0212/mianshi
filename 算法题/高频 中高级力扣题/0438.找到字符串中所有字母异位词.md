# 0438. 找到字符串中所有字母异位词

[LeetCode 原题](https://leetcode.com/problems/find-all-anagrams-in-a-string/)

## 题意与思路

维护长度等于模式串 p 的滑动窗口。窗口的 26 个字符计数与 p 相同，就记录左端下标。

时间 O(26n+m)，空间 O(26)，不含输出。

## 输入约定与测试

仅小写英文字母；空模式串按扩展约定返回空数组；非法字符拒绝。

- 正常：`['cbaebabacd', 'abc']` → `[0, 6]`。
- 重叠匹配：`['abab', 'ab']` → `[0, 1, 2]`。
- 空输入：`['', 'a']` → `[]`。
- 空模式：`['abc', '']` → `[]`。
- 非法字符：`['Ab', 'ab']` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(s, p):
    if any((not 'a' <= c <= 'z' for c in s + p)):
        raise ValueError('非法字符')
    if not p or len(p) > len(s):
        return []
    want = [0] * 26
    window = [0] * 26
    ans = []
    for c in p:
        want[ord(c) - 97] += 1
    for (i, c) in enumerate(s):
        window[ord(c) - 97] += 1
        if i >= len(p):
            window[ord(s[i - len(p)]) - 97] -= 1
        if i >= len(p) - 1 and window == want:
            ans.append(i - len(p) + 1)
    return ans
if __name__ == '__main__':
    actual = solve('cbaebabacd', 'abc')
    expected = [0, 6]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('abab', 'ab')
    expected = [0, 1, 2]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('', 'a')
    expected = []
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('abc', '')
    expected = []
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve('Ab', 'ab')
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(s, p string) []int {
	for _, c := range s + p {
		if c < 'a' || c > 'z' {
			panic("非法字符")
		}
	}
	ans := []int{}
	if len(p) == 0 || len(p) > len(s) {
		return ans
	}
	var want, window [26]int
	for i := range p {
		want[p[i]-'a']++
	}
	for i := range s {
		window[s[i]-'a']++
		if i >= len(p) {
			window[s[i-len(p)]-'a']--
		}
		if i >= len(p)-1 && window == want {
			ans = append(ans, i-len(p)+1)
		}
	}
	return ans
}

func main() {
	{ // 正常
		actual := solve("cbaebabacd", "abc")
		expected := []int{0, 6}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重叠匹配
		actual := solve("abab", "ab")
		expected := []int{0, 1, 2}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空输入
		actual := solve("", "a")
		expected := []int{}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空模式
		actual := solve("abc", "")
		expected := []int{}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve("Ab", "ab")
	}()
}
```

#include <array>
#include <cassert>
#include <iostream>
#include <stdexcept>
#include <string>
#include <vector>

using namespace std;
vector<int> solve(const string& s,const string& p) {
    for(char c:s+p)if(c<'a'||c>'z')throw invalid_argument("bad character");
    vector<int> ans;
    if(p.empty()||p.size()>s.size())return ans;
    array<int,26> want {},window {};
    for(char c:p)++want[c-'a'];
    for(size_t i=0;i<s.size();++i) {
        ++window[s[i]-'a'];
        if(i>=p.size())--window[s[i-p.size()]-'a'];
        if(i+1>=p.size()&&window==want)ans.push_back((int)(i+1-p.size()));
    }
    return ans;
}

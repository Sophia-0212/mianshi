# 0380. O(1) 时间插入、删除和获取随机元素

[LeetCode 原题](https://leetcode.com/problems/insert-delete-getrandom-o1/)

## 题意与思路

动态数组存值，哈希表存值→下标。删除时把末尾元素覆盖到待删位置，更新其下标后弹出末尾。随机选数组下标即可等概率返回元素。

插入平均/均摊 O(1)，删除、随机读取平均 O(1)，空间 O(n)。

## 输入约定与测试

操作 [0,x] 插入、[1,x] 删除、[2] 随机读取；插入/删除返回 1 或 0。空集合随机读取、非法操作拒绝。随机测试在多元素时检查成员合法性，在单元素时断言确定结果。

- 正常与重复：`[[[0, 1], [0, 1], [0, 2], [1, 1], [2], [1, 3]]]` → `[1, 0, 1, 1, 2, 0]`。
- 空操作：`[[]]` → `[]`。
- 删除后重建：`[[[0, 7], [2], [1, 7], [0, 9], [2]]]` → `[1, 7, 1, 1, 9]`。
- 空集合取随机值：`[[[2]]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
import random

def insert(values, positions, value):
    if value in positions:
        return False
    positions[value] = len(values)
    values.append(value)
    return True

def remove(values, positions, value):
    if value not in positions:
        return False
    index = positions[value]
    last = values[-1]
    values[index] = last
    positions[last] = index
    values.pop()
    del positions[value]
    return True

def getRandom(values):
    if not values:
        raise ValueError('空集合')
    return values[random.randrange(len(values))]

def solve(operations):
    values = []
    positions = {}
    answers = []
    for op in operations:
        if not op or op[0] not in (0, 1, 2) or len(op) != (1 if op[0] == 2 else 2):
            raise ValueError('非法操作')
        if op[0] == 0:
            answers.append(int(insert(values, positions, op[1])))
        elif op[0] == 1:
            answers.append(int(remove(values, positions, op[1])))
        else:
            answers.append(getRandom(values))
    probe = [10, 20, 30]
    assert all((getRandom(probe) in probe for _ in range(100)))
    return answers
if __name__ == '__main__':
    actual = solve([[0, 1], [0, 1], [0, 2], [1, 1], [2], [1, 3]])
    expected = [1, 0, 1, 1, 2, 0]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = []
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([[0, 7], [2], [1, 7], [0, 9], [2]])
    expected = [1, 7, 1, 1, 9]
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([[2]])
    except (ValueError, IndexError):
        print('用例 4：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"math/rand"
	"reflect"
)

func insert(values *[]int, positions map[int]int, value int) bool {
	if _, ok := positions[value]; ok {
		return false
	}
	positions[value] = len(*values)
	*values = append(*values, value)
	return true
}
func remove(values *[]int, positions map[int]int, value int) bool {
	index, ok := positions[value]
	if !ok {
		return false
	}
	a := *values
	last := a[len(a)-1]
	a[index] = last
	positions[last] = index
	*values = a[:len(a)-1]
	delete(positions, value)
	return true
}
func getRandom(values []int) int {
	if len(values) == 0 {
		panic("空集合")
	}
	return values[rand.Intn(len(values))]
}
func solve(operations [][]int) []int {
	values := []int{}
	positions := map[int]int{}
	ans := []int{}
	for _, op := range operations {
		if len(op) == 0 || op[0] < 0 || op[0] > 2 {
			panic("非法操作")
		}
		if (op[0] == 2 && len(op) != 1) || (op[0] != 2 && len(op) != 2) {
			panic("非法参数")
		}
		result := 0
		if op[0] == 0 {
			if insert(&values, positions, op[1]) {
				result = 1
			}
		} else if op[0] == 1 {
			if remove(&values, positions, op[1]) {
				result = 1
			}
		} else {
			result = getRandom(values)
		}
		ans = append(ans, result)
	}
	for i := 0; i < 100; i++ {
		v := getRandom([]int{10, 20, 30})
		if v != 10 && v != 20 && v != 30 {
			panic("随机值不在集合中")
		}
	}
	return ans
}

func main() {
	{ // 正常与重复
		actual := solve([][]int{[]int{0, 1}, []int{0, 1}, []int{0, 2}, []int{1, 1}, []int{2}, []int{1, 3}})
		expected := []int{1, 0, 1, 1, 2, 0}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空操作
		actual := solve([][]int{})
		expected := []int{}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 删除后重建
		actual := solve([][]int{[]int{0, 7}, []int{2}, []int{1, 7}, []int{0, 9}, []int{2}})
		expected := []int{1, 7, 1, 1, 9}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 4：已拒绝非法输入")
		}()
		solve([][]int{[]int{2}})
	}()
}
```

## C++

保存为 `main.cpp`，运行 `c++ -std=c++17 main.cpp -o main && ./main`。

```cpp
#include <algorithm>
#include <cassert>
#include <iostream>
#include <random>
#include <stdexcept>
#include <unordered_map>
#include <vector>

using namespace std;
bool insertValue(vector<int>& values,unordered_map<int,size_t>& positions,int value) {
    if(positions.count(value))return false;
    positions[value]=values.size();
    values.push_back(value);
    return true;
}
bool removeValue(vector<int>& values,unordered_map<int,size_t>& positions,int value) {
    auto it=positions.find(value);
    if(it==positions.end())return false;
    size_t index=it->second;
    int last=values.back();
    values[index]=last;
    positions[last]=index;
    values.pop_back();
    positions.erase(value);
    return true;
}
int getRandom(const vector<int>& values,mt19937& engine) {
    if(values.empty())throw invalid_argument("empty set");
    uniform_int_distribution<size_t> pick(0,values.size()-1);
    return values[pick(engine)];
}
vector<int> solve(const vector<vector<int>>& operations) {
    vector<int> values,ans;
    unordered_map<int,size_t> positions;
    mt19937 engine(random_device {}());
    for(auto& op:operations) {
        if(op.empty()||op[0]<0||op[0]>2||op.size()!=(op[0]==2?1U:2U))throw invalid_argument("bad operation");
        if(op[0]==0)ans.push_back(insertValue(values,positions,op[1]));
        else if(op[0]==1)ans.push_back(removeValue(values,positions,op[1]));
        else ans.push_back(getRandom(values,engine));
    }
    vector<int> probe {10,20,30};
    for(int i=0;i<100;++i) {
        int v=getRandom(probe,engine);
        assert(find(probe.begin(),probe.end(),v)!=probe.end());
    }
    return ans;
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常与重复
        auto actual = solve(vector<vector<int>> {
            vector<int> {0, 1}, vector<int> {0, 1}, vector<int> {0, 2}, vector<int> {1, 1}, vector<int> {2}, vector<int> {1, 3}
        });
        auto expected = vector<int> {1, 0, 1, 1, 2, 0};
        cout << "用例 1：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空操作
        auto actual = solve(vector<vector<int>> {});
        auto expected = vector<int> {};
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 删除后重建
        auto actual = solve(vector<vector<int>> {
            vector<int> {0, 7}, vector<int> {2}, vector<int> {1, 7}, vector<int> {0, 9}, vector<int> {2}
        });
        auto expected = vector<int> {1, 7, 1, 1, 9};
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        bool rejected=false;
        try {
            solve(vector<vector<int>> {
                vector<int> {2}
            });
        } catch(const invalid_argument&) {
            rejected=true;
        }
        assert(rejected);
        cout << "用例 4：已拒绝非法输入\n";
    }
}
```

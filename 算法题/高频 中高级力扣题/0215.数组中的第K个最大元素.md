# 0215. 数组中的第 K 个最大元素

[LeetCode 原题](https://leetcode.com/problems/kth-largest-element-in-an-array/)

## 题意与思路

维护大小不超过 k 的小顶堆。堆里保留最大的 k 个元素，堆顶就是第 k 大；重复数字分别计数。

时间 O(n log(k+1))，空间 O(k)。

## 输入约定与测试

k 必须在 [1,n]；空数组或越界 k 拒绝。

- 正常：`[[3, 2, 1, 5, 6, 4], 2]` → `5`。
- 重复：`[[3, 2, 3, 1, 2, 4, 5, 5, 6], 4]` → `4`。
- 单元素：`[[7], 1]` → `7`。
- 空输入：`[[], 1]` → `拒绝非法输入`。
- 非法 k：`[[1], 0]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
import heapq

def solve(nums, k):
    if not 1 <= k <= len(nums):
        raise ValueError('k 越界')
    heap = []
    for x in nums:
        heapq.heappush(heap, x)
        if len(heap) > k:
            heapq.heappop(heap)
    return heap[0]
if __name__ == '__main__':
    actual = solve([3, 2, 1, 5, 6, 4], 2)
    expected = 5
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([3, 2, 3, 1, 2, 4, 5, 5, 6], 4)
    expected = 4
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([7], 1)
    expected = 7
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([], 1)
    except (ValueError, IndexError):
        print('用例 4：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    try:
        solve([1], 0)
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"container/heap"
	"fmt"
	"reflect"
)

type MinHeap []int

func (h MinHeap) Len() int           { return len(h) }
func (h MinHeap) Less(i, j int) bool { return h[i] < h[j] }
func (h MinHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *MinHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *MinHeap) Pop() any          { a := *h; x := a[len(a)-1]; *h = a[:len(a)-1]; return x }
func solve(a []int, k int) int {
	if k < 1 || k > len(a) {
		panic("k 越界")
	}
	h := &MinHeap{}
	for _, x := range a {
		heap.Push(h, x)
		if h.Len() > k {
			heap.Pop(h)
		}
	}
	return (*h)[0]
}

func main() {
	{ // 正常
		actual := solve([]int{3, 2, 1, 5, 6, 4}, 2)
		expected := 5
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复
		actual := solve([]int{3, 2, 3, 1, 2, 4, 5, 5, 6}, 4)
		expected := 4
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单元素
		actual := solve([]int{7}, 1)
		expected := 7
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 4：已拒绝非法输入")
		}()
		solve([]int{}, 1)
	}()
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([]int{1}, 0)
	}()
}
```

## C++

保存为 `main.cpp`，运行 `c++ -std=c++17 main.cpp -o main && ./main`。

```cpp
#include <cassert>
#include <functional>
#include <iostream>
#include <queue>
#include <stdexcept>
#include <vector>

using namespace std;
int solve(const vector<int>& a,int k) {
    if(k<1||k>(int)a.size()) throw invalid_argument("invalid k");
    priority_queue<int,vector<int>,greater<int>> heap;
    for(int x:a) {
        heap.push(x);
        if((int)heap.size()>k)heap.pop();
    }
    return heap.top();
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常
        auto actual = solve(vector<int> {3, 2, 1, 5, 6, 4}, 2);
        auto expected = 5;
        cout << "用例 1：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 重复
        auto actual = solve(vector<int> {3, 2, 3, 1, 2, 4, 5, 5, 6}, 4);
        auto expected = 4;
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 单元素
        auto actual = solve(vector<int> {7}, 1);
        auto expected = 7;
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        bool rejected=false;
        try {
            solve(vector<int> {}, 1);
        } catch(const invalid_argument&) {
            rejected=true;
        }
        assert(rejected);
        cout << "用例 4：已拒绝非法输入\n";
    } {
        bool rejected=false;
        try {
            solve(vector<int> {1}, 0);
        } catch(const invalid_argument&) {
            rejected=true;
        }
        assert(rejected);
        cout << "用例 5：已拒绝非法输入\n";
    }
}
```

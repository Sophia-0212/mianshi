# OPT 页面置换（自编模拟）

**自编模拟题，无 LeetCode 题号。**

## 题意与思路

已知完整访问序列。命中不置换；缺页且容量已满时，淘汰下一次访问最晚的页面，不再访问视为最晚。本题返回总缺页数。

直观模拟：最坏 O(n²·capacity)，空间 O(capacity)。可预处理未来位置优化；本题优先展示规则。

## 输入约定与测试

输入为页面编号序列与容量。空序列缺页数为 0；容量 0 每次访问均缺页；负容量拒绝。

- 正常置换：`[[1, 2, 3, 1, 4, 1, 2], 3]` → `4`。
- 空序列：`[[], 3]` → `0`。
- 重复命中：`[[1, 1, 1], 1]` → `1`。
- 零容量：`[[1, 2], 0]` → `2`。
- 负容量：`[[1], -1]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(pages, capacity):
    if capacity < 0:
        raise ValueError('负容量')
    resident = set()
    faults = 0
    for (i, page) in enumerate(pages):
        if page in resident:
            continue
        faults += 1
        if capacity == 0:
            continue
        if len(resident) == capacity:

            def next_use(value):
                for j in range(i + 1, len(pages)):
                    if pages[j] == value:
                        return j
                return len(pages)
            resident.remove(max(resident, key=next_use))
        resident.add(page)
    return faults
if __name__ == '__main__':
    actual = solve([1, 2, 3, 1, 4, 1, 2], 3)
    expected = 4
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], 3)
    expected = 0
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, 1, 1], 1)
    expected = 1
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, 2], 0)
    expected = 2
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([1], -1)
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(pages []int, capacity int) int {
	if capacity < 0 {
		panic("负容量")
	}
	resident := map[int]bool{}
	faults := 0
	for i, page := range pages {
		if resident[page] {
			continue
		}
		faults++
		if capacity == 0 {
			continue
		}
		if len(resident) == capacity {
			victim, farthest := 0, -1
			for value := range resident {
				next := len(pages)
				for j := i + 1; j < len(pages); j++ {
					if pages[j] == value {
						next = j
						break
					}
				}
				if next > farthest {
					farthest = next
					victim = value
				}
			}
			delete(resident, victim)
		}
		resident[page] = true
	}
	return faults
}

func main() {
	{ // 正常置换
		actual := solve([]int{1, 2, 3, 1, 4, 1, 2}, 3)
		expected := 4
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空序列
		actual := solve([]int{}, 3)
		expected := 0
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复命中
		actual := solve([]int{1, 1, 1}, 1)
		expected := 1
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 零容量
		actual := solve([]int{1, 2}, 0)
		expected := 2
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([]int{1}, -1)
	}()
}
```

```cpp
#include <cassert>
#include <iostream>
#include <stdexcept>
#include <unordered_set>
#include <vector>

using namespace std;
int solve(const vector<int>& pages,int capacity) {
    if(capacity<0)throw invalid_argument("negative capacity");
    unordered_set<int> resident;
    int faults=0;
    for(int i=0;i<(int)pages.size();++i) {
        int page=pages[i];
        if(resident.count(page))continue;
        ++faults;
        if(!capacity)continue;
        if((int)resident.size()==capacity) {
            int victim=0,farthest=-1;
            for(int value:resident) {
                int next=pages.size();
                for(int j=i+1;j<(int)pages.size();++j)if(pages[j]==value) {
                    next=j;
                    break;
                }
                if(next>farthest) {
                    farthest=next;
                    victim=value;
                }
            }
            resident.erase(victim);
        }
        resident.insert(page);
    }
    return faults;
}
```

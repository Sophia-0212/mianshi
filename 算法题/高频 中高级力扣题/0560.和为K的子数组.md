# 0560. 和为 K 的子数组

[LeetCode 原题](https://leetcode.com/problems/subarray-sum-equals-k/)

## 题意与思路

前缀和：当前和为 sum，需要查此前有多少个前缀和等于 sum-k。先查询再记录，避免把空子数组算入。初始计数 0→1。

平均时间 O(n)，空间 O(n)。

## 输入约定与测试

允许负数，因此不能直接用普通滑动窗口。空数组返回 0；Go/C++ 使用 64 位前缀和与结果。

- 正常：`[[1, 1, 1], 2]` → `2`。
- 负数：`[[1, -1, 0], 0]` → `3`。
- 重复前缀：`[[0, 0, 0], 0]` → `6`。
- 空数组：`[[], 0]` → `0`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(nums, k):
    counts = {0: 1}
    total = answer = 0
    for x in nums:
        total += x
        answer += counts.get(total - k, 0)
        counts[total] = counts.get(total, 0) + 1
    return answer
if __name__ == '__main__':
    actual = solve([1, 1, 1], 2)
    expected = 2
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, -1, 0], 0)
    expected = 3
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([0, 0, 0], 0)
    expected = 6
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], 0)
    expected = 0
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(a []int, k int) int64 {
	counts := map[int64]int64{0: 1}
	var sum, answer int64
	for _, x := range a {
		sum += int64(x)
		answer += counts[sum-int64(k)]
		counts[sum]++
	}
	return answer
}

func main() {
	{ // 正常
		actual := solve([]int{1, 1, 1}, 2)
		expected := int64(2)
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 负数
		actual := solve([]int{1, -1, 0}, 0)
		expected := int64(3)
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 重复前缀
		actual := solve([]int{0, 0, 0}, 0)
		expected := int64(6)
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空数组
		actual := solve([]int{}, 0)
		expected := int64(0)
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
}
```

## C++

保存为 `main.cpp`，运行 `c++ -std=c++17 main.cpp -o main && ./main`。

```cpp
#include <cassert>
#include <iostream>
#include <unordered_map>
#include <vector>

using namespace std;
long long solve(const vector<int>& a,int k) {
    unordered_map<long long,long long> counts {
        {0,1}
    };
    long long sum=0,answer=0;
    for(int x:a) {
        sum+=x;
        answer+=counts[sum-k];
        ++counts[sum];
    }
    return answer;
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常
        auto actual = solve(vector<int> {1, 1, 1}, 2);
        auto expected = 2;

        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);

        // 负数
        auto actual = solve(vector<int> {1, -1, 0}, 0);
        auto expected = 3;
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 重复前缀
        auto actual = solve(vector<int> {0, 0, 0}, 0);
        auto expected = 6;
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空数组
        auto actual = solve(vector<int> {}, 0);

        cout << "用例 4：期望 ";

        show(actual);
        cout << "\n";
        assert(actual == expected);
    }
}
```

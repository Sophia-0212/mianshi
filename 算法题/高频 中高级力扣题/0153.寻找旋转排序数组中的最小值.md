# 0153. 寻找旋转排序数组中的最小值

[LeetCode 原题](https://leetcode.com/problems/find-minimum-in-rotated-sorted-array/)

## 题意与思路

比较中点和右端：中点较大则最小值在右侧；否则最小值在左半区，且可能就是中点。

时间 O(log n)，空间 O(1)。

## 输入约定与测试

数组由严格升序数组旋转得到，不含重复。不扫描验证排序前提；空数组拒绝。

- 正常：`[[3, 4, 5, 1, 2]]` → `1`。
- 单元素：`[[1]]` → `1`。
- 未旋转：`[[1, 2, 3]]` → `1`。
- 空数组：`[[]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(a):
    if not a:
        raise ValueError('数组不能为空')
    (l, r) = (0, len(a) - 1)
    while l < r:
        m = (l + r) // 2
        if a[m] > a[r]:
            l = m + 1
        else:
            r = m
    return a[l]
if __name__ == '__main__':
    actual = solve([3, 4, 5, 1, 2])
    expected = 1
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1])
    expected = 1
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([1, 2, 3])
    expected = 1
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([])
    except (ValueError, IndexError):
        print('用例 4：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(a []int) int {
	if len(a) == 0 {
		panic("数组不能为空")
	}
	l, r := 0, len(a)-1
	for l < r {
		m := l + (r-l)/2
		if a[m] > a[r] {
			l = m + 1
		} else {
			r = m
		}
	}
	return a[l]
}

func main() {
	{ // 正常
		actual := solve([]int{3, 4, 5, 1, 2})
		expected := 1
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单元素
		actual := solve([]int{1})
		expected := 1
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 未旋转
		actual := solve([]int{1, 2, 3})
		expected := 1
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 4：已拒绝非法输入")
		}()
		solve([]int{})
	}()
}
```

#include <cassert>
#include <iostream>
#include <stdexcept>
#include <vector>

using namespace std;
int solve(const vector<int>& a) {
    if(a.empty()) throw invalid_argument("empty input");
    int l=0,r=(int)a.size()-1;
    while(l<r) {
        int m=l+(r-l)/2;
        if(a[m]>a[r]) l=m+1;
        else r=m;
    }
    return a[l];
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常
        auto actual = solve(vector<int> {3, 4, 5, 1, 2});

        cout << " 实际 ";

        cout << "\n";

# 0011. 盛最多水的容器

[LeetCode 原题](https://leetcode.com/problems/container-with-most-water/)

## 题意与思路

双指针从两端向内。面积由短板决定；移动长板不可能在宽度变小时提高面积，因此移动短板。

时间 O(n)，空间 O(1)。

## 输入约定与测试

高度必须非负；少于两根柱子返回 0。Go/C++ 用 64 位面积。

- 正常：`[[1, 8, 6, 2, 5, 4, 8, 3, 7]]` → `49`。
- 空输入：`[[]]` → `0`。
- 单柱：`[[7]]` → `0`。
- 零高度：`[[0, 0]]` → `0`。
- 非法高度：`[[-1, 2]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(height):
    if any((x < 0 for x in height)):
        raise ValueError('高度不能为负')
    (left, right, best) = (0, len(height) - 1, 0)
    while left < right:
        best = max(best, (right - left) * min(height[left], height[right]))
        if height[left] <= height[right]:
            left += 1
        else:
            right -= 1
    return best
if __name__ == '__main__':
    actual = solve([1, 8, 6, 2, 5, 4, 8, 3, 7])
    expected = 49
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = 0
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([7])
    expected = 0
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([0, 0])
    expected = 0
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([-1, 2])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

## Go

保存为 `main.go`，运行 `go run main.go`。

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(h []int) int64 {
	for _, x := range h {
		if x < 0 {
			panic("高度不能为负")
		}
	}
	l, r := 0, len(h)-1
	var best int64
	for l < r {
		short := h[l]
		if h[r] < short {
			short = h[r]
		}
		area := int64(r-l) * int64(short)
		if area > best {
			best = area
		}
		if h[l] <= h[r] {
			l++
		} else {
			r--
		}
	}
	return best
}

func main() {
	{ // 正常
		actual := solve([]int{1, 8, 6, 2, 5, 4, 8, 3, 7})
		expected := int64(49)
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空输入
		actual := solve([]int{})
		expected := int64(0)
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 单柱
		actual := solve([]int{7})
		expected := int64(0)
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 零高度
		actual := solve([]int{0, 0})
		expected := int64(0)
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([]int{-1, 2})
	}()
}
```

## C++

```cpp
#include <algorithm>
#include <cassert>
#include <iostream>
#include <stdexcept>
#include <vector>

using namespace std;
long long solve(const vector<int>& h) {
    for(int x:h) if(x<0) throw invalid_argument("negative height");
    int l=0,r=(int)h.size()-1;
    long long best=0;
    while(l<r) {
        best=max(best,1LL*(r-l)*min(h[l],h[r]));
        if(h[l]<=h[r]) ++l;
        else --r;
    }
    return best;
}
template<class T> void show(const vector<T>& xs);
template<class T> void show(const T& x) {
    cout << boolalpha << x;
}
template<class T> void show(const vector<T>& xs) {
    cout << "[";
    for (size_t i=0; i<xs.size(); ++i) {
        if(i) cout << ", ";
        show(xs[i]);
    }
    cout << "]";
}
int main() {
    {
        // 正常
        auto actual = solve(vector<int> {1, 8, 6, 2, 5, 4, 8, 3, 7});
        auto expected = 49;
        cout << "用例 1：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 空输入
        auto actual = solve(vector<int> {});
        auto expected = 0;
        cout << "用例 2：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 单柱
        auto actual = solve(vector<int> {7});
        auto expected = 0;
        cout << "用例 3：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        // 零高度
        auto actual = solve(vector<int> {0, 0});
        auto expected = 0;
        cout << "用例 4：期望 ";
        show(expected);
        cout << " 实际 ";
        show(actual);
        cout << "\n";
        assert(actual == expected);
    } {
        bool rejected=false;
        try {
            solve(vector<int> {-1, 2});
        } catch(const invalid_argument&) {
            rejected=true;
        }
        assert(rejected);
        cout << "用例 5：已拒绝非法输入\n";
    }
}
```

# 0036. 有效的数独

[LeetCode 原题](https://leetcode.com/problems/valid-sudoku/)

## 题意与思路

分别记录每行、每列、每个 3×3 宫已出现的数字。宫编号为 `(行/3)*3+列/3`。只判断已有数字是否冲突，不求解数独。

固定 9×9 棋盘：时间、空间均 O(1)。

## 输入约定与测试

棋盘用 9 个字符串表示，`.` 表示空格。尺寸或字符非法返回 false；全空棋盘有效。

- 正常：`[['53..7....', '6..195...', '.98....6.', '8...6...3', '4..8.3..1', '7...2...6', '.6....28.', '...419..5', '....8..79']]` → `True`。
- 重复：`[['55.......', '.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........']]` → `False`。
- 全空棋盘：`[['.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........']]` → `True`。
- 空输入：`[[]]` → `False`。
- 非法字符：`[['x........', '.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........']]` → `False`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(board):
    if len(board) != 9 or any((len(row) != 9 for row in board)):
        return False
    (rows, cols, boxes) = ([set() for _ in range(9)], [set() for _ in range(9)], [set() for _ in range(9)])
    for i in range(9):
        for j in range(9):
            c = board[i][j]
            if c == '.':
                continue
            if c not in '123456789':
                return False
            b = i // 3 * 3 + j // 3
            if c in rows[i] or c in cols[j] or c in boxes[b]:
                return False
            rows[i].add(c)
            cols[j].add(c)
            boxes[b].add(c)
    return True
if __name__ == '__main__':
    actual = solve(['53..7....', '6..195...', '.98....6.', '8...6...3', '4..8.3..1', '7...2...6', '.6....28.', '...419..5', '....8..79'])
    expected = True
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(['55.......', '.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........'])
    expected = False
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(['.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........'])
    expected = True
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([])
    expected = False
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve(['x........', '.........', '.........', '.........', '.........', '.........', '.........', '.........', '.........'])
```

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(board []string) bool {
	if len(board) != 9 {
		return false
	}
	for _, row := range board {
		if len(row) != 9 {
			return false
		}
	}
	var rows, cols, boxes [9][9]bool
	for i := 0; i < 9; i++ {
		for j := 0; j < 9; j++ {
			c := board[i][j]
			if c == '.' {
				continue
			}
			if c < '1' || c > '9' {
				return false
			}
			d, b := int(c-'1'), (i/3)*3+j/3
			if rows[i][d] || cols[j][d] || boxes[b][d] {
				return false
			}
			rows[i][d], cols[j][d], boxes[b][d] = true, true, true
		}
	}
	return true
}

func main() {
	{ // 正常
		actual := solve([]string{"53..7....", "6..195...", ".98....6.", "8...6...3", "4..8.3..1", "7...2...6", ".6....28.", "...419..5", "....8..79"})

		}

	{ // 重复
```

```cpp
    if(board.size()!=9) return false;
    for(auto& row:board) if(row.size()!=9) return false;
    bool rows[9][9]= {},cols[9][9]= {},boxes[9][9]= {};
    for(int i=0;i<9;++i) for(int j=0;j<9;++j) {
```

# 0057. 插入区间

[LeetCode 原题](https://leetcode.com/problems/insert-interval/)

## 题意与思路

先收集新区间左边的区间，再合并所有重叠区间，最后追加右边区间。

时间 O(n)，输出空间 O(n)。

## 输入约定与测试

原区间必须按起点排序且互不重叠；端点相等算重叠。非法区间或顺序拒绝；不修改输入。

- 正常：`[[[1, 3], [6, 9]], [2, 5]]` → `[[1, 5], [6, 9]]`。
- 空数组：`[[], [2, 5]]` → `[[2, 5]]`。
- 包含：`[[[1, 5]], [2, 3]]` → `[[1, 5]]`。
- 端点重叠：`[[[1, 2], [5, 7]], [2, 5]]` → `[[1, 7]]`。
- 非法区间：`[[], [5, 2]]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(intervals, new):
    if len(new) != 2 or new[0] > new[1]:
        raise ValueError('非法区间')
    for (i, p) in enumerate(intervals):
        if len(p) != 2 or p[0] > p[1] or (i and intervals[i - 1][1] >= p[0]):
            raise ValueError('原区间无序或重叠')
    (left, right) = new
    ans = []
    i = 0
    while i < len(intervals) and intervals[i][1] < left:
        ans.append(intervals[i][:])
        i += 1
    while i < len(intervals) and intervals[i][0] <= right:
        left = min(left, intervals[i][0])
        right = max(right, intervals[i][1])
        i += 1
    ans.append([left, right])
    ans.extend((p[:] for p in intervals[i:]))
    return ans
if __name__ == '__main__':
    actual = solve([[1, 3], [6, 9]], [2, 5])
    expected = [[1, 5], [6, 9]]
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([], [2, 5])
    expected = [[2, 5]]
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([[1, 5]], [2, 3])
    expected = [[1, 5]]
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve([[1, 2], [5, 7]], [2, 5])
    expected = [[1, 7]]
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve([], [5, 2])
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

```go
package main

import (
	"fmt"
	"reflect"
)

func solve(a [][]int, new []int) [][]int {
	if len(new) != 2 || new[0] > new[1] {
		panic("非法区间")
	}
	for i, p := range a {
		if len(p) != 2 || p[0] > p[1] || (i > 0 && a[i-1][1] >= p[0]) {
			panic("原区间无序或重叠")
		}
	}
	l, r, i := new[0], new[1], 0
	ans := [][]int{}
	for i < len(a) && a[i][1] < l {
		ans = append(ans, []int{a[i][0], a[i][1]})
		i++
	}
	for i < len(a) && a[i][0] <= r {
		if a[i][0] < l {
			l = a[i][0]
		}
		if a[i][1] > r {
			r = a[i][1]
		}
		i++
	}
	ans = append(ans, []int{l, r})
	for ; i < len(a); i++ {
		ans = append(ans, []int{a[i][0], a[i][1]})
	}
	return ans
}

func main() {
	{ // 正常
		actual := solve([][]int{[]int{1, 3}, []int{6, 9}}, []int{2, 5})
		expected := [][]int{[]int{1, 5}, []int{6, 9}}
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空数组
		actual := solve([][]int{}, []int{2, 5})
		expected := [][]int{[]int{2, 5}}
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 包含
		actual := solve([][]int{[]int{1, 5}}, []int{2, 3})
		expected := [][]int{[]int{1, 5}}
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 端点重叠
		actual := solve([][]int{[]int{1, 2}, []int{5, 7}}, []int{2, 5})
		expected := [][]int{[]int{1, 7}}
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve([][]int{}, []int{5, 2})
	}()
}
```

```cpp
vector<vector<int>> solve(const vector<vector<int>>& a,const vector<int>& fresh) {
    if(fresh.size()!=2||fresh[0]>fresh[1]) throw invalid_argument("bad interval");
    for(size_t i=0;i<a.size();++i) if(a[i].size()!=2||a[i][0]>a[i][1]||(i&&a[i-1][1]>=a[i][0])) throw invalid_argument("bad input");
    int l=fresh[0],r=fresh[1];
    size_t i=0;
    vector<vector<int>> ans;
    while(i<a.size()&&a[i][1]<l) ans.push_back(a[i++]);
    while(i<a.size()&&a[i][0]<=r) {
        l=min(l,a[i][0]);
        r=max(r,a[i][1]);
        ++i;
    }
    ans.push_back( {l,r});
    while(i<a.size()) ans.push_back(a[i++]);
    return ans;
}
```

# 0424. 替换后的最长重复字符

[LeetCode 原题](https://leetcode.com/problems/longest-repeating-character-replacement/)

## 题意与思路

滑动窗口统计字符数量。窗口长度减去窗口内最多字符数，就是需要替换的次数；超过 k 就收缩。本实现每次扫描固定 26 个计数，保持判断直观。

时间 O(26n)=O(n)，空间 O(26)。

## 输入约定与测试

仅允许 A～Z；k 非负；空串返回 0，非法字符或负 k 拒绝。

- 正常：`['AABABBA', 1]` → `4`。
- 足够替换：`['ABAB', 2]` → `4`。
- 空串：`['', 0]` → `0`。
- 不允许替换：`['ABCDE', 0]` → `1`。
- 非法字符：`['abc', 1]` → `拒绝非法输入`。
- 非法 k：`['A', -1]` → `拒绝非法输入`。

三种实现均为普通函数，不依赖 `Solution`。节点或状态结构仅保存数据；`solve` 是可直接调用的入口。每段包含独立入口与断言。

## Python

保存为 `main.py`，运行 `python3 main.py`。

```python
def solve(s, k):
    if k < 0 or any((not 'A' <= c <= 'Z' for c in s)):
        raise ValueError('非法输入')
    counts = [0] * 26
    left = best = 0
    for (right, c) in enumerate(s):
        counts[ord(c) - 65] += 1
        while right - left + 1 - max(counts) > k:
            counts[ord(s[left]) - 65] -= 1
            left += 1
        best = max(best, right - left + 1)
    return best
if __name__ == '__main__':
    actual = solve('AABABBA', 1)
    expected = 4
    print('用例 1：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('ABAB', 2)
    expected = 4
    print('用例 2：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('', 0)
    expected = 0
    print('用例 3：期望', expected, '实际', actual)
    assert actual == expected
    actual = solve('ABCDE', 0)
    expected = 1
    print('用例 4：期望', expected, '实际', actual)
    assert actual == expected
    try:
        solve('abc', 1)
    except (ValueError, IndexError):
        print('用例 5：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
    try:
        solve('A', -1)
    except (ValueError, IndexError):
        print('用例 6：已拒绝非法输入')
    else:
        raise AssertionError('应拒绝非法输入')
```

package main

import (
	"fmt"
	"reflect"
)

func solve(s string, k int) int {
	if k < 0 {
		panic("非法 k")
	}
	for _, c := range s {
		if c < 'A' || c > 'Z' {
			panic("非法字符")
		}
	}
	var counts [26]int
	left, best := 0, 0
	maxCount := func() int {
		m := 0
		for _, n := range counts {
			if n > m {
				m = n
			}
		}
		return m
	}
	for right := 0; right < len(s); right++ {
		counts[s[right]-'A']++
		for right-left+1-maxCount() > k {
			counts[s[left]-'A']--
			left++
		}
		if size := right - left + 1; size > best {
			best = size
		}
	}
	return best
}

func main() {
	{ // 正常
		actual := solve("AABABBA", 1)
		expected := 4
		fmt.Printf("用例 1：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 足够替换
		actual := solve("ABAB", 2)
		expected := 4
		fmt.Printf("用例 2：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 空串
		actual := solve("", 0)
		expected := 0
		fmt.Printf("用例 3：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	{ // 不允许替换
		actual := solve("ABCDE", 0)
		expected := 1
		fmt.Printf("用例 4：期望 %v 实际 %v\n", expected, actual)
		if !reflect.DeepEqual(actual, expected) {
			panic("结果不符")
		}
	}
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")
			}
			fmt.Println("用例 5：已拒绝非法输入")
		}()
		solve("abc", 1)
	}()
	func() {
		defer func() {
			if recover() == nil {
				panic("应拒绝非法输入")

	}()
}

int solve(const string& s,int k) {
    if(k<0) throw invalid_argument("negative k");
    for(char c:s)if(c<'A'||c>'Z')throw invalid_argument("bad character");
    array<int,26> count {};
    int left=0,best=0;
    for(int right=0;right<(int)s.size();++right) {
        ++count[s[right]-'A'];
        while(right-left+1-*max_element(count.begin(),count.end())>k) --count[s[left++]-'A'];
        best=max(best,right-left+1);
    }
    return best;
}
